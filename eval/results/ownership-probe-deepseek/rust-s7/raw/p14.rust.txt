 ```rust
fn total(v: Vec<i32>, flag: bool) -> usize {
    if flag {
        let w = v.clone();
        return w.len() * 10;
    }
    let mut u = v.clone();
    u.push(3);
    u.len() * 20
}

fn main() {
    println!("{}", total(vec![1, 2], true));
    println!("{}", total(vec![1, 2], false));
}
```
