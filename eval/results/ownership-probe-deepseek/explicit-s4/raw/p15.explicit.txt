fn f(o: Option<Vec<Int>>) -> Option<Int> {
    let v = o?
    let r = len(&v) * 10 + len(&v)
    drop v
    Some(r)
}

fn main() {
    match f(Some(push(push(vec(), 1), 2))) {
        Some(n) => print(n),
        None => print(0),
    }
    match f(None) {
        Some(n) => print(n),
        None => print(99),
    }
}
