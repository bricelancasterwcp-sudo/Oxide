fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in 2.to_int().to_float().sqrt().trunc() + 1.to_float().to_int() {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let mut count = 0
    for i in 0.to_int().to_float().sqrt().trunc() + 1.to_float().to_int() {
        if is_prime(i) {
            count = count + 1
        }
    }
    print(count)
}
