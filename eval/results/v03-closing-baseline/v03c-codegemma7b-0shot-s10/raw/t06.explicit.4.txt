fn is_perfect(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    let mut sum = 1
    for i in 2.range(n) {
        if n % i == 0 {
            sum = sum + i
        }
    }
    return sum == n
}

fn main() {
    for i in 0.range(1000) {
        if is_perfect(i) {
            print(i)
            print("\n")
        }
    }
}
