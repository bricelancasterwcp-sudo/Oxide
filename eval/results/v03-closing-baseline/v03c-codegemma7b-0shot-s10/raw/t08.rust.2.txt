fn main() {
    let mut values = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut largest_value = std::i32::MIN;

    for value in values {
        if value > 0 {
            positive_count += 1;
        }
        if value > largest_value {
            largest_value = value;
        }
    }

    println!("Positive count: {}", positive_count);
    println!("Largest value: {}", largest_value);
}
