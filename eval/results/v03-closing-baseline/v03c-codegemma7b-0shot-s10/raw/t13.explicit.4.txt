fn is_palindrome(text: &Str) -> Bool {
    let mut i = 0
    let mut j = text.len() - 1
    while i < j {
        if text.chars().nth(i) != text.chars().nth(j) {
            return false
        }
        i = i + 1
        j = j - 1
    }
    true
}

fn main() {
    let level = "level"
    let stereo = "stereo"
    let rotor = "rotor"
    print(is_palindrome(&level))
    print(is_palindrome(&stereo))
    print(is_palindrome(&rotor))
}
