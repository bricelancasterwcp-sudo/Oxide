fn perimeter(l: Float, w: Float) -> Float {
    2.0 * (l + w)
}

fn main() {
    let r1 = Rectangle { length: 3.0, width: 4.0 }
    let t1 = Triangle { side1: 3.0, side2: 4.0, side3: 5.0 }
    let r2 = Rectangle { length: 2.0, width: 5.0 }
    print(perimeter(r1.length, r1.width))
    print(perimeter(t1.side1, t1.side2))
    print(perimeter(r2.length, r2.width))
}

fn print(v: &Vec<Int>) {
    print("[")
    for i in 0.upto(v.len()) {
        print(format("{}", v.get(i).unwrap()))
        if i < v.len() - 1 {
            print(", ")
        }
    }
    print("]")
}
