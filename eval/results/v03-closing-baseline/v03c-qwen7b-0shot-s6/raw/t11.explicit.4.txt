fn sort(v: Vec<Int>) -> Vec<Int> {
    for i in 0.range(v.len()) {
        for j in (i + 1).range(v.len()) {
            if get(v, i) > get(v, j) {
                let temp = get(v, i).unwrap()
                let val_j = get(v, j).unwrap()
                let temp_idx = i
                let val_j_idx = j
                drop temp
                drop val_j
                set(v, temp_idx, val_j)
                set(v, val_j_idx, temp)
            }
        }
    }
    v
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(clone(&nums))
    for x in &sorted {
        print_str(&x)
    }
    drop nums
    drop sorted
}

    let mut v = v
    v.set(index, value)
    v
