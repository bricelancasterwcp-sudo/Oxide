```rust
fn main() {
    let rectangle1 = Rectangle { width: 3, height: 4 };
    let triangle = Triangle { side1: 3, side2: 4, side3: 5 };
    let rectangle2 = Rectangle { width: 2, height: 5 };

    println!("Perimeter of rectangle 1: {}", rectangle1.perimeter());
    println!("Perimeter of triangle: {}", triangle.perimeter());
    println!("Perimeter of rectangle 2: {}", rectangle2.perimeter());
}

struct Rectangle {
    width: i32,
    height: i32,
}

impl Rectangle {
    fn perimeter(&self) -> i32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    side1: i32,
    side2: i32,
    side3: i32,
}

impl Triangle {
    fn perimeter(&self) -> i32 {
        self.side1 + self.side2 + self.side3
    }
}
```
