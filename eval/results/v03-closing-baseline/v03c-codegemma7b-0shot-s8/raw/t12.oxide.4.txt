fn reverse(s: Str) -> Str {
    let mut rev = vec()
    for c in chars(s) {
        rev = push(rev, c)
    }
    str_join(rev.as_slice())
}

fn main() {
    print(reverse("stack"))
}

fn str_join(s: Slice<Str>) -> Str {
    let mut out = Str("")
    for i in 0.upto(len(s) - 1) {
        out = concat(out, s.get(i).unwrap())
    }
    out
}

fn chars(s: Str) -> Vec<Str> {
    let mut c = vec()
    for i in 0.upto(str_len(s) - 1) {
        c = push(c, str_get(s, i).unwrap())
    }
    c
}

fn str_get(s: Str, i: Int) -> Option<Str> {
    if i >= 0 && i < str_len(s) {
        Some(Str(s.chars().nth(i).unwrap()))
    } else {
        None
    }
}

fn len(v: Vec<T>) -> Int {
    let mut c = 0
    for _ in v {
        c = c + 1
    }
    c
}

fn str_len(s: Str) -> Int {
    let mut c = 0
    for _ in s.chars() {
        c = c + 1
    }
    c
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let mut c = v
    c.push(x)
    c
}

fn concat(a: Str, b: Str) -> Str {
    let mut c = String(a.chars().chain(b.chars()))
    Str(c)
}
