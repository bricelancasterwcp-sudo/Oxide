    let mut ok = 0
    let mut fail = 0
    for s in vec_str("12", "x", "30") {
        match parse_int(s) {
            Some(n) => ok == ok + n,
            None => fail == fail + 1,
        }
    }
    print(ok)
    print(fail)
}

    if n >= 0 && n <= 9 {
        Ok(n)
    } else {
        Err("not a digit".to_string())
    }
}
