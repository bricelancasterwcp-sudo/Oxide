fn area(l: Int, w: Int) -> Int {
    l * w
}

fn perimeter(l: Int, w: Int) -> Int {
    2 * l + 2 * w
}

fn main() {
    print(perimeter(3, 4))
    print(perimeter(3, 4, 5))
    print(perimeter(2, 5))
}
