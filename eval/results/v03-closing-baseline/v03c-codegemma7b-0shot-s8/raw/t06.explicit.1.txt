fn divisors(n: Int) -> Vec<Int> {
    let mut factors = vec()
    for i in 2.to_int().unwrap() + 1.to_int().unwrap() {
        if n % i == 0 {
            factors.push(i)
            if i != n / i {
                factors.push(n / i)
            }
        }
    }
    factors
}

fn is_perfect(n: Int) -> Bool {
    if n == 0 {
        return false
    }
    let factors = divisors(n)
    let sum = factors.iter().fold(0, Int || acc, x || acc + x)
    return sum == n
}

fn main() {
    for n in 0.to_int().unwrap() + 1.to_int().unwrap() {
        if is_perfect(n) {
            print(n)
        }
    }
}
