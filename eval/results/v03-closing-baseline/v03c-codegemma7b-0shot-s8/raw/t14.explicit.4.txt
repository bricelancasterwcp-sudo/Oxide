fn count_s(s: &Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if c == "s" {
            count = count + 1
        }
    }
    count
}

fn count_distinct(s: &Str) -> Int {
    let mut seen: Vec<Str> = vec()
    for c in chars(s) {
        if !contains(&seen, &c) {
            seen = push(seen, c)
        }
    }
    len(&seen)
}

fn contains(v: &Vec<Str>, x: &Str) -> Bool {
    for i in 0 < len(v) {
        if v.get(i) == x {
            return true
        }
    }
    false
}

fn chars(s: &Str) -> Vec<Str> {
    let mut result: Vec<Str> = vec()
    for i in 0 < str_len(s) {
        result = push(result, s.get(i))
    }
    result
}

fn len(v: &Vec<Str>) -> Int {
    let mut count = 0
    for _ in v {
        count = count + 1
    }
    count
}

fn push(v: Vec<Str>, x: &Str) -> Vec<Str> {
    let mut result = v
    result.append(x)
    result
}

fn get(v: &Vec<Str>, i: Int) -> Option<Str> {
    if i < len(v) {
        Some(v.get(i))
    } else {
        None
    }
}

fn trunc(x: Float) -> Int {
    if x < 0 {

}
