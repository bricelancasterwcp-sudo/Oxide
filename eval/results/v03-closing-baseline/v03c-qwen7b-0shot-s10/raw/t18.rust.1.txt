```rust
fn main() {
    let rectangle1 = Rectangle { width: 3, height: 4 };
    let triangle = Triangle { a: 3, b: 4, c: 5 };
    let rectangle2 = Rectangle { width: 2, height: 5 };

    println!("Perimeter of rectangle1: {}", rectangle1.perimeter());
    println!("Perimeter of triangle: {}", triangle.perimeter());
    println!("Perimeter of rectangle2: {}", rectangle2.perimeter());
}

struct Rectangle {
    width: i32,
    height: i32,
}

impl Rectangle {
    fn perimeter(&self) -> i32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    a: i32,
    b: i32,
    c: i32,
}

impl Triangle {
    fn perimeter(&self) -> i32 {
        self.a + self.b + self.c
    }
}
```
