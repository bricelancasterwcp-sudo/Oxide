fn sort(v: &Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    for x in v {
        let mut found = false
        for i in 0.len() < result.len() {
            if result.get(i) > Some(x) {
                result.insert(i, x)
                found = true
                break
            }
        }
        if !found {
            result.push(x)
        }
    }
    result
}

    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(&nums)
    for x in &sorted {
        print_str(x)
    }
    drop nums
    drop sorted
}
