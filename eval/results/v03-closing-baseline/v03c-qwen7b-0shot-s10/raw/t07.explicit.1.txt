fn sum_multiples() -> Int {
    let mut sum = 0
    let mut count = 0
    let mut multiple = 3
    while count < 8 {
        sum = sum + multiple
        count = count + 1
        multiple = multiple + 3
    }
    sum
}

fn main() {
    let multiples_sum = sum_multiples()
    print(multiples_sum)
}
