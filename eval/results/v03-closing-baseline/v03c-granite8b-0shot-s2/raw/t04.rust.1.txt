```rust
const N: u32 = 100;

fn main() {
    let mut count = 0;
    for i in 2..N {
        if is_prime(i) {
            count += 1;
        }
    }
    println!("{}", count);
}

fn is_prime(n: u32) -> bool {
    if n == 2 {
        return true;
    }
    if n % 2 == 0 {
        return false;
    }
    for i in 3..(n as f64).sqrt().ceil() as u32 {
        if n % i == 0 {
            return false;
        }
    }
    true
}
```
