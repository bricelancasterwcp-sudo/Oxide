fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return False
    }
    for i in range(2, trunc(to_float(n).sqrt()) + 1) {
        if n % i == 0 {
            return False
        }
    }
    True
}

fn count_primes() -> Int {
    let mut count = 0
    for i in range(2, 100) {
        if is_prime(i) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes())
}
