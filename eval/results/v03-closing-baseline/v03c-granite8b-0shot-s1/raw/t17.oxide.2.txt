struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < len(v) {
        Some(get_unsafe(v, i))
    } else {
        None
    }
}

fn get_unsafe(v: Vec<Int>, i: Int) -> Int {
    clone(get_ref(v, i))
}

fn get_ref(v: Vec<Int>, i: Int) -> Ref<Int> {
    get_ref_at(v, i)
}

fn get_ref_at(v: Vec<Int>, i: Int) -> Ref<Int> {
    v + i
}

fn main() {
    let v = push(push(vec(), 10), 20)
    print(get(v, 0)?.clone())
    print(get(v, 2)?.clone())
}
