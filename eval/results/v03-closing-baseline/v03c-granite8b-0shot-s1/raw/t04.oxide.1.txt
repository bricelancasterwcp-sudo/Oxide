fn is_prime(n: Int) -> Bool {
    if n <= 3 {
        return true
    } else if n % 2 == 0 || n % 3 == 0 {
        return false
    }
    let mut i = 5
    while i * i <= n {
        if n % i == 0 || n % (i + 2) == 0 {
            return false
        }
        i = i + 6
    }
    true
}

fn count_primes(n: Int) -> Int {
    let mut count = 0
    for i in range(2, n) {
        if is_prime(i) {
            count = count + 1
        }
    }
    count
}

    print(count_primes(99))
}
