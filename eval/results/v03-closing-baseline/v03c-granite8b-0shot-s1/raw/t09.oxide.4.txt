struct IntList { head: Int, tail: Vec<Int> }

fn reverse(input: Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    for x in input {
        result = push(result, x)
    }
    result
}

fn main() {
    let input = vec(1, 2, 3, 4, 5)
    let mut result = vec()
    for x in input {
        result = push(result, x)
    }
    for x in reverse(input) {
        print(x)
    }
}
