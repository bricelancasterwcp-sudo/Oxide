fn print_int(n: Int) {
    print(int_to_str(n))
}

fn main() {
    let rect = Rectangle { width: 3, height: 4 }
    let triangle = Triangle { side1: 3, side2: 4, side3: 5 }
    let square = Rectangle { width: 2, height: 5 }
    print_int(perimeter(&rect))
    print_int(perimeter(&triangle))
    print_int(perimeter(&square))
}

struct Rectangle {
    width: Int,
    height: Int,
}

struct Triangle {
    side1: Int,
    side2: Int,
    side3: Int,
}

fn perimeter(shape: &Shape) -> Int {
    match shape {
        Shape(r) => 2 * (r.width + r.height),
        Triangle(t) => t.side1 + t.side2 + t.side3,
    }
}

enum Shape {
    Rectangle(Rectangle),
    Triangle(Triangle),
}

fn area(shape: &Shape) -> Int {
    match shape {
        Shape(r) => r.width * r.height,
        Triangle(t) => {
            let s = (t.side1 + t.side2 + t.side3) / 2
            s * (s - t.side1) * (s - t.side2) * (s - t.side3)
        },
    }
}

enum TriangleSides {
    Side1(Int),
    Side2(Int),
    Side3(Int),
}

    Rectangle(Rectangle),
    Triangle(TriangleSides, TriangleSides, TriangleSides),
}
