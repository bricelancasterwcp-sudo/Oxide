fn task(vec: &Vec<Int>) -> Vec<Int> {
    let mut sorted = clone(vec)
    let mut i = 0
    let mut j = 1
    while i < len(&sorted) - 1 {
        while j < len(&sorted) && get(&sorted, j) < get(&sorted, j - 1) {
            j = j + 1
        }
        if j < len(&sorted) {
            swap(&mutsorted, i, j)
            i = i + 1
            j = i + 1
        }
    }
    sorted
}

fn main() {
    let list = push(push(push(push(vec(), 5), 3), 8), 1)
    let sorted = push(push(list, 9), 2)
    print(get(&sorted, 0))
    drop sorted
}
