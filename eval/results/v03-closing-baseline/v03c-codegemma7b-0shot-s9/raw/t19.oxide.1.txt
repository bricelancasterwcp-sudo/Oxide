fn area(r: Rect) -> Int {
    r.width * r.height
}

fn perimeter(r: Rect) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let r = Rect { width: 7, height: 4 }
    print(area(r))
    print(perimeter(r))
}
