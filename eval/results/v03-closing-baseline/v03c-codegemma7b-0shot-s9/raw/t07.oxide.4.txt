fn main() {
    let mut acc = vec(0, 0, 0, 0, 0, 0, 0, 0)
    for n in range(1, 33) {
        let x = n * 3
        if len(acc) < 8 {
            acc = push(acc, x)
        } else {
            print(sum(acc))
            break
        }
    }
}

fn sum(v: Vec<Int>) -> Int {
    let mut s = 0
    for x in v {
        s = s + x
    }
    s
}

fn len(v: Vec<Int>) -> Int {
    let mut i = 0
    for _ in v {
        i = i + 1
    }
    i
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a + 1 < b {
        v = push(v, i)
    }
    v
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v2 = vec()
    for e in v {
        v2 = push(v2, e)
    }
    v2 = push(v2, x)
    v2
}

fn print(x: Int) {
    print_str(int_to_str(x))
}

fn print_str(s: Str) {
    for c in chars(s) {
        print(c)
    }
}

fn int_to_str(n: Int) -> Str {
    let mut s = Str("")
    for d in 1000000000 / 10 < n {
        s = concat(s, int_to_str(n / d))
        n = n % d
    }
    s = concat(s, int_to_str(n))
    s
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    let mut i = 0
    while i < str_len(s) {
        v = push(v, str_len(substr(s, 0, i + 1)))
        i = i + 1
    }
    v
}

fn substr(s: Str, a: Int, b: Int) -> Str {
    let mut s2 = Str("")
    let mut i = a
    while i < b {
        s2 = concat(s2, get(chars(s), i)?)
        i = i + 1
    }
    s2
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < len(v) {
        Some(v)
    } else {
        None
    }
}

fn clone(x: Int) -> Int {
    x
}

fn str_len(s: Str) -> Int {
    let mut i = 0
    for _ in chars(s) {
        i = i + 1
    }
    i
}

fn concat(a: Str, b: Str) -> Str {
    let mut s = Str("")
    for c in chars(a) {
        s = concat(s, c)
    }
    for c in chars(b) {
        s = concat(s, c)
    }
    s
}
