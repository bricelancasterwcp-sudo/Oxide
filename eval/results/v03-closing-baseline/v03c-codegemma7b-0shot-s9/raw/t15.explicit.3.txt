fn main() {
    let input = vec_str("12", "x", "30")
    let mut sum = 0
    let mut fail = 0
    for word in &input {
        match parse_int(word) {
            Some(n) => {
                sum = sum + n
            },
            None => {
                fail = fail + 1
            },
        }
    }
    print(sum)
    print(fail)
}

fn vec_str(s: &Str, rest: &Str, rest_rest: &Str) -> Vec<Str> {
    vec(s, rest, rest_rest)
}

fn parse_int(s: &Str) -> Option<Int> {
    match s.parse() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn print(i: Int) {
    let s = int_to_str(i)
    print_str(&s)
}

fn print_str(s: &Str) {
    let c = chars(s)
    for i in c {
        print_char(i)
    }
    print_char(Unit)
}

fn print_char(c: Str) {
    let n = str_len(&c)
    for i in 0 < n {
        push(push(vec(), i), c)
    }
}

fn int_to_str(n: Int) -> Str {
    let mut s = Str
    if n < 0 {
        s = push(s, char(45))
        n = trunc(n)
    }
    while n > 0 {
        let rem = n % 10
        let digit = char(rem + 48)
        s = push(s, digit)
        n = trunc(n / 10)
    }
    if s == Str {
        s = push(s, char(48))
    }
    str_rev(&s)
}

fn str_rev(s: &Str) -> Str {
    let mut rev = Str
    for i in s {
        rev = push(rev, i)
    }
    rev
}

fn chars(s: &Str) -> Vec<Str> {
    let mut chars = vec()
    for i in 0 < str_len(s) {
        let c = get(s, i)
        if c != None {
            chars = push(chars, c)
        }
    }
    chars
}

fn get(s: &Str, i: Int) -> Option<Str> {
    if i >= str_len(s) {
        None
    } else {
        Some(char(str_len(s) - 1 - i + 48))
    }
}

fn str_len(s: &Str) -> Int {
    let mut i = 0
    for _ in s {
        i = i + 1
    }
    i
}

fn push(v: Vec<Str>, x: Str) -> Vec<Str> {
    let mut new = v
    new = new + x
    new
}

fn vec() -> Vec<Str> {
    let mut new = Vec
    new
}

fn clone(x: &Str) -> Str {
    let mut new = Str
    for i in x {
        new = push(new, i)
    }
    new
}
