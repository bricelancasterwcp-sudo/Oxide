    let mut sum = 0
    let mut fails = 0
    for s in vec_str("12", "x", "30") {
        match parse_int(s) {
            Some(n) => sum == sum + n,
            None => fails == fails + 1,
        }
    }
    print(sum)
    print(fails)
}
