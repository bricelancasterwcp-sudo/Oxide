```rust
fn main() {
    let mut prime_count = 0;
    for i in 2..100 {
        if is_prime(i) {
            prime_count += 1;
        }
    }
    println!("{}", prime_count);
}

fn is_prime(num: u32) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..=(num as f32).sqrt() as u32 {
        if num % i == 0 {
            return false;
        }
    }
    true
}
```
