fn is_palindrome(s: Str) -> Option<Bool> {
    let mut i = 0
    let mut j = len(s) - 1
    while i < j {
        let left = get(s, i)?
        let right = get(s, j)?
        if left != right {
            return Some(false)
        }
        i = i + 1
        j = j - 1
    }
    Some(true)
}
