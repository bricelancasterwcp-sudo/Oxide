fn main() {
    let list = push(push(push(vec(), 3), 8), -2)
    let list = push(push(list, 12), 7)
    let length = 0
    let largest = 0
    for number in &list {
        if number > largest {
            largest = number
        }
        if number > 0 {
            length = length + 1
        }
    }
    print_int(length)
    print_str("\n")
    print_int(largest)
    print_str("\n")
}
