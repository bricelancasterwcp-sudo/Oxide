fn parse_int(s: Str) -> Option<Int> {
    if s == "x" {
        None
    } else {
        Some(trunc(to_float(s).unwrap()))
    }
}

    drop s
}

fn to_float(n: Int) -> Option<Float> {
    if n == 0 {
        None
    } else {
        Some(n.to_float())
    }
}

fn trunc(x: Float) -> Int {
    x.trunc()
}
