struct R { l: Str, v: Vec<Int> }

fn f(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn g(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn m() {
    let r = R { l: "lab", v: push(push(vec(), 3), 42) }
    print(f(r.v))
    let r2 = R { l: "lab2", ..r }
    match g(r2.v) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

fn main() {
    m()
}
