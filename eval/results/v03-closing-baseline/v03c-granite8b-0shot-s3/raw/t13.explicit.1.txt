fn read_backward(text: &Str) -> Bool {
    let reversed = text.chars().rev().collect()
    let mut left = 0
    let mut right = text.len() - 1
    while left < right {
        if text.chars().nth(left).unwrap() != reversed.chars().nth(left).unwrap() {
            return false
        }
        left = left + 1
        right = right - 1
    }
    true
}

fn main() {
    print(read_backward("level"))
    print(read_backward("stereo"))
    print(read_backward("rotor"))
}
