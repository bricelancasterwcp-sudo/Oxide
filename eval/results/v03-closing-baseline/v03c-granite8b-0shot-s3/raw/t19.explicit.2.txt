fn area(r: &Rectangle) -> Int {
    r.width * r.height
}

fn perimeter(r: &Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let rect = Rectangle { width: 7, height: 4 }
    print_str("Area: ")
    print(area(&rect))
    print_str("\nPerimeter: ")
    print(perimeter(&rect))
}

struct Rectangle {
    width: Int,
    height: Int,
}

fn main() {
    let rect = Rectangle { width: 7, height: 4 }
    print_str("Area: ")
    print(area(&rect))
    print_str("\nPerimeter: ")
    print(perimeter(&rect))
}
