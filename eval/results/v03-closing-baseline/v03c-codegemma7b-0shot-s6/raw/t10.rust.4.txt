use std::io;

fn main() {
    let mut list = vec![4, 1, 7, 3, 9, 2];
    let mut counter = 0;

    for num in &mut list {
        if *num > 3 {
            println!("{}", num);
            counter += 1;
        } else {
            list.remove(num);
        }
    }

    println!("{} values were kept", counter);
}
