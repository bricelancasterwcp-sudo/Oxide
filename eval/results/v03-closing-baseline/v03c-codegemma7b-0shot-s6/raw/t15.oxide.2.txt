    let mut sum = 0
    let mut count = 0
    for s in vec_str("12", "x", "30") {
        match parse_int(s) {
            Some(n) => {
                sum = sum + n
                count = count + 1
            },
            None => continue_next(),
        }
    }
    print(sum)
    print(count)
}

fn parse_int(s: Str) -> Option<Int> {
    let x = s.chars().nth(0)?
    if x >= char_from_int(48) && x <= char_from_int(57) {
        Some(int_from_char(x))
    } else {
        None
    }
}
