use std::io;

fn main() {
    let items = [
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];

    let mut total_cost = 0;
    let mut highest_cost = 0;
    let mut highest_cost_item = "";

    for (name, unit_price, quantity) in items {
        let cost = unit_price * quantity;
        total_cost += cost;

        if cost > highest_cost {
            highest_cost = cost;
            highest_cost_item = name;
        }
    }

    println!("Total cost: {}", total_cost);
    println!("Item with highest cost: {}", highest_cost_item);
}
