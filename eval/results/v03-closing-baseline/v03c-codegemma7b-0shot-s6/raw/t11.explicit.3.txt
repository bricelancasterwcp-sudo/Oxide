    let mut nums = push(push(push(push(push(vec(), 5), 3), 8), 1), 9)
    print(get(&nums, 0))
    print(get(&nums, 1))
    print(get(&nums, 2))
    print(get(&nums, 3))
    print(get(&nums, 4))
    drop nums
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        None
    } else {
        Some(v.get(i)?)
    }
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = v
    result.push(x)
    result
}

fn print(x: Option<Int>) {
    if letSome(y) == x {
        print(y)
    } else {
        print("None")
    }
}
