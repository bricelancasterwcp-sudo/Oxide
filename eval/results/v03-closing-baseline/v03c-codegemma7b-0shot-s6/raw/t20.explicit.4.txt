fn main() {
    let pen = Item { price: 3, quantity: 4 }
    let pad = Item { price: 9, quantity: 2 }
    let ink = Item { price: 5, quantity: 3 }
    let cost = pen.cost() + pad.cost() + ink.cost()
    println(cost)
    println(most_expensive(&pen, &pad, &ink))
}

struct Item {
    price: Int,
    quantity: Int,
}

fn cost(self: &Item) -> Int {
    self.price * self.quantity
}

fn most_expensive(a: &Item, b: &Item, c: &Item) -> String {
    if a.cost() >= b.cost() && a.cost() >= c.cost() {
        "Pen"
    } else if b.cost() >= a.cost() && b.cost() >= c.cost() {
        "Pad"
    } else {
        "Ink"
    }
}

fn clone(x: &Item) -> Item {
    Item(x.price, x.quantity)
}
