fn str_to_int(s: &Str) -> Result<Int, String> {
    match parse_int(s) {
        Ok(n) => Ok(n),
        Err(_) => Err(format("failed to parse {}", s)),
    }
}

fn add(a: Int, b: Int) -> Int {
    a + b
}

fn parse_int(s: &Str) -> Result<Int, String> {
    match s.parse() {
        Ok(n) => Ok(n),
        Err(_) => Err(format("{} is not an integer", s)),
    }
}

fn result_ok(r: &Result<Int, String>) -> Option<Int> {
    match r {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn result_is_err(r: &Result<Int, String>) -> Option<String> {
    match r {
        Ok(_) => None,
        Err(e) => Some(e.to_string()),
    }
}
