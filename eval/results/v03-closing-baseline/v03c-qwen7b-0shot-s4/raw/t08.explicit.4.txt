fn count_positive(v: &Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn max_value(v: &Vec<Int>) -> Int {
    let mut max = v.get(0).unwrap()
    for x in v {
        if x > max {
            max = x
        }
    }
    max
}

fn main() {
    let nums = push(push(push(vec(), 3), 8), push(push(vec(), -2), push(vec(), 12)))
    let nums = push(nums, 7)
    let pos_count = count_positive(&nums)
    print_str("Positive count: ")
    print(pos_count)
    print_str("\n")
    let max_val = max_value(&nums)
    print_str("Max value: ")
    print(max_val)
    print_str("\n")
    drop nums
}
