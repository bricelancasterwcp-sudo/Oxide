fn sum_multiples_of_three() -> Int {
    let mut multiples = vec()
    for i in range(1, 9) {
        multiples = push(multiples, i * 3)
    }
    let sum = sum_big(&multiples, 0)
    drop multiples
    sum
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn main() {
    let total = sum_multiples_of_three()
    print(total)
}
