struct Reading { label: Str, values: Vec<Int> }

fn get_value(v: Vec<Int>, pos: Int) -> Option<Int> {
    if pos >= 0 && pos < len(v) {
        Some(get(v, pos)?)
    } else {
        None
    }
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(vec(), 10), 20), 30) }
    let value1 = get_value(r.values, 1)
    let value5 = get_value(r.values, 5)
    match value1 {
        Some(n) => print(n),
        None => print(-1),
    }
    match value5 {
        Some(n) => print(n),
        None => print(-1),
    }
}
