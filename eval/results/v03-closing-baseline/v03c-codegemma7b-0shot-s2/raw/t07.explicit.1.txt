fn sum_of_multiples(n: Int) -> Int {
    let mut sum = 0
    for i in range(1, n + 1) {
        if i % 3 == 0 {
            sum = sum + i
        }
    }
    sum
}

fn main() {
    let multiples = range(1, 25)
    let result = multiples.iter().filter(is_multiple_of_three).map(sum_of_multiples).fold(0, plus)
    print(result)
    drop multiples
    drop result
}

fn is_multiple_of_three(n: Int) -> Bool {
    n % 3 == 0
}

fn plus(a: Int, b: Int) -> Int {
    a + b
}
