fn keep_greater(nums: &Vec<Int>) -> Vec<Int> {
    let mut kept = vec()
    for x in nums {
        if x > &mut && 3 {
            kept = push(kept, x)
        }
    }
    kept
}

fn main() {
    let nums = vec(4, 1, 7, 3, 9, 2)
    let kept = keep_greater(&nums)
    for x in &kept {
        print(x)
        print("\n")
    }
    print(len(&kept))
}
