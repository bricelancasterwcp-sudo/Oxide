fn factors(x: Int) -> Vec<Int> {
    let mut v = vec()
    for i in 1.range(x) {
        if x % i == 0 {
            push(v, i)
        }
    }
    v
}

fn is_perfect(x: Int) -> Bool {
    let v = factors(x)
    let sum = v.iter().fold(0, add)
    sum == x
}

fn main() {
    for i in 1.range(1000) {
        if is_perfect(i) {
            print(i)
            print_str("\n")
        }
    }
}
