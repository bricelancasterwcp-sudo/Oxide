struct Palindrome { text: Str }

fn is_palindrome(p: Palindrome) -> Bool {
    let len = str_len(p.text)
    for i in range(0, len / 2) {
        let char_at_i = get(p.text, i)
        let char_at_len_minus_i_minus_1 = get(p.text, len - i - 1)
        if char_at_i != char_at_len_minus_i_minus_1 {
            return false
        }
    }
    true
}

fn main() {
    let pal1 = Palindrome { text: "level" }
    print(is_palindrome(pal1))
    let pal2 = Palindrome { text: "stereo" }
    print(is_palindrome(pal2))
    let pal3 = Palindrome { text: "rotor" }
    print(is_palindrome(pal3))
}
