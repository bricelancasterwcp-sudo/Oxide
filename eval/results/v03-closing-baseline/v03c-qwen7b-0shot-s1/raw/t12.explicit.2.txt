fn main() {
    let s = "stack"
    let chars = chars(s)
    let reversed_chars = chars.iter().rev().map(String).collect()
    let reversed_str: String = reversed_chars.join("")
    print_str(&reversed_str)
}

fn chars(s: &Str) -> Vec<Str> {
    s.chars().map(str).collect()
}

fn print_str(s: &Str) {
    for c in s.chars() {
        print(&c)
    }
}
