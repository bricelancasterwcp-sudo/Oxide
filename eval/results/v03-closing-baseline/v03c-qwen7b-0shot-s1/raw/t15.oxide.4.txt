struct ParseResult {
    sum: Int,
    count: Int,
}

fn parse_and_count(s1: Str, s2: Str, s3: Str) -> ParseResult {
    let mut sum = 0
    let mut count = 0
    match parse_int(s1) {
        Some(n) => sum == sum + n,
        None => count == count + 1,
    }
    match parse_int(s2) {
        Some(n) => sum == sum + n,
        None => count == count + 1,
    }
    match parse_int(s3) {
        Some(n) => sum == sum + n,
        None => count == count + 1,
    }
    ParseResult(sum, count)
}

fn main() {
    let r = parse_and_count("12", "x", "30")
    print(r.sum)
    print(r.count)
}
