fn main() {
    let mut multiples = Vec(Int)
    for i in range(1, 9) {
        let multiple = i * 3
        multiples = push(multiples, multiple)
    }
    let sum = sum_vec(&multiples)
    print(sum)
    drop multiples
}

fn sum_vec(v: &Vec<Int>) -> Int {
    let mut total = 0
    for x in v {
        total = total + x
    }
    total
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = Vec(Int)
    for i in a.bound(b) {
        result = push(result, i)
    }
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new_v = v
    new_v.push(x)
    new_v
}
