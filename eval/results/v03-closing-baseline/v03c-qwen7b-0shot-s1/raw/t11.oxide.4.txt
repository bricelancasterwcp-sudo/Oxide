fn main() {
    let numbers = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(numbers)
    for x in sorted {
        print_str(x.to_string())
    }
}

fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut v = v
    for i in range(0, v.len()) {
        for j in range(0, v.len() - i - 1) {
            if v.get(j).unwrap() > v.get(j + 1).unwrap() {
                let temp = v.get(j).unwrap()
                v.set(j, v.get(j + 1).unwrap())
                v.set(j + 1, temp)
            }
        }
    }
    v
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = Vec()
    for i in a < b {
        vec_push(v, i)
    }
    v

fn unwrap(opt: Option<Int>) -> Int {
    match opt {
        Some(x) => x,
        None => panic("unwrap called on None"),
    }
}
