struct CountInfo { s_count: Int, distinct_count: Int }

fn count_chars(s: Str) -> CountInfo {
    let mut count = CountInfo { s_count: 0, distinct_count: 0 }
    let mut chars_set = Vec()
    for c in chars(s) {
        if c == "s" {
            count.s_count = count.s_count + 1
        }
        if !contains(chars_set, c) {
            chars_set = push(chars_set, c)
            count.distinct_count = count.distinct_count + 1
        }
    }
    count
}

fn main() {
    let s = "mississippi"
    let info = count_chars(s)
    print(info.s_count)
    print_str(" ")
    print(info.distinct_count)
}
