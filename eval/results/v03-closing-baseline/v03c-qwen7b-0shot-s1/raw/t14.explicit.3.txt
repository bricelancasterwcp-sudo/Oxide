    let text = "mississippi"
    let count_s = text.chars().filter(char_eq(char_of_strs_str_s())).count()
    let distinct_chars = text.chars().collect()
    print(count_s)
    print_str("\n")
    print(distinct_chars.len())
    drop text
}
