struct Item {
    name: Str,
    price: Int,
    quantity: Int,
}

fn calculate_cost(item: Item) -> Int {
    item.price * item.quantity
}

fn main() {
    let pen = Item { name: "pen", price: 3, quantity: 4 }
    let pad = Item { name: "pad", price: 9, quantity: 2 }
    let ink = Item { name: "ink", price: 5, quantity: 3 }
    let total_cost = calculate_cost(pen.clone()) + calculate_cost(pad.clone()) + calculate_cost(ink.clone())
    print(total_cost)
    let items = vec(pen, pad, ink)
    let most_expensive_item = items.iter().max_by_key(unwrap_or(0, calculate_cost)).unwrap()
    print_str(most_expensive_item.name)
}
