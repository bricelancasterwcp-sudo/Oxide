fn main() {
    let level = "level".chars().all(char_eq_backwards(&level))
    print(level)
    drop level
    let stereo = "stereo".chars().all(char_eq_backwards(&stereo))
    print(stereo)
    drop stereo
    let rotor = "rotor".chars().all(char_eq_backwards(&rotor))
    print(rotor)
    drop rotor
}

fn char_eq_backwards(text: &Vec<Str>) -> Bool {
    let len = len(text)
    let half = len / 2
    for i in 0 < half {
        if text.get(i) != text.get(len - i - 1) {
            return false
        }
    }
    true
}

fn print(x: Bool) {
    if x {
        print_str("true")
    } else {
        print_str("false")
    }
}

fn print_str(s: Str) {
    for c in chars(s) {
        print_char(c)
    }
}

fn print_char(c: Str) {
    let bytes = str_len(c)
    for i in 0 < bytes {
        print_byte(c.get(i))
    }
}

fn print_byte(b: Byte) {
    let ch = to_char(b)
    if ch >= "0".to_char() && ch <= "9".to_char() {
        print_char(ch)
    } else if ch >= "a".to_char() && ch <= "z".to_char() {
        print_char(ch)
    } else if ch >= "A".to_char() && ch <= "Z".to_char() {
        print_char(ch)
    } else {
        print_char("\\x")
        print_byte(to_hex_digit(b / 16))
        print_byte(to_hex_digit(b % 16))
    }
}

fn to_hex_digit(n: Int) -> Byte {
    if n >= 0 && n < 10 {
        n + "0".to_byte()
    } else if n >= 10 && n < 16 {
        n - 10 + "a".to_byte()
    } else {
        "0".to_byte()
    }
}

fn to_char(b: Byte) -> Str {
    let mut chars = vec()
    let mut n = b
    while n > 0 {
        chars.push(to_char_digit(n % 10))
        n = n / 10
    }
    chars.reverse()
    concat(chars, "")
}

fn to_char_digit(n: Int) -> Byte {
    n + "0".to_byte()
}
