fn sum_of_divisors(n: Int) -> Int {
    let mut total = 0
    for i in range(1, n) {
        if n % i == 0 {
            total = total + i
        }
    }
    total
}

fn is_perfect(n: Int) -> Bool {
    n == sum_of_divisors(n)
}

fn main() {
    for n in range(1, 1000) {
        if is_perfect(n) {
            print(n)
        }
    }
}
