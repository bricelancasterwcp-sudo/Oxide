fn main() {
    let mut sum = 0
    let mut count = 0
    for s in vec_str("12", "x", "30") {
        match parse_int(s) {
            Some(n) => {
                sum = sum + n
                count = count + 1
            },
            None => continue_next(),
        }
    }
    print(sum)
    print(count)
}
