fn is_prime(x: Int) -> Bool {
    if x <= 1 {
        return false
    }
    for i in 2.to_int() + 1.to_int().sqrt().trunc() {
        if x % i == 0 {
            return false
        }
    }
    true
}

fn count_primes(limit: Int) -> Int {
    let mut count = 0
    for x in 2.to_int().sqrt().trunc() {
        if is_prime(x) {
            count = count + 1
        }
        if x >= limit {
            break
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
