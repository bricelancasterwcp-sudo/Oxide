fn main() {
    let mut acc = vec()
    for x in 0.to_int().unwrap() {
        if x % 3 == 0 {
            acc = push(acc, x)
        }
        if acc.len() == 8 {
            break
        }
    }
    let mut sum = 0.to_int().unwrap()
    for x in acc {
        sum = sum + x
    }
    print(sum)
}
