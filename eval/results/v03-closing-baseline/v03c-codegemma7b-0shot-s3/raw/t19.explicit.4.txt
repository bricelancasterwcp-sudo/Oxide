struct Rectangle {
    width: Int,
    height: Int,
}

fn new(width: Int, height: Int) -> Rectangle {
    Rectangle(width, height)
}

fn width(r: &Rectangle) -> Int {
    r.width
}

fn height(r: &Rectangle) -> Int {
    r.height
}

fn area(r: &Rectangle) -> Int {
    r.width * r.height
}

fn perimeter(r: &Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let rect = new(7, 4)
    print(area(&rect))
    print(perimeter(&rect))
    drop rect
}
