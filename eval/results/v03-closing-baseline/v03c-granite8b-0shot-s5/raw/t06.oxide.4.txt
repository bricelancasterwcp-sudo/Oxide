struct IntWrapper { value: Int }

fn divisors(n: Int) -> Vec<Int> {
    let mut v = vec()
    for i in range(1, n) {
        if n % i == 0 {
            v = push(v, i)
        }
    }
    v
}

fn add(a: Int, b: Int) -> Int {
    a + b
}

fn reduce(init: Int, v: Vec<Int>, add: Int) -> Int {
    let mut sum = init
    for x in v {
        sum = add(sum, x)
    }
    sum
}

fn is_perfect(n: Int) -> Bool {
    let sum_of_divisors = reduce(0, divisors(n), add)
    sum_of_divisors == n
}

fn main() {
    for n in range(2, 1000) {
        if is_perfect(n) {
            print(IntWrapper { value: n })
        }
    }
}
