struct IntWrapper { value: Int }

fn divisors(n: Int) -> Vec<Int> {
    let mut v = vec()
    for i in range(1, n) {
        if n % i == 0 {
            v = push(v, i)
        }
    }
    v
}

fn main() {
    let max_number = 1000
    let mut n = 2
    while n < max_number {
        let sum_of_divisors = reduce(0, divisors(n), add)
        if sum_of_divisors == n {
            print(n)
        }
        n = n + 1
    }
}
