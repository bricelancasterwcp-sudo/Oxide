fn is_prime(x: Int) -> Bool {
    for i in range(2, x) {
        if (x % i == 0) {
            return false
        }
    }
    true
}

fn count_primes(x: Int) -> Int {
    let count = 0
    for i in range(2, x) {
        if (is_prime(i)) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
