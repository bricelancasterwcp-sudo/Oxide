fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < len(v) {
        Some(get_unsafe(v, i).clone())
    } else {
        None
    }
}

fn main() {
    let v = vec(10, 20, 30)
    match get(v, 1) {
        Some(x) => print(x),
        None => print_str("out of bounds"),
    }
    match get(v, 5) {
        Some(x) => print(x),
        None => print_str("out of bounds"),
    }
}
