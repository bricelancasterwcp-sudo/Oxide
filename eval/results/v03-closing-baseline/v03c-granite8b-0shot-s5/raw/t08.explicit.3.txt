fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn main() {
    let nums = push(push(vec(), 3), 8)
    let nums = push(nums, -2)
    let nums = push(nums, 12)
    let nums = push(nums, 7)
    let count = 0
    let largest = -100
    for x in &nums {
        if x > 0 {
            count = count + 1
        }
        if x > largest {
            largest = x
        }
    }
    print_str("Count: ")
    print(count)
    print_str(" | Largest: ")
    print(largest)
}
