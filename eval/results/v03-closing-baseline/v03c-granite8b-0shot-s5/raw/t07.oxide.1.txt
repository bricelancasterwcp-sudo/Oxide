fn main() {
    let multiples = vec(3, 6, 9, 12, 15, 18, 21, 24).filter(is_multiple_of(3)).collect()
    print_sum(multiples)
}

fn print_sum(multiples: Vec<Int>) {
    let sum = reduce(0, multiples, add)
    print(sum)
}

fn is_multiple_of(n: Int) -> Bool {
    move || x % n == 0
}

fn reduce(init: Int, v: Vec<Int>, f: Int, fn1: Int, fn2: Int) -> Int {
    match v {
        Vec(n, ith, rest) => if n == 0 { init } else { reduce(f(init, ith), rest, f, fn1, fn2) },
        Nullary => init,
    }
}

fn add(x: Int, y: Int) -> Int {
    x + y
}

fn push(v: Vec, x: Int) -> Vec {
    let cap = get(v, capacity)
    let first = get(v, first)
    let last = get(v, last)
    if last == cap {
        return v
    } else {
        let v = update(v, last, x)
        let last = last + 1
        update(v, last, last)
    }
}

fn print(x: Int) {
    let s = to_string(x)
    let ptr = get(s, ptr)
    let len = get(s, len)
    sys_write(1, ptr, len)
}

fn to_string(x: Int) -> Str {
    let s = Int { value: 10 }
    let s = update(s, 0, x)
    let s = update(s, 1, 1)
    s
}

fn sys_write(fd: Int, ptr: Int, len: Int) {
    let _ = fd
    let _ = ptr
    let _ = len
}
