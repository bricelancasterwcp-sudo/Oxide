fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    let result = match i >= 0 && i < len(&v) {
        True => Some(v.get(i)),
        False => None,
    }
    match result {
        Some(r) => r,
        None => Option.none(),
    }
}
