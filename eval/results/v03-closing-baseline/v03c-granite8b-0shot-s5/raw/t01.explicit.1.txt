fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_big(&nums, 10))
    drop nums
    print(sum_big(&bigger, 10))
    drop bigger
}

fn square(x: Int) -> Int {
    x * x
}

fn sum_squares(n: Int) -> Int {
    let sum = 0
    for i in range(0, n) {
        sum = sum + square(i)
    }
    sum
}

fn main() {
    print(sum_squares(10))
}
