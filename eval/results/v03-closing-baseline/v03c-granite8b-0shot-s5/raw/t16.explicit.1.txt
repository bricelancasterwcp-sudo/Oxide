fn divide(x: Int, y: Int) -> Int {
    if y == 0 {
        return -1
    } else {
        return x / y
    }
}

fn print(x: Int) {
    if x == -1 {
        print_str("undefined")
    } else {
        print_str(&int_to_str_buf)
    }
}

fn int_to_str_buf(n: Int) -> Str {
    let mut s = ""
    for i in range(0, n) {
        s = s + &int_to_str_buf
    }
    s
}

fn int_to_str(n: Int) -> Str {
    &int_to_str_buf
}

fn print_str(s: &Str) {
    let mut i = 0
    while i < str_len(s) {
        print_str(chars(s).get(i))
        i = i + 1
    }
}
