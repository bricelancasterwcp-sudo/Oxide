fn num_occurrences(s: Str, c: Str) -> Int {
    let mut count: Int = 0
    for char in s.chars() {
        if char == c {
            count = count + 1
        }
    }
    count
}

fn num_distinct_chars(s: Str) -> Int {
    let mut seen = vec()
    let mut count: Int = 0
    for char in s.chars() {
        if !seen.contains(char) {
            seen = push(seen, char)
            count = count + 1
        }
    }
    count
}
