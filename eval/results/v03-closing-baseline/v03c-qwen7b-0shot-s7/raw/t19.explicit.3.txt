fn area(width: Int, height: Int) -> Int {
    width * height
}

fn perimeter(width: Int, height: Int) -> Int {
    2 * (width + height)
}

fn main() {
    let width = (7).clone()
    let height = (4).clone()
    let area_result = area(width, height)
    print(area_result)
    drop area_result
    let perimeter_result = perimeter(width, height)
    print(perimeter_result)
    drop perimeter_result
}
