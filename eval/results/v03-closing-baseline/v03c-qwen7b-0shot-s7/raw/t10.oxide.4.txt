struct VecInt { values: Vec<Int> }

fn filter(v: VecInt, threshold: Int) -> VecInt {
    let mut result = VecInt { values: vec() }
    for x in v.values {
        if x > threshold {
            result.values.push(x)
        }
    }
    result
}

fn main() {
    let v = VecInt { values: vec(4, 1, 7, 3, 9, 2) }
    let filtered = filter(v, 3)
    for x in filtered.values {
        print(x)
        print_str("\n")
    }
    print(filtered.values.len())
    print_str("\n")
}
