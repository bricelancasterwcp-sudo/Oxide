fn main() {
    let pen_cost = 3 * 4
    let pad_cost = 9 * 2
    let ink_cost = 5 * 3
    let total_cost = pen_cost + pad_cost + ink_cost
    print(total_cost)
    let max_cost = pen_cost.max(pad_cost).max(ink_cost)
    if max_cost == pen_cost {
        print("pen")
    } else if max_cost == pad_cost {
        print("pad")
    } else {
        print("ink")
    }
}
