fn area(width: Int, height: Int) -> Int {
    width * height
}

fn perimeter(width: Int, height: Int) -> Int {
    2 * (width + height)
}

fn main() {
    let width = (7.0.to_int()).to_int()
    let height = (4.0.to_int()).to_int()
    let area_val = area(width, height)
    let perimeter_val = perimeter(width, height)
    print(area_val)
    print(perimeter_val)
}
