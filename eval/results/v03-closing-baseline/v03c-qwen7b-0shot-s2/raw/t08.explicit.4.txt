fn main() {
    let numbers = vec(3, 8, -2, 12, 7)
    let mut positive_count = 0
    let mut max_value: Option<Int> = None
    for num in &numbers {
        if num > 0 {
            positive_count = positive_count + 1
        }
        max_value = match max_value {
            Some(max) => Some(if num > max { num } else { max }),
            None => Some(num),
        }
    }
    print(positive_count)
    print(match max_value {
            Some(max) => max,
        })
    drop numbers
}
