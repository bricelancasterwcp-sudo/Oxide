fn sum_multiples_of_three(limit: Int) -> Int {
    let mut sum = 0
    let mut count = 0
    let mut i = 0
    while count < limit {
        i = i + 1
        if i % 3 == 0 {
            sum = sum + i
            count = count + 1
        }
    }
    sum
}

fn main() {
    let sum = sum_multiples_of_three(8)
    print(sum)
}
