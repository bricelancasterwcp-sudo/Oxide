struct List { head: Option<Int>, tail: Option<Box<List>> }

fn push(l: Option<Box<List>>, x: Int) -> Option<Box<List>> {
    Some(Box { head: Some(x), tail: l })
}

fn sort(l: Option<Box<List>>, acc: Option<Box<List>>) -> Option<Box<List>> {
    match l {
        None => acc,
        Some(node) => sort(node.tail, push(acc, node.head.unwrap())),
    }
}

fn print_list(l: Option<Box<List>>) {
    match l {
        None => print("\n"),
        Some(node) => {
            print("{}\n", node.head.unwrap() + print_list(node.tail))
        },
    }
}
