struct Reading { label: Str, values: Vec<Int> }

fn main() {
    let r = Reading { label: "lab", values: vec(4, 1, 7, 3, 9, 2) }
    let mut kept = Vec()
    for x in r.values {
        if x > 3 {
            push(kept, x)
        }
    }
    for x in kept {
        print(x)
    }
    print(len(kept))
}
