    while i < n {

fn is_prime(n: Int) -> Bool {
    if n == 2 || n == 3 {
        return true
    }
    if n < 2 || n % 2 == 0 || n % 3 == 0 {
        return false
    }
    let mut i = 5
    while i * i <= n {
        if n % i == 0 || n % (i + 2) == 0 {
            return false
        }
        i = i + 6
    }
    true
}

fn int_to_str(n: Int) -> Str {
    let mut s = vec()
    let mut n = n
    let mut buf = vec()
    while n > 0 {
        let r = n % 10
        buf = push(buf, 48 + r)
        n = n / 10
    }
    let mut buf = reverse(buf)
    while !buf.is_empty() {
        s = push(s, buf.first())
        buf = buf + 1
    }
    s
}

fn reverse(v: Vec<Int>) -> Vec<Int> {
    let mut v = clone(&v)
    let mut i = 0
    let mut j = v.len() - 1
    while i < j {
        let t = v.get(i)
        v.set(i, v.get(j))
        v.set(j, t)
        i = i + 1
        j = j - 1
    }
    v
}

    }
    Some(v.get(i))
}

fn parse_int(s: Str) -> Option<Int> {
    let n = chars(s).len()
    let mut i = 0
    while i < n && s.get(i) == 48 {
        i = i + 1
    }
    if i == n {
        return None
    }
    let mut result = 0
    while i < n {
        let d = s.get(i) - 48
        if d < 0 || d > 9 {
            return None
        }
        result = result * 10 + d
        i = i + 1
    }
    Some(result)
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    let n = s.len()
    let mut i = 0
    while i < n {
        v = push(v, s.slice(i, i + 1))
        i = i + 1
    }
    v
}
