fn main() {
    let mut count = 0;
    for i in 2..100 {
        let mut is_prime = true;
        for j in 2..i {
            if i % j == 0 {
                is_prime = false;
                break;
            }
        }
        if is_prime {
            count += 1;
        }
    }
    println!("{}", count);
}
