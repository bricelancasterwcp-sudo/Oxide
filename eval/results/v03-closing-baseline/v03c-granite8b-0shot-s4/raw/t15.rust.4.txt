fn main() {
    let inputs = ["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;

    for input in inputs {
        if let Ok(num) = input.parse::<i32>() {
            sum += num;
        } else {
            count += 1;
        }
    }

    println!("Sum of parsed integers: {}", sum);
    println!("Count of failed parses: {}", count);
}
