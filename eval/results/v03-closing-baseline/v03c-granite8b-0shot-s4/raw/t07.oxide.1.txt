struct S { a: Int, b: Int }

fn main() {
    let v = vec(1, 2, 3, 4, 5)
    print(sum(map(v, square)))
}

fn map(v: Vec<Int>, f: Int) -> Vec<Int> {
    let mut r = vec()
    for x in v {
        push(r, f(x))
    }
    r
}

fn square(x: Int) -> Int {
    x * x
}

fn sum(v: Vec<Int>) -> Int {
    let mut r = 0
    for x in v {
        r = r + x
    }
    r
}

fn filter(v: Vec<Int>, p: Int) -> Vec<Int> {
    let mut r = vec()
    for x in v {
        if p(x) {
            push(r, x)
        }
    }
    r
}

fn main() {
    let v = vec(1, 2, 3, 4, 5, 6, 7, 8)
    print(sum(filter(v, is_multiple_of_3)))
}

fn is_multiple_of_3(x: Int) -> Bool {
    x % 3 == 0
}

fn main() {
    let v = vec(1, 2, 3, 4, 5, 6, 7, 8)
    let x = first_big(v)
    match x {
        Some(y) => print(y + 1),
        None => print_str("no big numbers"),
    }
}

fn first_big(v: Vec<Int>) -> Option<Int> {
    let mut r = None
    for x in v {
        if x > 10 {
            r = Some(x)
            break
        }
    }
    r
}

fn main() {
    let v = vec(1, 2, 3, 4, 5)
    match second(v) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}
