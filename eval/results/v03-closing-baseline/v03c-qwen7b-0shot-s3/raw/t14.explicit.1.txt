fn main() {
    let text = "mississippi"
    let count_s = text.chars().filter(char_eq_s).count()
    let distinct_chars = text.chars().collect_to_set().len()
    print_int(count_s)
    print_int(distinct_chars)
}

fn char_eq_s(c: Str) -> Bool {
    c == "s"
}

fn collect_to_set(v: Vec<Str>) -> Set<Str> {
    let mut s = Set()
    for x in v {
        s.insert(x)
    }
    s
}

fn print_int(i: Int) {
    print_str(i.to_str())
}

fn to_str(n: Int) -> Str {
    match n {
        None => "0",
        Some(x) => x.to_str(),
    }
}

fn print_str(s: Str) {
    let mut chars = s.chars()
    while letSome(c) == chars.next() {
        print_char(c)
    }
}

fn print_char(c: Char) {
    let ascii = c.as_ascii()
    let ptr = &ascii
    unsafe * (ptr.add(0)) == ascii
}
