fn main() {
    let pen = (3.0 * 4.0).to_int()
    let pad = (9.0 * 2.0).to_int()
    let ink = (5.0 * 3.0).to_int()
    let total_cost = pen + pad + ink
    print_str("Combined cost of all items: ")
    print(total_cost)
    print_str("\n")
    let items = vec(("pen".to_str().to_vec()), ("pad".to_str().to_vec()), ("ink".to_str().to_vec()))
    let mut max_cost = 0
    let mut max_item = ""
    for item in items {
        let item_cost = match item {
            Vec(a) => {
                let name = a.as_str()
                let cost = match name {
                    Some(s) => match s.as_str() {
                        Some(name) => match name {
                            Some(name) => match name.as_str() {
                                Some(name) => match name {
                                    Some(name) => match name.as_str() == "pen" || name.as_str() == "pad" || name.as_str() == "ink" {
                                        Some(_) => 1,
                                        None => 0,
                                    },
                                    None => 0,
                                },
                                None => 0,
                            },
                            None => 0,
                        },
                        None => 0,
                    },
                    None => 0,
                }
            },
            _ => 0,
        }
        if item_cost > max_cost {
            max_cost = item_cost
            max_item = item.as_str()
        }
    }
    print_str("Item with the highest cost: ")
    print_str(max_item)
    print_str("\n")
}
