fn main() {
    let strings = ["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;

    for s in strings {
        match s.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => count += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Count of failed parses: {}", count);
}
