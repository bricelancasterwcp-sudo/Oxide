```rust
fn main() {
    struct Rectangle {
        width: u32,
        height: u32,
    }

    impl Rectangle {
        fn perimeter(&self) -> u32 {
            2 * (self.width + self.height)
        }
    }

    struct Triangle {
        side1: u32,
        side2: u32,
        side3: u32,
    }

    impl Triangle {
        fn perimeter(&self) -> u32 {
            self.side1 + self.side2 + self.side3
        }
    }

    let rect1 = Rectangle { width: 3, height: 4 };
    let triangle = Triangle { side1: 3, side2: 4, side3: 5 };
    let rect2 = Rectangle { width: 2, height: 5 };

    println!("{}", rect1.perimeter());
    println!("{}", triangle.perimeter());
    println!("{}", rect2.perimeter());
}
```
