```rust
fn main() {
    let values = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut largest_value = i32::MIN;

    for &value in &values {
        if value > 0 {
            positive_count += 1;
        }
        if value > largest_value {
            largest_value = value;
        }
    }

    println!("There are {} positive values.", positive_count);
    println!("The largest value is {}.", largest_value);
}
```
