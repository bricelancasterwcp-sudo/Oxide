fn str_from_int(n: Int) -> Str {
    if n == 0 {
        "0"
    } else {
        let mut digits = vec()
        let mut n = n
        while n > 0 {
            digits = push(digits, char_from_int(n % 10))
            n = n / 10
        }
        let mut s = Str()
        for i in range(0, len(&digits)) {
            s = s + chars(digits).get(i)
        }
        s
    }
}
