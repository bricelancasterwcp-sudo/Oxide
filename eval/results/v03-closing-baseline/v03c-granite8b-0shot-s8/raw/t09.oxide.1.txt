struct List { head: Int, tail: Option<Box<List>> }

fn main() {
    let list = List { head: 5, tail: Some(Box { head: 4, tail: Some(Box { head: 3, tail: Some(Box { head: 2, tail: Some(Box { head: 1, tail: None }) }) }) }) }
    print(list)
    print(print_list(list))
}

fn print_list(list: List) -> Str {
    match list.tail {
        Some(boxcdr) => {
            let rest = print_list(cdr)
            concat(concat(concat(int_to_str(list.head), str_len("\n")), rest), str_len("\n"))
        },
        None => int_to_str(list.head),
    }
}
