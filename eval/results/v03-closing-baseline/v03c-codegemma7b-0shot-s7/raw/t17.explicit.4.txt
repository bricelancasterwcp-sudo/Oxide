fn main() {
    let v = vec(10, 20, 30)
    print(get(&v, 1).unwrap_or(-1))
    print(get(&v, 5).unwrap_or(-1))
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < v.len() {
        Some(v.get(i).unwrap())
    } else {
        None
    }
}

fn vec(x: Int, y: Int, z: Int) -> Vec<Int> {
    return vec(x, y, z)
}

fn unwrap_or(x: Option<Int>, y: Int) -> Int {
    if letSome(x) == x {
        return x
    } else {
        return y
    }
}

fn len(x: &Vec<Int>) -> Int {
    return x.len()
}
