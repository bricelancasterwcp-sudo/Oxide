fn is_prime(x: u32) -> bool {
    if x <= 1 {
        return false;
    }
    for i in 2..=(x as f64).sqrt() as u32 {
        if x % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let mut count = 0;
    for i in 0..100 {
        if is_prime(i) {
            count += 1;
        }
    }
    println!("{}", count);
}
