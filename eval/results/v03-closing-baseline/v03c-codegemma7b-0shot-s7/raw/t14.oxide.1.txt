fn main() {
    let word = "mississippi".to_string()
    let mut count = 0
    let mut distinct = 0
    for c in word.chars() {
        if c == "s".chars().next().unwrap() {
            count = count + 1
        }
        if distinct == 0 {
            distinct = distinct + 1
        }
    }
    print(count)
    print(distinct)
}
