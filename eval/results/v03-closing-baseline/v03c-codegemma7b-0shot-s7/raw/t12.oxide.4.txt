fn reverse_string(text: Str) -> Str {
    let mut result = vec()
    for c in text.chars().rev() {
        result.push(c)
    }
    string_from(result)
}

fn string_from(v: Vec<Char>) -> Str {
    let mut result = ""
    for c in v.iter().rev() {
        result = c + result
    }
    result
}

fn main() {
    let text = "stack"
    print(reverse_string(text))
}
