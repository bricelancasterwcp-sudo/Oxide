fn main() {
    let mut nums = vec(3, 8, -2, 12, 7)
    let mut count = 0
    let mut max = nums.first()
    for x in &nums {
        if x > &mut * max {
            max = x
        }
        if x > &mut * 0 {
            count = count + 1
        }
    }
    print(count)
    print(max)
    drop nums
}
