```rust
fn main() {
    let text = "mississippi";

    let mut s_count = 0;
    let mut distinct_chars = HashSet::new();

    for c in text.chars() {
        if c == 's' {
            s_count += 1;
        }
        distinct_chars.insert(c);
    }

    println!("'s' occurs {} times", s_count);
    println!("There are {} distinct characters", distinct_chars.len());
}
```
