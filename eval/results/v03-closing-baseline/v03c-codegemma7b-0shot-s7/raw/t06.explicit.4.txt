fn is_perfect(num: Int) -> Bool {
    let mut sum = 0
    for i in 1.to(num - 1) {
        if num % i == 0 {
            sum = sum + i
        }
    }
    return sum == num
}

fn main() {
    for i in 1.to(1000) {
        if is_perfect(i) {
            print(i)
            print_str("\n")
        }
    }
}
