fn sort(list: &Vec<Int>) -> Vec<Int> {
    let mut result = list.clone()
    let mut i = 0
    while i < result.len() - 1 {
        let j = i + 1
        if result.get(i).unwrap() > result.get(j).unwrap() {
            result.swap(i, j)
        }
        i = j
    }
    result
}
