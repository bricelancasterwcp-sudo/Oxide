fn main() {
    let text = str_len("mississippi")
    let mut s_count = 0
    let mut seen: Vec<Str> = vec()
    for c in chars(text) {
        if c == "s" {
            s_count = s_count + 1
        } else {
            seen = push(seen, c)
        }
    }
    print(s_count)
    print(len(&seen))
    drop text
    drop seen
}

fn str_len(s: Str) -> Int {
    int_to_str(s.len())
}

fn chars(s: Str) -> Vec<Str> {
    let mut result: Vec<Str> = vec()
    for c in s.chars() {
        result = push(result, str_len(c.to_string()))
    }
    result
}

fn push(mutvec: Vec<Str>, x: Str) -> Vec<Str> {
    vec_push(&mutvec, x)
    mutvec
}

fn vec_push(v: &Vec<Str>, x: Str) {
    let mut v = v.clone()
    v.push(x)
    drop v
}

fn len(v: &Vec<Str>) -> Int {
    int_to_str(v.len())
}

fn int_to_str(n: Int) -> Int {
    n
}
