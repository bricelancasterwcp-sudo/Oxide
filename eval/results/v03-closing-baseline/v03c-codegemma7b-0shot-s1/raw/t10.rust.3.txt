use std::collections::LinkedList;

fn main() {
    let mut values = LinkedList::new();
    values.push_back(4);
    values.push_back(1);
    values.push_back(7);
    values.push_back(3);
    values.push_back(9);
    values.push_back(2);

    let mut kept_count = 0;

    for mut value in values.iter_mut() {
        if *value > 3 {
            println!("{}", value);
            kept_count += 1;
            *value = 4; // Replace with any value you want to keep
        }
    }

    println!("{} values kept", kept_count);
}
