fn main() {
    let pen = pen(3, 4)
    let pad = pad(9, 2)
    let ink = ink(5, 3)
    let all_items = push(push(push(vec(), pen), pad), ink)
    let cost = sum_cost(&all_items)
    print(cost)
    let most_expensive = most_expensive(&all_items)
    print_str(most_expensive)
}

fn pen(unit_price: Int, quantity: Int) -> Item {
    Item(unit_price, quantity)
}

fn pad(unit_price: Int, quantity: Int) -> Item {
    Item(unit_price, quantity)
}

fn ink(unit_price: Int, quantity: Int) -> Item {
    Item(unit_price, quantity)
}

fn sum_cost(items: &Vec<Item>) -> Int {
    let mut total = 0
    for item in items {
        total = total + cost(item)
    }
    total
}

fn cost(item: &Item) -> Int {
    item.unit_price * item.quantity
}

fn most_expensive(items: &Vec<Item>) -> Str {
    let mut most_expensive = "None"
    let mut max_cost = 0
    for item in items {
        let current_cost = cost(item)
        if current_cost > max_cost {
            max_cost = current_cost
            most_expensive = item.name
        }
    }
    most_expensive
}

struct Item {
    unit_price: Int,
    quantity: Int,
    name: Str,
}

fn item(unit_price: Int, quantity: Int, name: Str) -> Item {
    Item(unit_price, quantity, name)
}

fn get(v: Vec<T>, i: Int) -> Option<T> {
    if (i >= 0 && i < len(&v)) {

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a.to(b - 1) {
        v = push(v, i)

}

    let mut v = vec()
    for i in 0.to(str_len(&s) - 1) {

    }
