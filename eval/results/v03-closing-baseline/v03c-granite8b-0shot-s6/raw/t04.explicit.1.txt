fn main() {
    let mut count: Int = 0
    let mut i: Int = 2
    while i < 100 {
        let mut j: Int = 2
        let mut is_prime: Bool = true
        while j < i {
            if i % j == 0 {
                is_prime = false
                break
            }
            j = j + 1
        }
        if is_prime {
            count = count + 1
        }
        i = i + 1
    }
    print(count)
}
