struct Pen { price: Int, quantity: Int }

struct Pad { price: Int, quantity: Int }

struct Ink { price: Int, quantity: Int }

fn cost(item: Pen) -> Int {
    item.price * item.quantity
}

fn total_cost(items: Vec<Pen>) -> Int {
    let mut total = 0
    for item in items {
        total = total + cost(item)
    }
    total
}

fn highest_cost(items: Vec<Pen>) -> Option<Pen> {
    let mut max_cost = -1
    let mut best_item = Pen { price: 0, quantity: 0 }
    for item in items {
        let c = cost(item)
        if c > max_cost {
            max_cost = c
            best_item = item
        }
    }
    if max_cost == -1 {
        None
    } else {
        Some(best_item)
    }
}
