fn main() {
    let mut v = vec(10, 20, 30)
    print(get(&v, 1))
    if get(&v, 5).is_none() {
        print(-1)
    }
    drop_in_place(v)
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < v.len() {
        Some(&v)
    } else {
        None
    }
}

fn len(v: &Vec<Int>) -> Int {
    v.len()
}

fn push(mutv: Vec<Int>, x: Int) -> Vec<Int> {
    mutv.push(x)
    mutv
}

fn vec(i: Int, ii: Int, iii: Int) -> Vec<Int> {
    vec(i, ii, iii)
}

fn print(i: Int) {
    print(i)
}

fn is_none(i: Option<Int>) -> Bool {
    i.is_none()
}

fn unwrap(i: Option<Int>) -> Int {
    i.unwrap()
}

fn is_some(i: Option<Int>) -> Bool {
    i.is_some()
}

fn drop_in_place(v: Vec<Int>) {
    drop v
}

fn main() {
    let mut v = vec(10, 20, 30)
    print(get(&v, 1))
    if get(&v, 5).is_none() {
        print(-1)
    }
    drop_in_place(v)
}
