use std::io;

fn main() {
    let mut sum = 0;
    let mut count = 0;

    for line in io::stdin().lines() {
        match line.unwrap().parse::<i32>() {
            Ok(n) => {
                sum += n;
                count += 1;
            }
            Err(_) => (),
        }
    }

    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
