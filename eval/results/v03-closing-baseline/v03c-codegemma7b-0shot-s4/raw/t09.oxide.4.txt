    let mut list = vec(1, 2, 3, 4, 5)
    while len(list) > 0 {
        print(get(list, len(list) - 1)?)
        list = pop(pop(list)?)?

fn pop(v: Vec<Int>) -> Option<Vec<Int>> {
    if len(v) == 0 {
        None
    } else {
        Some(vec(range(0, len(v) - 1)?, v.last()?))
    }
}

    if i < len(v) {
