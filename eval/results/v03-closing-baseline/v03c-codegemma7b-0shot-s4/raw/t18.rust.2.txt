use std::fmt::Display;

struct Rectangle {
    width: u32,
    height: u32,
}

impl Display for Rectangle {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        write!(f, "Perimeter of rectangle: {}", 2 * (self.width + self.height))
    }
}

struct Triangle {
    side1: u32,
    side2: u32,
    side3: u32,
}

impl Display for Triangle {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        write!(f, "Perimeter of triangle: {}", self.side1 + self.side2 + self.side3)
    }
}

fn main() {
    let rectangle = Rectangle { width: 3, height: 4 };
    let triangle = Triangle { side1: 3, side2: 4, side3: 5 };
    let another_rectangle = Rectangle { width: 2, height: 5 };

    println!("{}", rectangle);
    println!("{}", triangle);
    println!("{}", another_rectangle);
}
