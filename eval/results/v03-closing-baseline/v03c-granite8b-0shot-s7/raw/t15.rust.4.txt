fn main() {
    let numbers = ["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;
    
    for number in numbers {
        match number.parse::<i32>() {
            Ok(num) => {
                sum += num;
            }
            Err(_) => {
                count += 1;
            }
        }
    }
    
    println!("Sum of valid numbers: {}", sum);
    println!("Count of invalid numbers: {}", count);
}
