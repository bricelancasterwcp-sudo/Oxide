fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= len(&v) {
        return None
    }

}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let v_len = len(&v)
    let v_capacity = v_len + 1
    let new_v = vec_with_capacity(&v_capacity)
    for i in range(0, v_len) {
        let v_i = get(&v, i)
        if v_i == None {
            return None
        }
        push(new_v, v_i)
    }
    push(new_v, x)
}
