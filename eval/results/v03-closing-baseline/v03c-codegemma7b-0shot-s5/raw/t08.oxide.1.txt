fn main() {
    let mut nums = vec(3, 8, -2, 12, 7)
    let mut positives = 0
    let mut largest = nums.get(0)?
    for x in nums {
        if x > 0 {
            positives = positives + 1
        }
        if x > largest {
            largest = x
        }
    }
    print(positives)
    print(largest)
}
