fn is_prime(x: Int) -> Bool {
    if x <= 1 {
        return false
    }
    for n in 2.to_int() + 1.to_int() {
        if x % n == 0 {
            return false
        }
    }
    true
}

fn main() {
    let mut count = 0
    for x in 2.to_int() + 1.to_int() {
        if is_prime(x) {
            count = count + 1
        }
    }
    print(count)
    drop count
}
