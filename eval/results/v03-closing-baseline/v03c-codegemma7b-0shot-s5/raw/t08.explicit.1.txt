fn main() {
    let nums = vec(3, 8, -2, 12, 7)
    let mut positive = 0
    let mut largest = None
    for x in &nums {
        if x > &mut && 0 {
            positive = positive + 1
        }
        if largest < &mut * &x {
            largest = Some(x)
        }
    }
    print(positive)
    print(largest)
}
