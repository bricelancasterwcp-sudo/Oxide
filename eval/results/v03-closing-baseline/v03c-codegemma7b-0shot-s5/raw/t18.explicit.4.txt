fn rect_area(a: Int, b: Int) -> Int {
    a * 2 + b * 2
}

fn triangle_area(a: Int, b: Int, c: Int) -> Int {
    let s = (a + b + c) / 2
    (s * (s - a) * (s - b) * (s - c)).sqrt()
}

fn main() {
    let r1 = rect_area(3, 4)
    print(r1)
    drop r1
    let t = triangle_area(3, 4, 5)
    print(t)
    drop t
    let r2 = rect_area(2, 5)
    print(r2)
    drop r2
}

fn rect_perimeter(a: Int, b: Int) -> Int {
    a * 2 + b * 2
}

fn triangle_perimeter(a: Int, b: Int, c: Int) -> Int {
    a + b + c
}

fn main() {
    let r1 = rect_perimeter(rect_area(3, 4))
    print(r1)
    drop r1
    let t = triangle_perimeter(triangle_area(3, 4, 5))
    print(t)
    drop t
    let r2 = rect_perimeter(rect_area(2, 5))
    print(r2)
    drop r2
}
