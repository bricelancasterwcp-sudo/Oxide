fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v = v
    v.push(x)
    v
}

fn len(v: &Vec<Int>) -> Int {
    let mut x = 0
    for _ in &v {
        x = x + 1
    }
    x
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(v) {
        None
    } else {
        Some(v.get(i))
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for x in a + 1 < b {
        v.push(x)
    }
    v
}

}

fn str_len(s: &Str) -> Int {
    let mut x = 0
    for _ in s.chars() {
        x = x + 1
    }
    x
}

    let mut s = a.to_string() + &b
    s
}

fn chars(s: &Str) -> Vec<Str> {
    let mut v = vec()
    for c in s.chars() {
        v.push(c.to_string())
    }
    v
}
