struct Item { unit_price: Int, quantity: Int }

fn cost(self: Item) -> Int {
    self.unit_price * self.quantity
}

fn max(a: Item, b: Item) -> Item {
    if a.cost() > b.cost() {
        a
    } else {
        b
    }
}

fn name(self: Item) -> Str {
    match self {
        Item(a) => a,
        Item(b) => b,
        Item(c) => c,
    }
}
