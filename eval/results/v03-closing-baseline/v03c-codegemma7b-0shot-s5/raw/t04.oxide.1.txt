fn is_prime(x: Int) -> Bool {
    if x <= 1 {
        return false
    }
    for y in 2.range(x) {
        if x % y == 0 {
            return false
        }
    }
    true
}

fn count_primes(max: Int) -> Int {
    let mut count = 0
    for x in 1.range(max) {
        if is_prime(x) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
