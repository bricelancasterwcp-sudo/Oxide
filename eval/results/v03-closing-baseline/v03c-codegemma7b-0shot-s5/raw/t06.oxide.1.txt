fn sum_of_divs(n: Int) -> Int {
    let mut acc = 0
    for x in 1.range(n) {
        if n % x == 0 {
            acc = acc + x
        }
    }
    acc
}

fn is_perfect(n: Int) -> Bool {
    sum_of_divs(n) == n
}

fn main() {
    for x in 1.range(1000) {
        if is_perfect(x) {
            print(x)
            print_str("\n")
        }
    }
}
