fn count_and_max(v: &Vec<Int>) -> Tuple<Int, Int> {
    let mut count = 0
    let mut max = -999999
    for x in &v {
        if x > 0 {
            count = count + 1
        }
        if x > max {
            max = x
        }
    }
    Tuple(count, max)
}

fn main() {
    let nums = push(push(push(push(vec(), 3), 8), -2), 12)
    nums = push(nums, 7)
    let result = count_and_max(&nums)
    print(result.fst())
    print_str("\n")
    print(result.snd())
    drop nums
}
