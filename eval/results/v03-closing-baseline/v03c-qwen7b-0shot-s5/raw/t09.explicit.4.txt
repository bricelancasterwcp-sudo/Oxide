fn main() {
    let nums = push(push(push(push(vec(), 1), 2), 3), 4, 5)
    for i in range(0, len(&nums)) {
        print(get(&nums, len(&nums) - 1 - i))
    }
    drop nums
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new_v = v.clone()
    new_v.push(x)
    new_v
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    v.get(i).cloned()
}

fn range(a: Int, b: Int) -> Vec<Int> {
    (a + 1).min(b).down_to(1).collect()
}

fn len(v: &Vec<Int>) -> Int {
    v.len()
}

fn print(x: Int) {
    println(x)
}

fn clone(v: &Vec<Int>) -> Vec<Int> {
    v.clone()
}

fn drop_name(name: Vec<Int>) {
    drop name
}

fn main() {
    let nums = push(push(push(push(vec(), 1), 2), 3), 4, 5)
    for i in range(0, len(&nums)) {
        print(get(&nums, len(&nums) - 1 - i))
    }
    drop_name(nums)
}
