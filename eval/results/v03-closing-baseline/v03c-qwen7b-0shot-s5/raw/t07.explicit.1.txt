    let mut multiples = Vec(Int)
    let limit = 24
    for i in range(1, 9) {

    let sum = sum_big(&multiples, 0)
    print(sum)
    drop multiples
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    concat(v, vec(x))
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut vec = Vec(Int)
    for i in a.while_(i < b) {
        vec = push(vec, i)
        i = i + 1
    }
    vec
}

fn vec() -> Vec<Int> {
    Vec(Int)
}

fn concat(a: Vec<Int>, b: Vec<Int>) -> Vec<Int> {
    let mut result = Vec(Int)
    for x in &a {
        result = push(result, x)
    }
    for x in &b {
        result = push(result, x)
    }
    result
}

fn int_to_str(n: Int) -> Str {
    let digits = vec("0123456789")
    let mut result = Str
    let mut n_copy = n
    while n_copy > 0 {
        let digit = n_copy % 10
        n_copy = n_copy / 10
        let char = digits.get(digit)
        result = concat(vec(char), result)
    }
    result
}
