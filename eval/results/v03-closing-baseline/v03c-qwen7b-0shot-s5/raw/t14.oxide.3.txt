struct Reading { label: Str, values: Vec<Str> }

fn count_s(text: Str) -> Int {
    let count = 0
    for c in chars(text) {
        if c == "s" {
            count = count + 1
        }
    }
    count
}

fn distinct_chars(text: Str) -> Int {
    let mut chars_map = Reading { label: "map", values: vec() }
    for c in chars(text) {
        let c_str = int_to_str(c)
        if get(chars_map.values, c_str) == None {
            push(chars_map.values, c_str)
        }
    }
    len(chars_map.values)
}

fn main() {
    let text = "mississippi"
    print(count_s(text))
    let distinct_count = distinct_chars(text)
    print(distinct_count)
}
