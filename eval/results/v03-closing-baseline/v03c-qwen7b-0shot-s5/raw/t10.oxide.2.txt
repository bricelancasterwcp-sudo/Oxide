fn main() {
    let numbers = vec(4, 1, 7, 3, 9, 2)
    let filtered: Vec<Int> = filter(numbers, x || x > 3)
    for n in filtered {
        print(n)
        print_str("\n")
    }
    print(len(filtered))
    print_str("\n")
}
