fn main() {
    let c = 12
    let x = String::from("x")
    let s = "30"
    if parse_int(&c) && parse_int(&x) && parse_int(&s) {
        println!("{}", sum(c, x, s))
        println!("{}", count_failures())
    } else {
        println!("Invalid input")
    }
}

fn sum(x: Int, y: Int, z: Int) -> Int {
    x + y + z
}

fn parse_int(s: &String) -> Option<Int> {
    match s.parse::<i32>() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn count_failures() -> u32 {
    let mut count = 0;
    for c in "12x30".chars() {
        if parse_int(&c) == None {
            count += 1
        }
    }
    count
}
