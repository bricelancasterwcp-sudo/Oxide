```
struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}
```

```python
class Reading:
    def __init__(self, label: str, values: list):
        self.label = label
        self.values = values

def first_big(self):
    found = -1
    for x in self.values:
        if x > 10:
            found = x
            break
    return found

def second(self):
    x = self.get(1)
    if x is not None:
        return x + 1
    else:
        return None

def main():
    r = Reading("lab", [3, 42])
    print(f"First big number: {r.first_big()}")
    print(f"Second number: {r.second()}")
```
