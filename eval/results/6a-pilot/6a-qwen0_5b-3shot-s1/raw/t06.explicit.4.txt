fn main() {
    for i in 2..=1000 {
        if is_perfect_number(i) {
            print!("{}", i);
        }
    }

    let mut perfect_numbers = vec![];
    for i in 2..=1000 {
        if is_perfect_number(i) {
            perfect_numbers.push(i);
        }
    }

    perfect_numbers.sort_unstable();
    for x in &perfect_numbers {
        println!("{}", x);
    }
}

fn is_perfect_number(n: u32) -> bool {
    let mut sum = 0;
    for i in 1..=n / 2 {
        if n % i == 0 {
            sum += i;
        }
    }
    sum == n
}
