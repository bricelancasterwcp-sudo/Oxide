fn main() {
    let mut v = push(push(push(vec![27], 1), 3), 42)
    let mut sum = 0;
    let mut steps = 0;

    while !v.is_empty() {
        if v.pop().unwrap() % 2 == 0 {
            sum += v.pop().unwrap();
        } else {
            sum += v.pop().unwrap() * 3 + 1;
        }
        steps += 1;
    }

    print(sum)
    print(steps)
}
