fn main() {
    let n = 100;
    let mut count = 0;
    for i in 2..n {
        if is_prime(i) {
            count += 1;
        }
    }
    println!("{}", count);
}

fn is_prime(num: isize) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..=((num as f64).sqrt() as isize + 1) {
        if num % i == 0 {
            return false;
        }
    }
    true
}
