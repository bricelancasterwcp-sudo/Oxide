fn main() {
    let mut v = vec![1];
    for i in &mut v {
        *i *= i;
    }
    print(sum());
    print(min());
}

fn sum() -> Int {
    let mut sum = 0;
    for x in v.iter() {
        sum += x;
    }
    sum
}

fn min() -> Option<Int> {
    if v.is_empty() {
        None
    } else {
        Some(v.iter().min().unwrap())
    }
}
