fn is_reverse(s: &str) -> bool {
    if s.len() == 0 { return true; }
    let mut i = 0;
    let mut j = s.len() - 1;
    while i < j {
        if s[i] != s[j] { return false; }
        i += 1;
        j -= 1;
    }
    true
}

fn main() {
    print!("{}", is_reverse("level")); // true
    println!("{}", is_reverse("stereo")); // true
    println!("{}", is_reverse("rotor")); // true
}
