fn main() {
    for n in 2..1000 {
        let sum = n as i64 * (n - 1) / 2;
        if is_perfect_sum(sum) {
            println!("{}", n);
        }
    }

    fn is_perfect_sum(num: i64) -> bool {
        let mut divisors = vec![];
        for i in 1..num {
            if num % i == 0 {
                divisors.push(i);
            }
        }
        divisors.iter().sum() == num
    }
}
