fn is_perfect_number(n: isize) -> bool {
    let mut sum = 1;
    for i in 2..=n / 2 {
        if n % i == 0 {
            sum += i;
            if i != n / i {
                sum += n / i;
            }
        }
    }
    sum == n
}
