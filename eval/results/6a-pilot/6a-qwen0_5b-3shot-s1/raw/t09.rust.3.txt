fn main() {
    let mut p = 1;
    for i in 1..7 {
        p *= i;
    }
    println!("{}", p);
    
    let v = vec![2, 5, 1, 6];
    let sum: i64 = v.iter().sum();
    let min = v.iter().min().unwrap();
    println!("{}", sum);
    println!("{}", min);
    
    // Corrected version
    for num in v {
        print!("{num}, ");
    }
    println!("");
    println!("{}", min);
}
