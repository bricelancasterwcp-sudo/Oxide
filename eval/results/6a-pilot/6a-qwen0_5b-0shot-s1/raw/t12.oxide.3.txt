struct Reading { label: Str, values: Vec<Int> }

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    v.push(x)
}

fn main() {
    let r = Reading { label: "lab", values: push(vec(), 3), }
    println(first_big(r.values))
}

fn first_big(v: Vec<Int>) -> Int {
    let mut max_value = -1;
    for x in v {
        if x > max_value {
            max_value = x;
        }
    }
    max_value
}
