    let one_hundred = 100;
    for divisor in [5, 0, 4] {
        match divisor {
            0 => println!("undefined"),
            _ => {
                // Corrected division by a float
                let quotient = (one_hundred as f64) / divisor as f64;
                if quotient == f64::INFINITY { // Check if division would result in infinity
                    println!("Division by zero is undefined.");
                } else {
                    println!("{}", quotient);
                }
            }
        }
    }
}
