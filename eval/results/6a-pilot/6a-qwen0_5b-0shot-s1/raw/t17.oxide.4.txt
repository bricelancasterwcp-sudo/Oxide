struct Reading { label: Str, values: Vec<Int> }

impl Reading {
    fn first_big(&self) -> Int {
        let mut found = -1;
        for x in self.values.iter() {
            if *x > 10 {
                found = *x;
                break;
            }
        }
        if found == -1 {
            return -1; // element not found, return -1
        } else {
            print!("{}", found);
        }
    }

    fn second(&self) -> Option<Int> {
        let x = self.values.get(1).cloned();
        Some(x + 1)
    }
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) };
    first_big(r);
    let r2 = Reading { label: "lab2", ..r };
    match second(r2.values) {
        Some(n) => print!("{}", n),
        None => print_str("too short"),
    }
}
