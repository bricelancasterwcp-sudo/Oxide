fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let mut total = 0;
    for x in &v {
        if x > limit {
            total = total + x.pow(2);
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

    println(sum_big(&nums, 10));
}
