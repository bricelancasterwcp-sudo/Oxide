fn main() {
    let (level, stereo, rotor) = ("level".to_string(), "stereo".to_string(), "rotor".to_string());
    println!("{}", is_palindrome(level)); // true
    println!("{}", is_palindrome(stereo)); // false
    println!("{}", is_palindrome(rotor)); // false
}

fn is_palindrome(s: String) -> bool {
    let mut i = 0;
    let mut j = s.len() - 1;
    while i < j {
        if s[i] != s[j] {
            return false;
        }
        i += 1;
        j -= 1;
    }
    true
}
