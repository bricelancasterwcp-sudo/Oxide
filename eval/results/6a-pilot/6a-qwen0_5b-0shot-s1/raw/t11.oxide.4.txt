struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 10 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?;
    Some(x + 1)
}

fn main() {
    let mut r = Reading { label: "lab", values: push(vec![3, 8, 1, 9, 2], 0) };
    print(first_big(r.values));            // field access copies: r stays usable
    println!("second:", second(r.values).unwrap())
}

#[derive(Debug)]
struct Point {
    x: f64,
    y: f64,
}

impl Point {
    fn new(x: f64, y: f64) -> Self {
        Point { x, y }
    }

    fn add(&mut self, other: Point) {
        self.x += other.x;
        self.y += other.y;
    }

    fn subtract(&mut self, other: Point) {
        self.x -= other.x;
        self.y -= other.y;
    }

    fn multiply(&mut self, scalar: f64) {
        self.x *= scalar;
        self.y *= scalar;
    }
}

fn print(x: i32) {
    println!("{:?}", x);
}

fn str_len(s: &str) -> usize {
    s.len()
}

fn concat(a: &str, b: &str) -> String {
    a.to_string() + &b
}

fn chars(s: &str) -> Vec<char> {
    s.chars().collect()
}

fn int_to_str(n: i32) -> String {
    n.to_string()
}

fn parse_int(s: &str) -> Option<i32> {
    if let Ok(num) = s.parse::<i32>() {
        Some(num)
    } else {
        None
    }
}

fn to_float(n: f64) -> f64 {
    n
}

fn trunc(x: f64) -> i32 {
    x.trunc() as i32
}
