```
use std::char;

fn main() {
    let str_vec = vec![
        "12".to_string(),
        "x".to_string(),
        "30".to_string()
    ];
    
    for str in &str_vec {
        if char::is_digit(str.chars().next().unwrap()) {
            let num = str.parse::<i32>().unwrap();
            println!("Sum: {}", num);
            print!("{}", num == 120 ? "Success" : "Fail");
        }
    }
}
```
