fn main() {
    let mut prime_count = 0;
    
    for i in 2..=99 {
        if is_prime(i) {
            prime_count += 1;
        }
    }
    
    println!("There are {} prime numbers less than 100.", prime_count);
}

fn is_prime(n: isize) -> bool {
    if n <= 1 {
        return false;
    }
    for i in 2..=((n as f64).sqrt() as isize) {
        if n % i == 0 {
            return false;
        }
    }
    true
}
