```
use std::cmp::Ordering;

struct Reading {
    label: Str,
    values: Vec<Int>,
}

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 10 {
            found = x;
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let mut x = get(v, 1)?;
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) };
    print(first_big(r.values));            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }

    let x1 = 5;
    let x2 = 0;
    let x3 = 4;

    println!("Division of 100 by each of 5, 0, and 4:");
    for i in 0..=4 {
        print!("{}/{}", x1 / (i + 1), x2 == 0 && x3 != 0 ? "undefined" : x3);
    }
}
```

### Explanation:

- **Reading Struct**: A struct `Reading` is defined with fields `label` and `values`.
- **first_big Function**: This function iterates through the vector of integers to find the first number greater than or equal to 10.
- **second Function**: This function finds the second largest number in the vector.
- **main Function**: Demonstrates how to use these functions. It prints the division results for different values of `x` and ensures that each operation is correctly handled.
