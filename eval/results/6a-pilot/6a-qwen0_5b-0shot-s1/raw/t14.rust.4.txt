fn main() {
    let text = String::from("mississippi");
    
    // Count occurrences of 's'
    let count_s = text.chars().filter(|&c| c == 's').count();
    
    // Use a HashSet to store distinct characters
    let mut distinct_chars: HashSet<char> = text.chars().collect();
    
    // Print the counts
    println!("Character 's' appears {} times.", count_s);
    println!("Distinct characters in the string are: {:?}", distinct_chars);
}
