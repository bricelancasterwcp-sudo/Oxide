```
fn main() {
    let data = vec!["12", "x", "30"];
    let mut success_count = 0;
    let mut failure_count = 0;

    for &str_val in data {
        match str_val.parse::<i32>() {
            Ok(val) => {
                println!("Parsed: {}", val);
                success_count += 1;
            }
            Err(_) => {
                println!("Failed to parse: {}", str_val);
                failure_count += 1;
            }
        }
    }

    println!("Successes: {}", success_count);
    println!("Failures: {}", failure_count);
}
```

This program now compiles successfully, and it correctly parses "12", "x", and "30" into integers, counts them, and prints the results.
