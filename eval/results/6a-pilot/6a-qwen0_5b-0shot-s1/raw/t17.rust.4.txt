fn main() {
    let numbers: [i32; 3] = [10, 20, 30];

    println!("Element found at position 1: {}", find_element_at_position(&numbers, 1));
    println!("Element found at position 5: {}", find_element_at_position(&numbers, 5));
}

fn find_element_at_position(numbers: &[i32], index: usize) -> i32 {
    if index < 0 || index >= numbers.len() {
        -1
    } else {
        &numbers[index]
    }
}
