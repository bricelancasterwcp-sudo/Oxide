fn find_perfect(n: Int) -> Bool {
    let mut sum = 0;
    for i in range(1, n) {
        if n % i == 0 {
            sum = sum + i;
        }
    }
    sum == n
}

fn main() {
    for num in range(2, 1000) {
        if find_perfect(num) {
            print(&num);
            print_str("\n");
        }
    }
}
