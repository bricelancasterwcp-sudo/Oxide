fn main() {
    let mut i = 2;
    while i <= 1000 {
        let mut sum = 1;
        for m in 2..=i / 2 {
            if i % m == 0 {
                sum += m;
            }
        }
        if sum == i {
            print(i);
            print_str("\n");
        }
        i += 1;
    }
}
