struct List {
    head: Option<Box<Node>>,
}

struct Node {
    value: Int,
    tail: Box<List>,
}

fn sort(list: Box<List>) -> Box<List> {
    let mut sorted = Box::new(List { head: None });
    let mut current = list.head.take();

    while let Some(mut node) = current {
        current = node.tail.head.take();
        sorted = insert(sorted, node.value);
    }

    sorted
}

fn insert(mut list: Box<List>, value: Int) -> Box<List> {
    if list.head.is_none() || list.head.as_ref().unwrap().value > value {
        Box::new(List { head: Some(Box::new(Node { value, tail: list })) })
    } else {
        let mut current = &mut list.head;
        while let Some(ref mut next) = *current {
            if next.value >= value {
                break;
            }
            current = &mut next.tail.head.take();
        }
        current.replace(Some(Box::new(Node { value, tail: Box::from(next.unwrap().tail) })));
        list
    }
}
