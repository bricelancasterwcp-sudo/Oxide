```rust
struct Rectangle {
    width: u32,
    height: u32,
}

struct Triangle {
    side1: u32,
    side2: u32,
    side3: u32,
}

impl Rectangle {
    fn new(width: u32, height: u32) -> Rectangle {
        Rectangle { width, height }
    }

    fn perimeter(&self) -> u32 {
        2 * (self.width + self.height)
    }
}

impl Triangle {
    fn new(side1: u32, side2: u32, side3: u32) -> Triangle {
        Triangle { side1, side2, side3 }
    }

    fn perimeter(&self) -> u32 {
        self.side1 + self.side2 + self.side3
    }
}

fn main() {
    let rect1 = Rectangle::new(3, 4);
    println!("{}", rect1.perimeter());

    let triangle = Triangle::new(3, 4, 5);
    println!("{}", triangle.perimeter());

    let rect2 = Rectangle::new(2, 5);
    println!("{}", rect2.perimeter());
}
```
