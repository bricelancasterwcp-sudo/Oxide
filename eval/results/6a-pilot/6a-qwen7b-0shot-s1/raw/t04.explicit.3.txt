fn main() {
    let mut count = 0;
    for n in 2..100 {
        if is_prime(n) {
            count += 1;
        }
    }
    print(count);
}

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        false
    } else {
        let mut i = 2;
        while i * i <= n {
            if n % i == 0 {
                return false;
            }
            i += 1;
        }
        true
    }
}
