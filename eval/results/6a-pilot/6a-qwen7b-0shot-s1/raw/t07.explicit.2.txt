fn sum_multiples(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0;
    for x in v {
        if *x % 3 == 0 && *x <= limit {
            total = total + *x;
        }
    }
    total
}

fn main() {
    let multiples = vec();
    let mut result = 0;
    for i in range(1, 8) {
        let multiple = i * 3;
        if multiple <= 24 {
            result = push(result, multiple);
        }
    }
    print(sum_multiples(&result, 24));
    drop result
}
