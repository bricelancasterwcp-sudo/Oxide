```rust
fn main() {
    let strings = vec!["12", "x", "30"];
    let mut sum = 0;
    let mut fail_count = 0;

    for s in &strings {
        match parse_int(s) {
            Some(n) => sum += n,
            None => fail_count += 1,
        }
    }

    print(sum);
    print_str(" ");
    print(fail_count);
}
```
