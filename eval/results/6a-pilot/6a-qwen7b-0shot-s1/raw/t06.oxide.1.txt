fn main() {
    let mut i = 2;
    while i <= 1000 {
        let mut sum = 1;
        let j = trunc(to_float(i).to_int());
        let k = trunc(sqrt(j));
        for m in 2..=k {
            if j % m == 0 {
                sum += m + (if m != j / m { j / m } else { 0 });
            }
        }
        if sum == i {
            print(i);
            print_str("\n");
        }
        i = i + 1;
    }
}
