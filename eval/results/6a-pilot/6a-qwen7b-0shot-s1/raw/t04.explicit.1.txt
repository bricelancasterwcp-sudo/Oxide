fn main() {
    let mut count = 0;
    for n in 2..100 {
        if is_prime(n) {
            count = count + 1;
        }
    }
    print(count);
}

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false;
    }
    for i in 2..(n / 2 + 1) {
        if n % i == 0 {
            return false;
        }
    }
    true
}
