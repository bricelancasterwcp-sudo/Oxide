fn sum_multiples(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in v {
        if *x % 3 == 0 && *x <= limit {
            total = total + *x
        }
    }
    total
}

fn main() {
    let mut multiples = vec()
    for i in range(1, 8) {
        multiples = push(multiples, i * 3)
    }
    print(sum_multiples(&multiples, 24))
    drop multiples
}
