    let nums = vec_str("12", "x", "30")
    let sum = nums.iter().filter_map(parse_int).fold(0, add)
    print(sum)
    print(len(nums) - len(nums.iter().filter_map(parse_int).collect()))
}

fn vec_str(first: Str, rest: Str, others: Str) -> Vec<Str> {
    let mut v = vec(first)
    for s in split(rest, others, " ") {
        v.push(s)
    }
    v
}

fn split(first: Str, rest: Str, sep: Str) -> Vec<Str> {
    let mut v = vec(first)
    let mut rest = rest
    while letSome(pos) == find(rest, sep) {
        let token = rest.slice(0, pos)
        v.push(token)
        rest = rest.slice(pos + len(sep), len(rest))
    }
    v.push(rest)
    v
}

fn find(haystack: Str, needle: Str) -> Option<Int> {
    for i in 0.upto(len(haystack) - len(needle)) {
        if haystack.slice(i, len(needle)) == needle {
            return Some(i)
        }
    }
    None
}

fn parse_int(s: Str) -> Option<Int> {
    if s.chars().all(is_digit) {
        Some(int_to_str(s).parse().unwrap())
    } else {
        None
    }
}

    "0123456789".chars().contains(c)
}
