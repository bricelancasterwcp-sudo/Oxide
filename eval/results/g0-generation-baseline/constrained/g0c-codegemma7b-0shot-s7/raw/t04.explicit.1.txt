fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }
    for i in 2.to_int() + 1.to_int() {
        if n % i == 0 {
            return false
        }
    }
    return true
}

fn main() {
    let mut count = 0.to_int()
    for n in 1.to_int() + 1.to_int() {
        if is_prime(n) {
            count = count + 1.to_int()
        }
    }
    print(count)
}
