fn main() {
    let list = vec(3, 8, -2, 12, 7)
    let mut positive = 0
    let mut largest = -1000000000
    for x in list {
        if x > 0 {
            positive = positive + 1
        }
        if x > largest {
            largest = x
        }
    }
    print(positive)
    print(largest)
}

fn get(x: Vec<Int>, y: Int) -> Option<Int> {
    if y >= len(x) {
        return None
    } else {
        return Some(x.get(y).unwrap())
    }
}
