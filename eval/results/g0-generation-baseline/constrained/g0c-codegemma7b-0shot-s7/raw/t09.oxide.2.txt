fn main() {
    let mut numbers = vec(1, 2, 3, 4, 5)
    while len(numbers) > 0 {
        print(pop(numbers)?.last()?)
    }
}

fn pop(v: Vec<Int>) -> Option<Vec<Int>> {
    if len(v) == 0 {
        None
    } else {
        Some(vec(range(0, len(v) - 1)?, v.last()?))
    }
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(v) {
        None
    } else {
        Some(v.get(i)?)
    }
}

fn len(v: Vec<Int>) -> Int {
    let mut count = 0
    for _ in v {
        count = count + 1
    }
    count
}

fn vec(args: Int, elements: Int) -> Vec<Int> {
    let mut result = Vec { len: 0, ptr: 0 }
    for _ in range(0, elements) {
        result = push(result, args)
    }
    result
}

fn range(a: Int, b: Int) -> Option<Int> {
    if a >= b {
        None
    } else {
        Some(a)
    }
}

fn print(x: Int) {
    print_str(to_str(x))
}

fn print_str(s: Str) {
    for c in chars(s) {
        print(int_to_str(ord(c)))
    }
    print_str("\n")
}

fn to_str(n: Int) -> Str {
    if n < 10 {
        return Str { len: 1, ptr: n + 48 }
    }
    let mut result = Str { len: 0, ptr: 0 }
    let mut number = n
    while number >= 10 {
        let digit = number % 10
        result = push(result, digit)
        number = number / 10
    }
    result = push(result, number)
    result
}

fn ord(c: Str) -> Int {
    if c.len() != 1 {
        panic("ord called on multi-character string")
    }
    return get(c, 0)? - 48
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = Vec { len: 0, ptr: 0 }
    for i in range(0, s.len()) {
        result = push(result, Str { len: 1, ptr: s.ptr + i })
    }
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = v
    result.len == result.len + 1
    result.ptr == x
    result
}

fn to_float(n: Int) -> Float {
    return Float { mantissa: n, exponent: 0 }
}

fn trunc(x: Float) -> Int {
    if x.mantissa >= 0 {
        return int_to_str(x.mantissa)
    } else {
        return int_to_str(-x.mantissa - 1)
    }
}
