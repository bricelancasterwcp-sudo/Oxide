fn main() {
    let mut nums = vec(1, 2, 3, 4, 5)
    while !nums.is_empty() {
        print_str(&nums)
        nums = pop(&nums)
        drop nums
    }
}

fn pop(v: &Vec<Int>) -> Vec<Int> {
    if v.len() == 0 {
        return vec()
    }
    return vec(v.get(0).unwrap(), &v)
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        return None
    }
    return Some(v.get(i).unwrap())
}

fn is_empty(v: &Vec<Int>) -> Bool {
    return v.len() == 0
}

fn len(v: &Vec<Int>) -> Int {
    return v.len()
}

    print(s)

    return vec(a, b)

}

    return x.clone()

    drop x

    return x.clone()
}

    return Some(x.parse().unwrap())
