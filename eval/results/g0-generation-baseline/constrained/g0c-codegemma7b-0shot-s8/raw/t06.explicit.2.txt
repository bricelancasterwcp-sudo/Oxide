fn divisors(n: Int) -> Vec<Int> {
    let mut factors = vec()
    for i in 2.to_int().unwrap() + 1.to_int().unwrap() {
        if n % i == 0 {
            factors = push(factors, i)
        }
    }
    factors
}

fn is_perfect(n: Int) -> Bool {
    let factors = divisors(n)
    let sum = factors.iter().fold(0, add)
    sum == n
}

fn main() {
    for i in 1.to_int().unwrap() + 1.to_int().unwrap() {
        if is_perfect(i) {
            print(i)
            print_str("\n")
        }
    }
}
