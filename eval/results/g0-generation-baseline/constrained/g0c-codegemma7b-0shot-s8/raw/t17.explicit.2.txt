    let list = vec(10, 20, 30)
    print(get(&list, 1).unwrap_or(-1))
    print(get(&list, 5).unwrap_or(-1))
    drop list
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}
