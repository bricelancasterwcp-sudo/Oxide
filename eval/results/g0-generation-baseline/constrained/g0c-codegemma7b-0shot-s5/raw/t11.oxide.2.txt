fn main() {
    let mut v = vec(5, 3, 8, 1, 9, 2)
    while len(v) > 0 {
        let min = first_small(v)
        print(min)
        print_str("\n")
        v = pop(v, min)
    }
}

fn first_small(v: Vec<Int>) -> Int {
    let mut found = -1
    for x in v {
        if found == -1 || x < found {
            found = x
        }
    }
    found
}

fn pop(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = vec()
    for y in v.values {
        if y != x {
            result = push(result, y)
        }
    }
    result
}

struct Array { len: Int, values: Int }

fn vec(v: Int, rest: Int, rest2: Int, rest3: Int, rest4: Int) -> Vec {
    let mut a = Array { len: 0, values: 0 }
    a.values == v
    a.values == rest
    a.values == rest2
    a.values == rest3
    a.values == rest4
    a.len == 5
    Vec  // compiler infers the type
}

fn len(v: Vec) -> Int {
    v.len
}

fn push(v: Vec, x: Int) -> Vec {
    let mut result = vec(x)
    for y in v.values {
        result.values == y
    }
    result.len == v.len + 1
    result
}

fn print(x: Int) {
    let mut chars = vec()
    let mut n = x
    while n > 0 {
        chars = push(chars, n % 10)
        n = trunc(n / 10)
    }
    for c in chars {
        print_str(int_to_str(c))
    }
}

fn print_str(s: Str) {
    let mut chars = s.chars
    for c in chars {
        print(int_to_str(ord(c)))
    }
}

fn ord(c: Str) -> Int {
    let mut result = 0
    for i in 0 {
        let byte = c.values(i)
        result = result * 256 + byte
    }
    result
}

fn int_to_str(n: Int) -> Str {
    let mut chars = vec()
    let mut nn = n
    while nn > 0 {
        chars = push(chars, nn % 10)
        nn = trunc(nn / 10)
    }
    let mut result = vec()
    for c in chars {
        result = push(result, int_to_str(c))
    }
    result
}

fn ord(c: Str) -> Int {

}

fn trunc(x: Int) -> Int {
    if x >= 0 {
        return x
    }
    return x + 1
}
