fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut temp = v
    temp.push(x)
    temp
}

fn step_by(start: Int, step: Int) -> Int {
    start + step
}

fn take(v: Vec<Int>, n: Int) -> Vec<Int> {
    let mut temp = vec()
    for i in 0.to_int().max(n).to_int().min(v.len()) {
        temp = push(temp, get(v, i)?)
    }
    temp
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        None
    } else {
        Some(v.get(i))
    }
}

fn print(x: Int) {

fn print_str(s: Str) {
    print(to_float(parse_int(s)?))
}

fn int_to_str(n: Int) -> Str {
    if n == 0 {
        return "0".to_string()
    }
    let mut digits = vec()
    let mut x = n
    while x != 0 {
        digits = push(digits, trunc(x % 10))
        x = trunc(x / 10)
    }
    let mut s = "0".to_string()
    for d in digits {
        s = concat(s, int_to_str(d))
    }
    s
}

fn parse_int(s: Str) -> Option<Int> {
    let mut x = 0
    let mut neg = false
    for c in s.chars() {
        if c == "-" {
            neg = true
        } else if c >= "0".chars().nth(0).unwrap() && c <= "9".chars().nth(0).unwrap() {
            x = x * 10 + (c.to_digit(10).unwrap() - "0".to_digit(10).unwrap())
        } else {
            return None
        }
    }
    if neg {
        return Some(-x)
    } else {
        return Some(x)
    }
}

fn filter_by(f: FnMut, v: Vec<Int>) -> Vec<Int> {
