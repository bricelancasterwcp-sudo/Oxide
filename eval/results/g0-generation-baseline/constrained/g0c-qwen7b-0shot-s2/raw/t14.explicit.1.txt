    let distinct_chars = s.chars().collect_uniq().len()
    print(count_s)
    print(distinct_chars)
}

fn char_eq(bns: &Byte, c: &Byte) -> Bool {
    bns == c
}

fn collect_uniq(vec: Vec<Str>) -> Vec<Str> {
    let mut result = Vec()
    for x in vec {
        if !result.contains(&x) {
            result.push(x)
        }
    }
    result
}

    vec.iter().filter(char_eq(value)).count()
}
