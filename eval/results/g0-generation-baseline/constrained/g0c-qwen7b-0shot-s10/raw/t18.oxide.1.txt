struct Rectangle { width: Int, height: Int }

struct Triangle { side1: Int, side2: Int, side3: Int }

fn perimeter_of_rectangle(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn perimeter_of_triangle(t: Triangle) -> Int {
    t.side1 + t.side2 + t.side3
}

fn main() {
    let r1 = Rectangle { width: 3, height: 4 }
    let t = Triangle { side1: 3, side2: 4, side3: 5 }
    let r2 = Rectangle { width: 2, height: 5 }
    print(perimeter_of_rectangle(r1))
    print(perimeter_of_triangle(t))
    print(perimeter_of_rectangle(r2))
}
