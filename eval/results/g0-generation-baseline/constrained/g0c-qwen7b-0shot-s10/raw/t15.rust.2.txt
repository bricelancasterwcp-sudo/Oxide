fn main() {
    let strings = vec!["12", "x", "30"];
    let mut sum = 0;
    let mut failed = 0;

    for s in strings {
        match s.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => failed += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Failed: {}", failed);
}
