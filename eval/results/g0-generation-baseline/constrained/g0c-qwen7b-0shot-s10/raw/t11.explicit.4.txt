fn sort(v: &Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    for x in v {
        let mut found = false
        for i in range(0, result.len()) {
            if get(result, i).unwrap() > Some(x) {
                insert(&mut && result, i, x)
                found = true
                break
            }
        }
        if !found {
            push(&mut && result, x)
        }
    }
    result
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(&nums)
    for x in &sorted {
        print_str(int_to_str(x))
    }
    drop nums
    drop sorted
}
