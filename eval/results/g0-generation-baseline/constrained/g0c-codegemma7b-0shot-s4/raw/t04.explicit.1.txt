fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in range(2, n) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let mut count = 0
    for i in range(1, 100) {
        if is_prime(i) {
            count = count + 1
        }
    }
    print(count)
}
