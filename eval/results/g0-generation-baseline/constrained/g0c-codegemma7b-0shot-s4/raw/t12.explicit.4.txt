fn reverse_str(s: &Str) -> Str {
    let mut result = str_len(s) - 1
    for i in (0.to_int() + 1).to_int() {
        if i > result {
            break
        }
        print(chars(s).get(result).unwrap())
        result = result - 1
    }
    result
}

fn main() {
    let text = str_len("stack")
    print(reverse_str(&text))
    drop text
}
