fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }
    let mut i = 2
    while i * i <= n {
        if n % i == 0 {
            return false
        }
        i = i + 1
    }
    true
}

fn count_primes(max: Int) -> Int {
    let mut count = 0
    for i in 0.to(max) {
        if is_prime(i) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
