```rust
fn main() {
    let items = vec![
        ("pen".to_string(), 3. * 4.),
        ("pad".to_string(), 9. * 2.),
        ("ink".to_string(), 5. * 3.),
    ];

    let mut max_cost = f64::MIN;
    let mut best_item = String::new();
    for (name, cost) in items.iter() {
        if *cost > max_cost {
            max_cost = *cost;
            best_item = name.to_string();
        }
    }

    println!("{:?}", items);
    println!("{}", best_item);
}
```
