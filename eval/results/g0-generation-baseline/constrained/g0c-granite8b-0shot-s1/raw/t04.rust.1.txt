fn is_prime(n: usize) -> bool {
    if n <= 1 {
        return false;
    }
    for i in 2..=(n as f64).sqrt().ceil() as usize {
        if n % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let mut count = 0;
    for i in 2..100 {
        if is_prime(i) {
            count += 1;
        }
    }
    println!("{}", count);
}
