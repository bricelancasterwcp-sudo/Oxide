fn main() {
    let n = 1000
    let mut i = 1
    let mut sum = 0
    while i < n {
        let mut m = i
        let mut d = 1
        while d < m {
            if m % d == 0 {
                let x = m / d
                sum = sum + x
                m = x
            }
            d = d + 1
        }
        if m == 1 {
            sum = sum + i
        }
        i = i + 1
    }
    print(sum)
}
