```rust
fn main() {
    rectangle(3, 4);
    triangle(3, 4, 5);
    rectangle(2, 5);
}

fn rectangle(a: i32, b: i32) {
    println!("{}", 2 * (a + b));
}

fn triangle(a: i32, b: i32, c: i32) {
    let s = (a + b + c) / 2;
    println!("{}", 2 * f64::sqrt(s as f64 * (s - a) as f64 * (s - b) as f64 * (s - c) as f64).round() as i32);
}
```
