fn main() {
    let shape1 = Rectangle { width: 3, height: 4 };
    let shape2 = Triangle { a: 3, b: 4, c: 5 };
    let shape3 = Rectangle { width: 2, height: 5 };

    println!("{:?}", shape1.perimeter());
    println!("{:?}", shape2.perimeter());
    println!("{:?}", shape3.perimeter());
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn perimeter(&self) -> u32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    a: u32,
    b: u32,
    c: u32,
}

impl Triangle {
    fn perimeter(&self) -> u32 {
        self.a + self.b + self.c
    }
}
