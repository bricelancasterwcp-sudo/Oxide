fn main() {
    let sum = sum_of_divisors(2000)
    for i in range(2, sum) {
        if sum_of_divisors(i) == i {
            print(i)
        }
    }
}

fn sum_of_divisors(n: Int) -> Int {
    let mut total = 0
    for d in range(1, n) {
        if n % d == 0 {
            total = total + d
        }
    }
    total
}

fn is_prime(n: Int) -> Bool {
    n > 1 && sum_of_divisors(n) == n
}

fn sum_of_primes_before(n: Int) -> Int {
    let mut total = 0
    for i in range(2, n) {
        if is_prime(i) {
            total = total + i
        }
    }
    total
}

fn main() {
    print_str(sum_of_primes_before(2000).to_str())
}
