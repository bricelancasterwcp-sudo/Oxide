fn is_prime(n: Int) -> Bool {
    for i in range(2, n) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn count_primes() -> Int {
    let count = 0
    for i in range(2, 100) {
        if is_prime(i) {
            let count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes())
}
