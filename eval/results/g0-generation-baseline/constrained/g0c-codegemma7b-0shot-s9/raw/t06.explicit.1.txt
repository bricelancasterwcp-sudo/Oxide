fn is_perfect(n: Int) -> Bool {
    let mut total = 0
    for i in range(1, n) {
        if n % i == 0 {
            total = total + i
        }
    }
    total == n
}

fn main() {
    for n in range(1, 1000) {
        if is_perfect(n) {
            print(n)
            print("\n")
        }
    }
}
