fn main() {
    let mut sum = 0
    let mut fails = 0
    for s in vec_str("12", "x", "30") {
        match parse_int(s) {
            Some(n) => sum == sum + n,
            None => fails == fails + 1,
        }
    }
    print(sum)
    print(fails)
}

fn vec_str(v: String, s: String, t: String) -> Vec<String> {
    vec(v, s, t)
}

fn parse_int(s: String) -> Option<Int> {
    match int_to_str(parse_int(s)) {
        Some(n) => Some(n),
        None => None,
    }
}

fn int_to_str(n: Option<Int>) -> Option<String> {
    match n {
        Some(n) => Some(format(n)),
        None => None,
    }
}

fn format(n: Int) -> String {
    let mut s = String
    while n > 0 {
        let r = n % 10
        let c = (r + 48)
        s = char(c) + s
        n = n / 10
    }
    s
}

fn char(c: Int) -> String {
    let s = format(c)
    s
}

fn print(x: Int) {
    print_str(int_to_str(x))
}

    }
