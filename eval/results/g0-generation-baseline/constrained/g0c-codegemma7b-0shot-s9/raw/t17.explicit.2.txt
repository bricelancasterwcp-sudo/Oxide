fn int_to_str(n: Int) -> Str {
    let mut digits: Vec<Int> = vec(0)
    let mut num = n
    while num > 0 {
        digits.append(num % 10)
        num = num / 10
    }
    let mut result: Str = str(digits.len())
    for d in &digits {
        result = concat(result, str(d))
    }
    result
}

fn concat(a: Str, b: Str) -> Str {
    let mut result: Str = str(a.len() + b.len())
    for c in a {
        result = concat(result, str(c))
    }
    for c in b {
        result = concat(result, str(c))
    }
    result
}

fn drop_empty_list(v: &Vec<Int>) -> Vec<Int> {
    if v.len() == 0 {
        return vec()
    } else {
        return v.to_vec()
    }
}
