struct Fib {
    a: Int,
    b: Int,
}

fn fib(n: Int) -> Int {
    let mut f = Fib { a: 1, b: 1 }
    let mut i = 0
    while i < n {
        i = i + 1
        let tmp = f.b
        f.b == f.a + tmp
        f.a == tmp
    }
    f.b
}

struct Point {
    x: Int,
    y: Int,
}

struct Circle {
    center: Point,
    radius: Int,
}

struct Shape {
    area: Int,
}

fn area(s: Shape) -> Int {
    s.area
}

fn main() {
    let c = Circle { center: Point { x: 1, y: 2 }, radius: 3 }
    let a = area(c)
    print(a)
}
