fn is_prime(n: u32) -> bool {
    if n <= 1 {
        return false;
    }
    for i in 2..(n as f32).sqrt().ceil() as u32 {
        if n % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let mut count = 0;
    for i in 2..100 {
        if is_prime(i) {
            count += 1;
        }
    }
    println!("{}", count);
}
