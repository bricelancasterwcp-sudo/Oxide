fn main() {
  let rect = Rectangle { length: 3, width: 4 };
  let triangle = Triangle { a: 3, b: 4, c: 5 };
  let rect2 = Rectangle { length: 2, width: 5 };

  println!("{}", rect.perimeter());
  println!("{}", triangle.perimeter());
  println!("{}", rect2.perimeter());
}

struct Rectangle {
  length: u32,
  width: u32,
}

impl Rectangle {
  fn perimeter(&self) -> u32 {
    2 * (self.length + self.width)
  }
}

struct Triangle {
  a: u32,
  b: u32,
  c: u32,
}

impl Triangle {
  fn perimeter(&self) -> u32 {
    self.a + self.b + self.c
  }
}
