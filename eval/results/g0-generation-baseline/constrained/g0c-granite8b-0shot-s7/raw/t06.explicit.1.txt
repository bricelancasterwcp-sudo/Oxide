fn main() {
    for i in 2.to(1000) {
        let sum = sum_factors(i) - i
        if sum == i {
            println(i)
        }
    }
}

fn sum_factors(n: Int) -> Int {
    let mut total = 0
    let mut i = 1
    while i <= n {
        if n % i == 0 {
            total = total + i
        }
        i = i + 1
    }
    total
}
