fn main() {
  let pen = (3, 4);
  let pad = (9, 2);
  let ink = (5, 3);

  let combined_cost = pen.0 * pen.1 + pad.0 * pad.1 + ink.0 * ink.1;
  let (max_cost, name) = if pen.0 * pen.1 > pad.0 * pad.1 && pen.0 * pen.1 > ink.0 * ink.1 {
    (pen.0 * pen.1, "pen")
  } else if pad.0 * pad.1 > ink.0 * ink.1 {
    (pad.0 * pad.1, "pad")
  } else {
    (ink.0 * ink.1, "ink")
  };

  println!("Combined cost: {}", combined_cost);
  println!("Item with the highest cost: {}", name);
}
