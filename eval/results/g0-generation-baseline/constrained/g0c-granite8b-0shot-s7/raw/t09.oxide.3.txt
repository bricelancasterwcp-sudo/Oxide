    let items = push(push(push(push(vec(), Item { value: 5 }), Item { value: 4 }), Item { value: 3 }), Item { value: 2 })

fn pop(v: Vec<Item>) -> Item {
    let first = get(v, 0)?
    let rest = slice(v, 1, len(v))
    Item(value == first.value, rest == rest)
