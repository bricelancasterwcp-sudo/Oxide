fn palindrome(s: Str) -> Bool {
    let len = str_len(s)
    for i in range(0, len / 2) {
        if s.chars().get(i) != s.chars().get(len - i - 1) {
            return false
        }
    }
    true
}

fn main() {
    print(palindrome("level"))    // true
    print(palindrome("stereo"))   // false
    print(palindrome("rotor"))    // true
}
