fn count_positive(v: &Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn max_value(v: &Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        max = Some(match max {
                Some(y) => if x > y { x } else { y },
                None => x,
            })
    }
    max
}

fn main() {
    let nums = vec(3, 8, -2, 12, 7)
    let count = count_positive(&nums)
    print(count)
    let max = max_value(&nums)
    print(max?)
    drop nums
}
