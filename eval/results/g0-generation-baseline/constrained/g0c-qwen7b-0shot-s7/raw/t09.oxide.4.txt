struct VecInt { values: Vec<Int> }

fn reverse_print(v: VecInt) {
    for i in range(0, len(v.values)) {
        let x = get(v.values.clone(), len(v.values) - 1 - i)?
        print(x)
    }
}

fn main() {
    let v = VecInt { values: vec(push(push(push(push(vec(), 1), 2), 3), 4), 5) }
    reverse_print(v)
}
