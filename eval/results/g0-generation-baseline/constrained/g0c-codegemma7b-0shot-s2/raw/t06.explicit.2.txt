fn factors(x: Int) -> Vec<Int> {
    let mut result = vec()
    for i in 1.to(x) {
        if x % i == 0 {
            push(result, i)
        }
    }
    result
}

fn is_perfect(x: Int) -> Bool {
    if x == 0 {
        return false
    }
    let factors = factors(x)
    let sum = factors.iter().fold(0, add)
    return sum == x
}

fn main() {
    for i in 0.to(1000) {
        if is_perfect(i) {
            print(i)
        }
    }
}
