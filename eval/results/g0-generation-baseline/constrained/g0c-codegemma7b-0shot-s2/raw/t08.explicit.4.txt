fn count_pos(v: &Vec<Int>) -> Int {
    let mut count = 0
    for x in &v {
        if x > &mut && 0 {
            count = count + 1
        }
    }
    count
}

fn max(v: &Vec<Int>) -> Int {
    let mut max = &mut && 0
    for x in &v {
        if x > max {
            max = x
        }
    }
    max.clone()
}

fn main() {
    let nums = vec(3, 8, -2, 12, 7)
    print(count_pos(&nums))
    print(max(&nums))
    drop nums
}
