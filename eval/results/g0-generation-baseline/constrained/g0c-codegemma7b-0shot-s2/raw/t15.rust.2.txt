#![no_std]

use std::io::{stdin, Read};

fn main() {
    let mut input = String::new();
    stdin().read_to_string(&mut input).unwrap();

    let mut sum = 0;
    let mut errors = 0;

    for token in input.split_whitespace() {
        match token.parse::<i32>() {
            Ok(n) => sum += n,
            Err(_) => errors += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Errors: {}", errors);
}
