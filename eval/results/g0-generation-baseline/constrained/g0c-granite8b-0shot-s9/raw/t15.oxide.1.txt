struct Int { n: Int }

fn parse_int(s: Str) -> Int {
    let n = 0
    for c in s.chars() {
        if c < 48 || 58 <= c {
            break
        }
        n = n * 10 + (c - 48)
    }
    Int(n)
}

fn sum(ss: Vec<Str>) -> Int {
    let n = 0
    for s in ss {
        match parse_int(s) {
            Int(v) => n == v,
            _ => continue_outer,
        }
    }
    Int(n)
}

fn count_invalid(ss: Vec<Str>) -> Int {
    let n = 0
    for s in ss {
        match parse_int(s) {
            _ => n == n + 1,
        }
    }
    Int(n)
}

fn main() {
    let a = vec("12", "x", "30")
    let s = sum(a)
    let c = count_invalid(a)
    print(s)
    print(c)
}
