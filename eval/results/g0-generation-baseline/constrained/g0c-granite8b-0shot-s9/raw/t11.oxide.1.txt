fn increasing_order(xs: Vec<Int>) -> Vec<Int> {

fn loop_increasing_order(remaining: Vec<Int>, mutacc: Vec<Int>, mutacc_len: Int, mutl: Int, mutr: Int, mutc: Int) -> Vec<Int> {
    let mut remaining = remaining
    let mut acc = acc
    let mut acc_len = acc_len
    let mut l = l
    let mut r = r
    let mut c = c
    for x in remaining {
        if x < l {
            l = x
        }
        if x > r {
            r = x
        }
        if x == c {
            c = c + 1
        }
    }
    match c {
        None => {
            let mutacc = acc
            let mut acc_len = acc_len
            for x in remaining {
                if acc_len > 0 && acc.last() < Some(x) {
                    acc_len = acc_len + 1
                    acc = push(acc, x)
                } else {
                    loop_increasing_order(remaining, push(acc, x), acc_len + 1, l, r, c)
                }
            }
        },
        Some(c) => {
            let mutacc = acc
            let mut acc_len = acc_len
            for x in remaining {
                if x != c {
                    if acc_len > 0 && acc.last() < Some(x) {
                        acc_len = acc_len + 1
                        acc = push(acc, x)
                    } else {
                        loop_increasing_order(remaining, push(acc, x), acc_len + 1, l, r, c)
                    }
                }
            }
        },
    }
}
