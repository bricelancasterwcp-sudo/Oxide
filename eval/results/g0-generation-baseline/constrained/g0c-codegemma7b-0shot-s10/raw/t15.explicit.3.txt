fn main() {
    let mut total = 0
    let mut count = 0
    for arg in std_input() {
        if letOk(num) == parse_int(arg) {
            total = total + num
            count = count + 1
        }
    }
    print(total)
    print(count)
}

fn std_input() -> Vec<String> {
    let mut input = Vec.new()
    while letOk(line) == std_input_line() {
        input.push(line)
    }
    input
}

fn std_input_line() -> Option<String> {
    let mut line = String.new()
    std_input_char().map(move || line.push(c))
}

fn std_input_char() -> Option<Char> {
    std_input_byte().map(move || char(b))
}

fn std_input_byte() -> Option<Byte> {
    std_input_bytes().next()
}

fn std_input_bytes() -> Bytes {
    std_input_bytes_from(std_input_stream())
}

fn std_input_bytes_from(stream: InputStream) -> Bytes {
    let mut bytes = Bytes.new()
    for byte in stream {
        bytes.push(byte)
    }
    bytes
}

fn std_input_stream() -> InputStream {
    std_input_stream_from(std_input_device())
}

fn std_input_stream_from(device: Device) -> InputStream {
    device.input()
}

fn std_input_device() -> Device {
    std_input_device_from(std_input_context())
}

fn std_input_device_from(context: Context) -> Device {
    context.device(0)
}

    std_input_context_from(std_input_handle())

fn std_input_socket_from(address: Address) -> Socket {

fn std_input_address() -> Address {

}

}

fn std_input_url() -> Url {
    std_input_url_from(std_input_string())
}

fn std_input_url_from(string: String) -> Url {
    Url.parse(string)
}

fn std_input_string() -> String {
    std_input_line().unwrap()
}

fn parse_int(arg: String) -> Option<Int> {
    match parse_int_impl(arg) {
        Ok(num) => Some(num),
        Err(_) => None,
    }
}

fn parse_int_impl(arg: String) -> Result<Int, Error> {
    let mut num = 0
    for c in arg.chars() {
        num = num * 10 + c - b
    }
    Ok(num)
}
