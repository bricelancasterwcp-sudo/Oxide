fn is_palindrome(text: Str) -> Bool {
    let reversed = reverse(text)
    text == reversed
}

fn reverse(text: Str) -> Str {
    let mut chars = chars(text)
    let mut rev = vec()
    while len(chars) > 0 {
        rev = push(rev, get(chars, 0)?)
        chars = slice(chars, 1, -1)
    }
    rev
}

fn main() {
    let level = "level"
    let stereo = "stereo"
    let rotor = "rotor"
    print(is_palindrome(level))
    print(is_palindrome(stereo))
    print(is_palindrome(rotor))
}
