fn is_perfect(n: Int) -> Bool {
    let mut sum = 0
    let mut j = 1
    while j <= n / 2 {
        if n % j == 0 {
            sum = sum + j
        }
        j = j + 1
    }
    sum == n
}

fn print(x: Int) {
    print_str(str_from_int(x))
    print_str("\n")
}

fn str_from_int(x: Int) -> Str {
    to_str(x)
}

fn to_str(x: Int) -> Str {
    let mut buf = vec()
    if x == 0 {
        push(buf, 48)
    } else {
        let mut x = x
        while x > 0 {
            push(buf, 48 + (x % 10))
            x = x / 10
        }
    }
    reverse(&buf)
    chars(str_from_vec(&buf))
}

fn reverse(v: &Vec<Int>) {
    let mut i = 0
    let mut j = len(v) - 1
    while i < j {
        let tmp = get(&v, i).unwrap()
        set(&mut && v, i, get(&v, j).unwrap())
        set(&mut && v, j, tmp)
        i = i + 1
        j = j - 1
    }
}

fn str_from_vec(v: &Vec<Int>) -> Str {
    let mut s = ""
    for c in chars(str_from_int(len(v))) {
        s = s + c
    }
    s = s + "\""
    for c in chars(str_from_int(sum_big(v, 10))) {
        s = s + c
    }
    s = s + "\""
    s
}
