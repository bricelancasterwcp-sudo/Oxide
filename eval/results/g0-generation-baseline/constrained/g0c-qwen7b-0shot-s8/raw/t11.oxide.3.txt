struct List { elements: Vec<Int> }

fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut result = v
    for i in range(0, len(result)) {
        for j in range(0, len(result) - i - 1) {
            if get(result, j).unwrap() > get(result, j + 1).unwrap() {
                let temp = get(result, j).unwrap()
                set(result, j, get(result, j + 1).unwrap())
                set(result, j + 1, temp)
            }
        }
    }
    result
}

fn main() {
    let list = List { elements: vec(5, 3, 8, 1, 9, 2) }
    let sorted = sort(list.elements.clone())
    for x in sorted {
        print(x)
    }
}
