fn main() {
    let s = "mississippi"
    let mut count_s = 0
    let mut seen = Vec()
    for c in chars(s) {
        if c == "s" {
            count_s = count_s + 1
        }
        if !contains(seen, c) {
            push(seen, c)
        }
    }
    print(count_s)
    print_str(" ")
    print(len(seen))
}
