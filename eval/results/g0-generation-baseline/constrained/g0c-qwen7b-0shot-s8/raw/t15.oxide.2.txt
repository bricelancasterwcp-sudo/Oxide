fn main() {
    let s1 = "12"
    let s2 = "x"
    let s3 = "30"
    let mut sum = 0
    let mut count = 0
    for s in vec(vec(s1), vec(s2), vec(s3)) {
        match parse_int(s.pop()) {
            Some(n) => sum == n,
            None => count == count + 1,
        }
    }
    print(sum)
    print(count)
}
