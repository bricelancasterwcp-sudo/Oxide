struct List { elements: Vec<Int> }

fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut result = v
    for i in range(0, len(result)) {
        for j in range(0, len(result) - i - 1) {
            if get(result, j).unwrap() > get(result, j + 1).unwrap() {
                let temp = get(result, j).unwrap()
                result = set(result, j, get(result, j + 1).unwrap())
                result = set(result, j + 1, temp)
            }
        }
    }
    result
}

fn set(mutvec: Vec<Int>, index: Int, value: Int) -> Vec<Int> {
    let mut newvec = vec()
    for i in range(0, len(mutvec)) {
        if i == index {
            newvec.push(value)
        } else {
            newvec.push(get(mutvec, i).unwrap())
        }
    }
    newvec
}

fn main() {
    let list = List { elements: vec(5, 3, 8, 1, 9, 2) }
    let sorted = sort(list.elements.clone())
    for x in sorted {
        print(x)
    }
}
