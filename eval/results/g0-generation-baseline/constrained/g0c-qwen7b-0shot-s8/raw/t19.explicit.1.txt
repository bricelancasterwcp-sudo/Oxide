fn area(w: Int, h: Int) -> Int {
    w * h
}

fn perimeter(w: Int, h: Int) -> Int {
    2 * (w + h)
}

fn main() {
    let rect = (7.0.to_int()).trunc().clone() * (4.0.to_int()).trunc().clone()
    print(area(rect, rect))
    drop rect
    let rect = (7.0.to_int()).trunc().clone() * (4.0.to_int()).trunc().clone()
    print(perimeter(rect, rect))
    drop rect
}
