fn is_palindrome(s: &Str) -> Bool {
    let len = str_len(s)
    let mut i = 0
    let mut j = len - 1
    while i < j {
        if s.get(i) != s.get(j) {
            return false
        }
        i = i + 1
        j = j - 1
    }
    true
}

fn main() {
    print(is_palindrome(&str_len))
    print(is_palindrome(&str_len))
    print(is_palindrome(&str_len))
}
