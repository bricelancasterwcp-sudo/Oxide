struct Item { name: Str, unit_price: Int, quantity: Int }

fn cost(item: Item) -> Int {
    item.unit_price * item.quantity
}

fn highest_cost_item(items: Vec<Item>) -> Item {
    let mut max_cost = -1
    let mut max_item = Item { name: "".clone(), unit_price: 0, quantity: 0 }
    for item in items {
        let item_cost = cost(item)
        if item_cost > max_cost {
            max_cost = item_cost
            max_item = item
        }
    }
    max_item
}

fn main() {
    let items = vec(Item { name: "pen".clone(), unit_price: 3, quantity: 4 }, Item { name: "pad".clone(), unit_price: 9, quantity: 2 }, Item { name: "ink".clone(), unit_price: 5, quantity: 3 })
    let total_cost = items.iter().map(cost).sum()
    print(total_cost)
    print_str("\n")
    let highest_cost_item = highest_cost_item(items)
    print(highest_cost_item.name)
}
