fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut v = v
    for i in 0.veclen() {
        let mut min_index = i
        for j in i + 1.veclen() {
            if v.get(j).unwrap() < v.get(min_index).unwrap() {
                min_index = j
            }
        }
        let temp = v.get(min_index).unwrap()
        v.set(min_index, v.get(i).unwrap())
        v.set(i, temp)
    }
    v
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(clone(&nums))
    for x in sorted {
        print_str(x.to_str())
    }
    drop sorted
}
