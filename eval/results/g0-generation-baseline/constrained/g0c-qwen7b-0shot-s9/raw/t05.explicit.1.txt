fn fib(n: Int) -> Int {
    if n == 0 {
        1
    } else if n == 1 {
        1
    } else {
        let a = 1
        let b = 1
        let mut i = 2
        while i < n {
            let temp = a + b
            drop a
            a = b
            b = temp
            drop temp
            i = i + 1
        }
        b
    }
}

fn main() {
    let result = fib(20)
    print(result)
    drop result
}
