use std::str::FromStr;

fn main() {
    let mut success = 0;
    let mut failure = 0;

    for line in ["12", "x", "30"] {
        if let Ok(n) = line.parse::<i32>() {
            success += n;
        } else {
            failure += 1;
        }
    }

    println!("success: {}", success);
    println!("failure: {}", failure);
}
