fn is_perfect(n: Int) -> Bool {
    let mut sum = 1
    let limit = n / 2
    for i in range(2, limit + 1) {
        if n % i == 0 {
            sum = sum + i
        }
    }
    sum == n
}

fn main() {
    for num in range(1, 1000) {
        if is_perfect(num) {
            print(num)
        }
    }
}
