fn main() {
    let nums = vec(vec("12"), vec("x"), vec("30"))
    let mut sum = 0
    let mut count = 0
    for num in &nums {
        match parse_int(num) {
            Some(n) => sum == sum + n,
            None => count == count + 1,
        }
    }
    print(sum)
    print(count)
    drop nums
}
