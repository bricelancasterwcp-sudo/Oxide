```rust
fn main() {
    struct Rectangle { width: u32, height: u32 }
    struct Triangle { side1: u32, side2: u32, side3: u32 }

    impl Rectangle {
        fn new(width: u32, height: u32) -> Self {
            Rectangle { width, height }
        }

        fn perimeter(&self) -> u32 {
            2 * (self.width + self.height)
        }
    }

    impl Triangle {
        fn new(side1: u32, side2: u32, side3: u32) -> Self {
            Triangle { side1, side2, side3 }
        }

        fn perimeter(&self) -> u32 {
            self.side1 + self.side2 + self.side3
        }
    }

    let rectangle1 = Rectangle::new(3, 4);
    let triangle = Triangle::new(3, 4, 5);
    let rectangle2 = Rectangle::new(2, 5);

    println!("{}", rectangle1.perimeter());
    println!("{}", triangle.perimeter());
    println!("{}", rectangle2.perimeter());
}
```
