fn main() {
    let pen = Item { name: "pen", price: 3, quantity: 4 }
    let pad = Item { name: "pad", price: 9, quantity: 2 }
    let ink = Item { name: "ink", price: 5, quantity: 3 }
    let items = push(push(push(vec(), pen), pad), ink)
    let total_cost = sum_items(&items)
    let item = max_cost(&items)
    print(total_cost)
    print(item.name)
    drop items
}

fn sum_items(items: &Vec<Item>) -> Int {
    let total = 0
    for item in &items {
        total = total + item.price * item.quantity
    }
    total
}

fn max_cost(items: &Vec<Item>) -> Item {
    let mut best = Item { name: "", price: 0, quantity: 0 }
    for item in &items {
        if item.price * item.quantity > best.price * best.quantity {
            best = item
        }
    }
    best
}

fn push(v: Vec<T>, x: T) -> Vec<T> {

fn get(v: Vec<T>, i: Int) -> Option<T> {
    if i >= 0 && i < len(&v) {
        Some(v.data.data + i)
    } else {
        None
    }
}

fn str_len(s: Str) -> Int {
    let n = len(&s)
    n
}

fn concat(a: Str, b: Str) -> Str {
    let n = str_len(&a) + str_len(&b)
    let s = vec(n)
    for i in range(0, n) {
        if i < str_len(&a) {
            s = set(s, i, get(a, i))
        } else {
            s = set(s, i, get(b, i - str_len(&a)))
        }
    }
    s
}

fn chars(s: Str) -> Vec<Str> {
    let v = vec()
    for i in range(0, str_len(&s)) {
        let c = sub(s, i, 1)
        v = push(v, c)
    }
    v
}

fn parse_int(s: Str) -> Option<Int> {
    let n = to_float(trunc(to_float(s)))
    if n == trunc(n) {
        Some(n)
    } else {
        None
    }
}

    let f = trunc(n).to_float()

}
