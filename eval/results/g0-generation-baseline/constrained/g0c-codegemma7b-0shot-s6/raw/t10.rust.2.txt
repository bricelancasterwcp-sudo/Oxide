use std::io;

fn main() {
    let mut list = vec![4, 1, 7, 3, 9, 2];
    let mut counter = 0;

    for num in list.iter() {
        if *num > 3 {
            println!("{}", num);
            counter += 1;
        } else {
            list.retain(|&x| x != *num);
        }
    }

    println!("{} values were kept", counter);
}
