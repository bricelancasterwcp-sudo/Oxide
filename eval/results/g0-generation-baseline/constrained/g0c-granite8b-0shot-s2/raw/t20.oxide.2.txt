fn pen() -> Int {
    3
}

fn pad() -> Int {
    9
}

fn ink() -> Int {
    5
}

fn max(a: Int, b: Int, c: Int) -> Int {
    if a > b && a > c {
        a
    } else if b > a && b > c {
        b
    } else {
        c
    }
}

fn main() {
    let pens = pen() * 4
    let pads = pad() * 2
    let ink = ink() * 3
    let total_cost = pens + pads + ink
    print(total_cost)
    let max_cost = max(pens, pads, ink)
    match max_cost {
        _ => print("ink"),
        _ => print("pen"),
        _ => print("pad"),
    }
}
