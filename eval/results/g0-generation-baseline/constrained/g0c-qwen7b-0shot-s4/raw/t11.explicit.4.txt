fn sort(vec: Vec<Int>) -> Vec<Int> {
    let mut sorted = vec
    let n = len(&sorted)
    for i in range(0, n) {
        for j in range(i + 1, n) {
            if get(sorted, i) > get(sorted, j) {
                let temp = get(sorted, i)
                let mut mutsorted = sorted
                mutsorted = set(&mutsorted, i, get(sorted, j))
                mutsorted = set(&mutsorted, j, temp)
                sorted = mutsorted
            }
        }
    }
    sorted
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(clone(&nums))
    for num in &sorted {
        print(num)
    }
    drop nums
    drop sorted
}
