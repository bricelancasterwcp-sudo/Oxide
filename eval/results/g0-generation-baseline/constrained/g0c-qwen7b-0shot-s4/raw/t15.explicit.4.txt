fn parse_and_sum(nums: Vec<Str>) -> Int {
    let mut sum = 0
    let mut count = 0
    for num in &nums {
        match parse_int(num) {
            Some(n) => {
                sum = sum + n
            },
            None => {
                count = count + 1
            },
        }
    }
    sum
}

fn main() {
    let nums = vec(vec("12"), vec("x"), vec("30"))
    let sum = parse_and_sum(nums.clone())
    let count = 3 - sum / 12 - count / 30
    print(sum)
    print(count)
    drop sum
    drop count
}
