fn count_s(text: Str) -> Int {
    let mut count = 0
    for c in chars(text) {
        if c == "s" {
            count = count + 1
        }
    }
    count
}

fn distinct_chars(text: Str) -> Int {
    let mut set = Vec()
    for c in chars(text) {
        if !contains(set, c) {
            set = push(set, c)
        }
    }
    len(set)
}

fn main() {
    let text = "mississippi"
    print(count_s(text))
    print_str(" ")
    print(distinct_chars(text))
}
