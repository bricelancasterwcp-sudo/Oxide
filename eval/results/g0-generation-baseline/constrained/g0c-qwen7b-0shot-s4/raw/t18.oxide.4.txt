struct Point { x: Float, y: Float }

fn perimeter(p1: Point, p2: Point, p3: Point) -> Float {
    let d1 = ((p2.x - p1.x).powf(2.0) + (p2.y - p1.y).powf(2.0)).sqrt()
    let d2 = ((p3.x - p2.x).powf(2.0) + (p3.y - p2.y).powf(2.0)).sqrt()
    let d3 = ((p1.x - p3.x).powf(2.0) + (p1.y - p3.y).powf(2.0)).sqrt()
    (d1 + d2 + d3).to_float()
}

fn main() {
    let p1 = Point { x: 0.0, y: 0.0 }
    let p2 = Point { x: 3.0, y: 4.0 }
    let p3 = Point { x: 8.0, y: 0.0 }
    print(perimeter(p1, p2, p3))
}
