fn main() {
    let text = "mississippi"
    let count_s = text.chars().filter(char_eq_s).count()
    let distinct_chars = text.chars().collect_vec().into_iter().distinct().count()
    print(count_s)
    drop count_s
    print(distinct_chars)
    drop distinct_chars
}

fn char_eq_s(c: Char) -> Bool {
    c == "s"
}
