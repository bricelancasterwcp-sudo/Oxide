fn main() {
    let pen_cost = (3.0 * 4.0).to_int()
    let pad_cost = (9.0 * 2.0).to_int()
    let ink_cost = (5.0 * 3.0).to_int()
    let total_cost = pen_cost + pad_cost + ink_cost
    print_str("Combined cost of all items: ")
    print(total_cost)
    print_str("\n")
    let items = vec(("pen".to_str()), ("pad".to_str()), ("ink".to_str()))
    let mut max_cost = 0
    let mut max_item = ""
    for item in items {
        let item_cost = match item {
            Some(name) => match name {
                Some(pen) => pen,
                Some(pad) => pad,
                Some(ink) => ink,
                _ => 0,
            },
            _ => 0,
        }
        if item_cost > max_cost {
            max_cost = item_cost
            max_item = item
        }
    }
    print_str("Item with the highest cost: ")
    print_str(max_item)
    print_str("\n")
    drop items
}
