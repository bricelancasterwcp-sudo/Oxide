fn area(width: Int, height: Int) -> Int {
    width * height
}

fn perimeter(width: Int, height: Int) -> Int {
    2 * (width + height)
}

fn main() {
    let rect = Rectangle(7, 4)
    print(area(&rect, rect.width()))
    print(perimeter(&rect, rect.height()))
    drop rect
}
