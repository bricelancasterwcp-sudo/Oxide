fn get_cost(unit_price: Int, quantity: Int) -> Int {
    unit_price * quantity
}

fn main() {
    let pen_cost = get_cost(3, 4)
    let pad_cost = get_cost(9, 2)
    let ink_cost = get_cost(5, 3)
    let total_cost = pen_cost + pad_cost + ink_cost
    print(total_cost)
    if pen_cost > pad_cost {
        if pen_cost > ink_cost {
            print("pen")
        } else {
            print("ink")
        }
    } else if pad_cost > ink_cost {
        print("pad")
    } else {
        print("ink")
    }
    drop pen_cost
    drop pad_cost
    drop ink_cost
    drop total_cost
}
