use std::io;

fn main() {
    let mut total_cost = 0;
    let mut most_expensive_item = "";
    let mut most_expensive_cost = 0;

    // Pen
    let pen_cost = 3 * 4;
    total_cost += pen_cost;
    if pen_cost > most_expensive_cost {
        most_expensive_item = "Pen";
        most_expensive_cost = pen_cost;
    }

    // Pad
    let pad_cost = 9 * 2;
    total_cost += pad_cost;
    if pad_cost > most_expensive_cost {
        most_expensive_item = "Pad";
        most_expensive_cost = pad_cost;
    }

    // Ink
    let ink_cost = 5 * 3;
    total_cost += ink_cost;
    if ink_cost > most_expensive_cost {
        most_expensive_item = "Ink";
        most_expensive_cost = ink_cost;
    }

    println!("Total cost: {}", total_cost);
    println!("Most expensive item: {}", most_expensive_item);
}
