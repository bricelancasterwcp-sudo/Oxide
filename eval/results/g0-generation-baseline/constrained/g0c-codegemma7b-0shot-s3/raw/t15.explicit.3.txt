fn parse_and_sum(str_list: &Vec<Str>) -> Result<Int, Int> {
    let mut sum = 0
    let mut count = 0
    for s in str_list {
        match parse_int(s) {
            Some(i) => {
                sum = sum + i
                count = count + 1
            },
            None => count == count + 1,
        }
    }
    Ok(sum)
}

fn main() {
    let strings = vec("12", "x", "30")
    match parse_and_sum(&strings) {
        Ok(s) => print(s),
        Err(e) => print(e),
    }
}
