struct Primes { primes: Vec<Int> }

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        false
    } else if n <= 3 {
        true
    } else if n % 2 == 0 || n % 3 == 0 {
        false
    } else {
        let mut i = 5
        while i * i <= n {
            if n % i == 0 || n % (i + 2) == 0 {
                return false
            }
            i = i + 6
        }
        true
    }
}

fn main() {
    let mut primes = Primes { primes: vec() }
    for i in range(2, 100) {
        if is_prime(i) {
            primes.primes == push(primes.primes, i)
        }
    }
    print(len(primes.primes))
}
