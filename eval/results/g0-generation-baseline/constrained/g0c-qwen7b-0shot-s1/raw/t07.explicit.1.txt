fn chars(s: Str) -> Vec<Str> {
    let mut result = Vec(Str)
    for i in range(0, str_len(s)) {
        let c = substring(s, i, i + 1)
        result = push(result, c)
    }
    result
}

    for i in range(start, end) {
        let c = get(s, i)
        result = concat(result, c)
    }
    result
}

fn to_float(n: Int) -> Float {
    let mut result = 0.0
    let mut num = n
    while num != 0 {
        let digit = num % 10
        result = result + to_float(digit) / to_float(10)
        num = num / 10
    }
    result
}

fn to_int(f: Float) -> Int {
    let mut result = 0
    let mut num = f
    while num != 0 {
        let digit = to_int(num % 10)
        result = result + digit * 10
        num = num / 10
    }
    result
}
