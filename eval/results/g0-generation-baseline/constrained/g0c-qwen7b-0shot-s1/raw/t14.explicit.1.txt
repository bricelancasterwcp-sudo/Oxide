fn main() {
    let text = "mississippi"
    let s_count = text.chars().filter(char_eq(&s)).count()
    let distinct_chars = text.chars().collect_set().len()
    print(s_count)
    print_str(" ")
    print(distinct_chars)
    drop text
}

fn char_eq(c: &Char) -> FnMut_Char {
    move || char_eq_helper(c)
}

fn char_eq_helper(c: &Char) -> Bool {
    c == &s
}

fn collect_set(v: Vec<Str>) -> HashSet<Str> {
    let mut set = HashSet.new()
    for x in &v {
        set.insert(x.clone())
    }
    set
}
