fn is_prime(x: Int) -> Bool {
    let mut i = 2
    while i * i <= x {
        if x % i == 0 {
            return false
        }
        i = i + 1
    }
    true
}

fn if_prime(x: Int) -> Int {
    if is_prime(x) {
        x
    } else {
        0
    }
}

fn filter_map(v: Vec<Int>, f: Fun<Int, Int>) -> Vec<Int> {
    let acc = vec()
    for x in v {
        let y = f(x)
        if y != 0 {
            acc = push(acc, y)
        }
    }
    acc
}

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}
