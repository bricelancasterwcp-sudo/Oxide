fn reverse<T>(mut v: Vec<T>) -> Vec<T> {
    let mut result = vec();
    while let Some(x) = pop(&mut v) {
        push(result, x);
    }
    result
}

fn pop<T>(v: &mut Vec<T>) -> Option<T> {
    v.pop()
}
