fn parse_int(s: Str) -> Option<Int> {
    parse_int(s)
}

fn sum_and_count(nums: Vec<Str>) -> (Int, Int) {
    let mut total = 0;
    let mut count = 0;
    for s in nums {
        match parse_int(s) {
            Some(n) => {
                total += n;
            },
            None => {
                count += 1;
            }
        }
    }
    (total, count)
}

fn main() {
    let nums = vec("12", "x", "30");
    let (sum, count) = sum_and_count(nums);
    print(sum);
    print(count);
}
