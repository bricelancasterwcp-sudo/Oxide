fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v;
    for i in 0..sorted.len() {
        for j in i + 1..sorted.len() {
            if sorted[i] > sorted[j] {
                let temp = sorted[i];
                sorted[i] = sorted[j];
                sorted[j] = temp;
            }
        }
    }
    sorted
}

fn main() {
    let v = push(push(push(push(push(vec(), 5), 3), 8), 1), 9), 2);
    let sorted = sort(v);
    for x in sorted {
        print(x);
        print_str("\n");
    }
}
