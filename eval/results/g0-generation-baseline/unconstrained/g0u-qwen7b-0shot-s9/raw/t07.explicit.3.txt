fn sum_multiples_of_three() -> Int {
    let mut sum = 0;
    let mut count = 0;
    let mut current = 3;
    while count < 8 {
        sum = sum + current;
        current = current + 3;
        count = count + 1;
    }
    sum
}

fn main() {
    let total = sum_multiples_of_three();
    print(total);
}
