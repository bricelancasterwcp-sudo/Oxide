```rust
fn main() {
    let rectangle1 = Rectangle { width: 3, height: 4 };
    let triangle = Triangle { sides: [3, 4, 5] };
    let rectangle2 = Rectangle { width: 2, height: 5 };

    println!("{}", rectangle1.perimeter());
    println!("{}", triangle.perimeter());
    println!("{}", rectangle2.perimeter());
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn perimeter(&self) -> u32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    sides: [u32; 3],
}

impl Triangle {
    fn perimeter(&self) -> u32 {
        self.sides.iter().sum()
    }
}
```
