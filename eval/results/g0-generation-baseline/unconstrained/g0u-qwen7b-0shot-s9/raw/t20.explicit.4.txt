```rust
struct Item {
    name: Str,
    unit_price: Int,
    quantity: Int,
}

fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 };
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 };
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 };

    let total_cost = pen.unit_price * pen.quantity + pad.unit_price * pad.quantity + ink.unit_price * ink.quantity;
    print_str(&format!("Combined cost: {}\n", total_cost));

    let mut items = vec![pen, pad, ink];
    items.sort_by_key(|item| -item.unit_price * item.quantity);
    let most_expensive = &items[0];
    print_str(&format!("Item with highest cost: {}\n", most_expensive.name));

    drop items;
}
```
