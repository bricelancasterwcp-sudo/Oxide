fn is_prime(x: Int) -> Bool {
    if x <= 1 {
        return false
    }
    for i in range(2, x) {
        if x % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let mut count = 0
    for x in range(1, 100) {
        if is_prime(x) {
            count = count + 1
        }
    }
    print(count)
}
