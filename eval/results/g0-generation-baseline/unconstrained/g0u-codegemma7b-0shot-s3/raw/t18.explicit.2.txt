fn rect_perimeter(a: Int, b: Int) -> Int {
    2 * (a + b)
}

fn tri_perimeter(a: Int, b: Int, c: Int) -> Int {
    a + b + c
}

fn main() {
    let r1 = rect_perimeter(3, 4);
    print(r1);
    println();
    let t1 = tri_perimeter(3, 4, 5);
    print(t1);
    println();
    let r2 = rect_perimeter(2, 5);
    print(r2);
    println();
}
