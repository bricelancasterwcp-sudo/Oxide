fn main() {
    let pen = Item { unit_price: 3, quantity: 4 };
    let pad = Item { unit_price: 9, quantity: 2 };
    let ink = Item { unit_price: 5, quantity: 3 };

    let total_cost = pen.cost() + pad.cost() + ink.cost();
    print(total_cost);

    let mut max_cost_item = pen;
    if pad.cost() > max_cost_item.cost() {
        max_cost_item = pad;
    }
    if ink.cost() > max_cost_item.cost() {
        max_cost_item = ink;
    }

    print_str(max_cost_item.name());
}

struct Item {
    unit_price: Int,
    quantity: Int,
}

impl Item {
    fn cost(&self) -> Int {
        self.unit_price * self.quantity
    }

    fn name(&self) -> Str {
        match self {
            Item { unit_price: 3, .. } => "pen",
            Item { unit_price: 9, .. } => "pad",
            Item { unit_price: 5, .. } => "ink",
        }
    }
}
