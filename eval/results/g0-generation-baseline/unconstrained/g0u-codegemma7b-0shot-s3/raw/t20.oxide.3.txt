struct Item {
    name: Str,
    unit_price: Float,
    quantity: Int,
}

fn main() {
    let pen = Item { name: "pen", unit_price: 3.0, quantity: 4 };
    let pad = Item { name: "pad", unit_price: 9.0, quantity: 2 };
    let ink = Item { name: "ink", unit_price: 5.0, quantity: 3 };

    let total_cost = pen.cost() + pad.cost() + ink.cost();
    let most_expensive = max_by(vec![pen, pad, ink], |item| item.cost());

    println!("Total cost: {}", total_cost);
    println!("Most expensive item: {}", most_expensive.name);
}

impl Item {
    fn cost(&self) -> Float {
        self.unit_price * self.quantity as Float
    }
}

fn max_by<T, F>(items: Vec<T>, mut f: F) -> T
where
    F: FnMut(&T) -> Float,
{
    let mut max_item = None;
    let mut max_value = -Float::INFINITY;

    for item in items {
        let value = f(&item);
        if value > max_value {
            max_value = value;
            max_item = Some(item);
        }
    }

    max_item.unwrap()
}
