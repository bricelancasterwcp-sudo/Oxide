struct Stack {
    data: Vec<Int>,
}

fn main() {
    let mut s = Stack { data: vec() }
    for c in "stack".chars() {
        push(s.data, c)
    }
    let mut result = vec()
    while len(s.data) > 0 {
        result = push(result, get(s.data, len(s.data) - 1)?)
        s.data = range(0, len(s.data) - 1)
    }
    print_str(str_cat(result))
}
