struct Pen { price: Int, quantity: Int }
struct Pad { price: Int, quantity: Int }
struct Ink { price: Int, quantity: Int }

fn cost(item: &Item) -> Int {
    item.price * item.quantity
}

fn highest_cost(items: Vec<Item>) -> (Int, Str) {
    let mut highest = (0, "")
    for item in items {
        let c = cost(&item)
        if c > highest.0 {
            highest = (c, item.name)
        }
    }
    highest
}
