fn filter(v: Vec<Int>, f: fn(Int) -> Bool) -> Vec<Int> {
    let mut out = vec()
    for x in v {
        if f(x) {
            out = push(out, x)
        }
    }
    out
}

fn foldl(v: Vec<Int>, f: fn(Int, Int) -> Int) -> Int {
    let mut acc = 0
    for x in v {
        acc = f(acc, x)
    }
    acc
}
