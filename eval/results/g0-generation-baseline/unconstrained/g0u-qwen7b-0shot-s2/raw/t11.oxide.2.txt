struct List { head: Option<Int>, tail: Option<Box<List>> }

fn empty_list() -> List {
    List { head: None, tail: None }
}

fn push_front(l: List, x: Int) -> List {
    List { head: Some(x), tail: Some(Box::new(l)) }
}

fn print_list(l: List) {
    match l.head {
        Some(x) => {
            print(x);
            print_str("\n");
            print_list(*l.tail.unwrap());
        },
        None => ()
    }
}

fn main() {
    let mut l = empty_list();
    l = push_front(l, 2);
    l = push_front(l, 9);
    l = push_front(l, 8);
    l = push_front(l, 3);
    l = push_front(l, 5);
    print_list(l);
}
