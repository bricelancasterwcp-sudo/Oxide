fn transform_steps(n: Int) -> Int {
    let mut steps = 0;
    let mut current = n;
    while current != 1 {
        if current % 2 == 0 {
            current = current / 2;
        } else {
            current = current * 3 + 1;
        }
        steps = steps + 1;
    }
    steps
}

fn main() {
    let steps = transform_steps(27);
    print(steps);
}
