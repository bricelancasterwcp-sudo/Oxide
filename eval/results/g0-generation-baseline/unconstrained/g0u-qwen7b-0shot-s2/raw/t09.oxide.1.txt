struct List { head: Option<Int>, tail: Option<Box<List>> }

fn reverse_print(lst: List) {
    match lst {
        List { head: Some(x), tail: Some(t) } => {
            reverse_print(*t);
            print(x);
        },
        List { head: Some(x), tail: None } => print(x),
        _ => ()
    }
}

fn main() {
    let lst = List { head: Some(1), tail: Some(Box::new(List { head: Some(2), tail: Some(Box::new(List { head: Some(3), tail: Some(Box::new(List { head: Some(4), tail: Some(Box::new(List { head: Some(5), tail: None })) })) })) })) })) };
    reverse_print(lst);
}
