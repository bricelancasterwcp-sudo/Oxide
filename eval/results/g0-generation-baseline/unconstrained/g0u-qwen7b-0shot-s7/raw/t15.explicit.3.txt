```rust
fn parse_int(s: &Str) -> Option<Int> {
    let parsed: Result<Int, _> = s.parse();
    match parsed {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn main() {
    let nums = ["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;
    for num in nums {
        match parse_int(num) {
            Some(n) => {
                sum = sum + n;
            },
            None => {
                count = count + 1;
            }
        }
    }
    print(sum);
    print(count);
}
```
