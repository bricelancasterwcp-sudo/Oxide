fn main() {
    let strings = vec!["12", "x", "30"];
    let mut sum = 0;
    let mut fail_count = 0;

    for s in strings {
        match s.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => fail_count += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Fail count: {}", fail_count);
}
