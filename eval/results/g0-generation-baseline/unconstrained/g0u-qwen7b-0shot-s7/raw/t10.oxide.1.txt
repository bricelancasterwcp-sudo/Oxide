```rust
struct List { values: Vec<Int> }

fn filter_greater_than_three(v: Vec<Int>) -> Vec<Int> {
    let mut result = vec();
    for x in v {
        if x > 3 {
            push(result, x);
        }
    }
    result
}

fn main() {
    let l = List { values: vec(4, 1, 7, 3, 9, 2) };
    let filtered = filter_greater_than_three(l.values);
    for x in filtered {
        print(x);
    }
    print(len(filtered));
}
```
