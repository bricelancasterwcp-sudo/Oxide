```rust
struct Multiples { list: Vec<Int> }

fn sum_multiples() -> Int {
    let mut acc = Multiples { list: vec() };
    for i in range(1, 9) {
        let multiple = 3 * i;
        acc.list = push(acc.list, multiple);
    }
    let sum = acc.list.iter().sum();
    sum
}

fn main() {
    let result = sum_multiples();
    print(result);
}
```
