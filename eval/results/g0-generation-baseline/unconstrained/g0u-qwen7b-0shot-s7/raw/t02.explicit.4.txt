fn transform(num: Int) -> Int {
    let mut steps = 0;
    let mut current = num;
    while current != 1 {
        if current % 2 == 0 {
            current = current / 2;
        } else {
            current = current * 3 + 1;
        }
        steps = steps + 1;
    }
    steps
}

fn main() {
    let result = transform(27);
    print_str(result.to_string());
}
