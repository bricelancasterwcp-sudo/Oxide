fn sum_of_multiples(n: Int) -> Int {
    let mut sum = 0;
    let mut count = 0;
    let mut i = 1;
    while count < n {
        if i % 3 == 0 {
            sum = sum + i;
            count = count + 1;
        }
        i = i + 1;
    }
    sum
}

fn main() {
    let result = sum_of_multiples(8);
    print(result);
    drop result;
}
