```rust
struct List { values: Vec<Int> }

fn count_positive(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn max_value(v: Vec<Int>) -> Int {
    let mut max = -1000
    for x in v {
        if x > max {
            max = x
        }
    }
    max
}

fn main() {
    let l = List { values: push(push(push(push(vec(), 3), 8), -2), 12) }
    print(count_positive(l.values))  // prints 3
    print(max_value(l.values))       // prints 12
}
```
