```rust
struct Item {
    name: &'static str,
    unit_price: Int,
    quantity: Int,
}

fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 };
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 };
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 };

    let total_cost = pen.unit_price * pen.quantity + pad.unit_price * pad.quantity + ink.unit_price * ink.quantity;
    print(total_cost);

    let items = vec![pen, pad, ink];
    let mut max_item = &items[0];
    for item in &items {
        if item.unit_price * item.quantity > max_item.unit_price * max_item.quantity {
            max_item = item;
        }
    }
    print(max_item.name);
    drop items;
}
```
