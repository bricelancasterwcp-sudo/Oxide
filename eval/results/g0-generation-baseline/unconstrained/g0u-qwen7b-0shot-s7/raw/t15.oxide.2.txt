```rust
fn parse_int(s: Str) -> Option<Int> {
    match parse_int(s) {
        Some(n) => Some(n),
        None => None,
    }
}

fn main() {
    let nums = vec("12", "x", "30");
    let mut sum = 0;
    let mut count = 0;
    for num in nums {
        match parse_int(num) {
            Some(n) => {
                sum += n;
            },
            None => {
                count += 1;
            }
        }
    }
    print(sum);
    print(count);
}
```
