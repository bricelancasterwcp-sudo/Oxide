fn main() {
    let mut total = 0
    let mut count = 0
    for arg in std::env::args().skip(1) {
        match parse_int(&arg) {
            Some(n) => {
                total = total + n
                count = count + 1
            }
            None => {}
        }
    }
    print("{}\n{}\n", total, count)
}
