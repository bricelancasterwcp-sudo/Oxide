fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false;
    }
    for i in 2..trunc(sqrt(n)) + 1 {
        if n % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let mut count = 0;
    for i in 0..100 {
        if is_prime(i) {
            count = count + 1;
        }
    }
    print(count)
}
