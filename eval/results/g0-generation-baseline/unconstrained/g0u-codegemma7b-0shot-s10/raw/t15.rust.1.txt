```rust
fn main() {
    let mut sum = 0;
    let mut failures = 0;

    for line in ["12", "x", "30"] {
        match line.parse::<i32>() {
            Ok(n) => sum += n,
            Err(_) => failures += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Failures: {}", failures);
}
```
