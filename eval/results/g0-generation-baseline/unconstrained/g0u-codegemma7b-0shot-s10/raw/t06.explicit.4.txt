fn is_perfect(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    let mut sum = 1
    for i in 2..(n/2 + 1) {
        if n % i == 0 {
            sum += i
        }
    }
    return sum == n
}

    for i in 1..1000 {

        }
    }
}
