fn main() {
    let mut list = vec![4, 1, 7, 3, 9, 2];
    let mut counter = 0;
    for num in list.iter_mut() {
        if *num > 3 {
            counter += 1;
            print!("{}\n", num);
        } else {
            list.remove(list.iter().position(|&x| x == *num).unwrap());
        }
    }
    print!("{} value(s) kept\n", counter);
}
