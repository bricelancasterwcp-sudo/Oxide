fn is_prime(n: Int) -> Bool {
    if n <= 1 { return false }
    let mut i = 5
    while i * i <= n {
        if n % i == 0 || n % (i + 2) == 0 { return false }
        i += 6
    }
    true
}

fn count_primes(max: Int) -> Int {
    let mut count = 0
    for n in 2..max {
        if is_prime(n) { count += 1 }
    }
    count
}

fn main() {
    print(count_primes(100))
}
