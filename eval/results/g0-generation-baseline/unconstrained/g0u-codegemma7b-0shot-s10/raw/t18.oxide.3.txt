fn main() {
    let r1 = Rectangle { width: 3, height: 4 };
    let t1 = Triangle { a: 3, b: 4, c: 5 };
    let r2 = Rectangle { width: 2, height: 5 };
    print(perimeter(r1));
    print(perimeter(t1));
    print(perimeter(r2));
}

struct Rectangle { width: Int, height: Int }
struct Triangle { a: Int, b: Int, c: Int }

fn perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn perimeter(t: Triangle) -> Int {
    t.a + t.b + t.c
}
