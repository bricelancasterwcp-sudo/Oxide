```
fn keep_big(v: Vec<Int>, limit: Int) -> Vec<Int> {
    let mut result = vec()
    for x in v {
        if x > limit {
            result = push(result, x)
        }
    }
    result
}

fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = keep_big(nums, 10)
    print(bigger)
    drop nums
    drop bigger
}
```
