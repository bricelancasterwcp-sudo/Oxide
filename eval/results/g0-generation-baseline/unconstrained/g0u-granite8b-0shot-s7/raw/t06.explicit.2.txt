fn perfect_numbers(n: Int) -> Vec<Int> {
    let mut v = vec()
    for x in range(1, n) {
        let mut s = 0
        for y in range(1, x) {
            if x % y == 0 {
                s = s + y
            }
        }
        if s == x {
            push(v, x)
        }
    }
    v
}

fn main() {
    for n in perfect_numbers(1000) {
        print(n)
    }
}
