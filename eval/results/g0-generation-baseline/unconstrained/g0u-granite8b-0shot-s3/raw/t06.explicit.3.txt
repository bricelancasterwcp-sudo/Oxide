fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_big(&nums, 10))
    drop nums
    print(sum_big(&bigger, 10))
    drop bigger
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn is_perfect(n: Int) -> Bool {
    let sum = reduce_sum(&v, 0)
    sum == n
}

fn reduce_sum(v: &Vec<Int>, acc: Int) -> Int {
    let sum = acc
    for x in &v {
        if x != 0 {
            let s = reduce_sum(&v, acc + x)
            sum = s
        }
    }
    sum
}
