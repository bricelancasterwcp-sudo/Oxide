fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false;
    }
    let mut i = 2
    while i < n {
        if n % i == 0 {
            return false;
        }
        i = i + 1
    }
    true
}

fn main() {
    let count = 0
    let mut i = 2
    while i < 100 {
        if is_prime(i) {
            count = count + 1
        }
        i = i + 1
    }
    print(count)
}
