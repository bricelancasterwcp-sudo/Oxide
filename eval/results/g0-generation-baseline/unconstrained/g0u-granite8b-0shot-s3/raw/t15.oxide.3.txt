```text
fn parse_and_sum(input: Str) -> Option<Int> {
    let mut sum = 0
    for c in chars(input) {
        match parse_int(c) {
            Some(n) => sum += n,
            None => ()
        }
    }
    Some(sum)
}

fn main() {
    match parse_and_sum("12 30") {
        Some(n) => print(n),
        None => print_str("no numbers")
    }
}
```
