fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in 2..trunc(sqrt(to_float(n))) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let primes = filter(vec(), |n| is_prime(n))
    let count = len(primes)
    for p in primes {
        if p >= 100 {
            break
        }
        print(p)
    }
    print(count)
}
