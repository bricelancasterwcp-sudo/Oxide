fn transform_steps(n: Int) -> Int {
    let mut count = 0
    let mut current = n
    while current != 1 {
        if current % 2 == 0 {
            current = current / 2
        } else {
            current = current * 3 + 1
        }
        count = count + 1
    }
    count
}

fn main() {
    let steps = transform_steps(27)
    print(steps)
}
