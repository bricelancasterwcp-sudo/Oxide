fn main() {
    let values = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut max_value = -2147483648;

    for x in values {
        if x > 0 {
            positive_count += 1;
        }
        if x > max_value {
            max_value = x;
        }
    }

    print(positive_count);
    print(max_value);
}
