fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut v = v;
    for i in 0..v.len() {
        for j in i+1..v.len() {
            if v[i] > v[j] {
                let temp = v[i];
                v[i] = v[j];
                v[j] = temp;
            }
        }
    }
    v
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2);
    let sorted = sort(clone(&nums));
    for x in &sorted {
        print_str(x)
    }
    drop sorted
}
