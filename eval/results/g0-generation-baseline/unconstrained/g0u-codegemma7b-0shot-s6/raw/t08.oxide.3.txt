fn main() {
    let list = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut largest = list[0];
    for x in list {
        if x > 0 {
            positive_count += 1;
        }
        if x > largest {
            largest = x;
        }
    }
    print(positive_count);
    print(largest);
}
