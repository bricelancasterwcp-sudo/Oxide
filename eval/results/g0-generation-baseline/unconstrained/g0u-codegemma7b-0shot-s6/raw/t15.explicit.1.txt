```rust
fn main() {
    let strings = ["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;
    for s in strings {
        match parse_int(s) {
            Some(n) => {
                sum += n;
                count += 1;
            },
            None => (),
        }
    }
    print(sum);
    print(count);
}
```
