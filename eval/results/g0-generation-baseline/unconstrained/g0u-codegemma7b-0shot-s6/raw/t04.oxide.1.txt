fn is_prime(n: Int) -> Bool {
    if n <= 1 { false }
    let mut i = 2
    while i * i <= n {
        if n % i == 0 { false }
        i += 1
    }
    true
}

fn main() {
    let mut count = 0
    let lim = 100
    for n in 2..lim {
        if is_prime(n) { count += 1 }
    }
    print(count)
}
