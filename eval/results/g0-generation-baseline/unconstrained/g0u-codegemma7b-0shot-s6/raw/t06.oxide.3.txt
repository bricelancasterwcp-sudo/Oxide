fn is_perfect(x: Int) -> Bool {
    if x <= 1 { return False }
    let mut sum = 1;
    let mut i = 2;
    while i * i <= x {
        if x % i == 0 {
            sum += i + x / i;
        }
        i += 1;
    }
    sum == x
}

fn main() {
    for i in 0..1000 {
        if is_perfect(i) { print(i); print_str("\n") }
    }
}
