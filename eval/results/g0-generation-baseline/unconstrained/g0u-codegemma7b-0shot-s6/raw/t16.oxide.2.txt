fn main() {
    let nums = vec![5, 0, 4];
    for d in nums {
        match div(100, d) {
            Some(q) => print(q),
            None => print_str("undefined"),
        }
        print_str("\n");
    }
}

fn div(a: Int, b: Int) -> Option<Int> {
    if b == 0 {
        None
    } else {
        Some(trunc(a / b))
    }
}
