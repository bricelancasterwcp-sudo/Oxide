use std::io::stdin;

fn main() {
    let mut sum = 0;
    let mut fail_count = 0;

    for line in stdin().lines() {
        match line.unwrap().parse::<i32>() {
            Ok(n) => sum += n,
            Err(_) => fail_count += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Failed: {}", fail_count);
}
