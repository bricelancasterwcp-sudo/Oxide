fn is_prime(x: Int) -> Bool {
    if x <= 1 { return false }
    for i in range(2, trunc(x.sqrt()) + 1) {
        if x % i == 0 { return false }
    }
    true
}

fn count_primes(max: Int) -> Int {
    let mut count = 0
    for i in range(2, max) {
        if is_prime(i) { count = count + 1 }
    }
    count
}

fn main() {
    print(count_primes(100))
}
