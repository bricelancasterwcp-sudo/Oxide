fn div(a: Int, b: Int) -> Option<Int> {
    if b == 0 {
        None
    } else {
        Some(a / b)
    }
}

fn main() {
    match div(100, 5) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
    match div(100, 0) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
    match div(100, 4) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
}
