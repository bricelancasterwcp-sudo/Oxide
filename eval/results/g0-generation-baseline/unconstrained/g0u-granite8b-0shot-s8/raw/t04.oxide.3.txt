fn main() {
    let mut i = 2;
    let mut count = 0;
    while i < 100 {
        if is_prime(i) {
            count = count + 1;
        }
        i = i + 1;
    }
    print(count);
}

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false;
    }
    if n <= 3 {
        return true;
    }
    if n % 2 == 0 || n % 3 == 0 {
        return false;
    }
    let mut i = 5;
    while i * i <= n {
        if n % i == 0 || n % (i + 2) == 0 {
            return false;
        }
        i = i + 6;
    }
    true
}
