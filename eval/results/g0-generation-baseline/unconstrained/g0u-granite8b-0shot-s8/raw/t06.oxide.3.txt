fn perfect_numbers() {
    let ps = [2, 3, 5, 7, 11, 13, 17, 19];
    for p in ps {
        let mut n = p * (p + 1);
        while n < 1000 {
            let mut sum = 0;
            for i in 1..n {
                if n % i == 0 {
                    sum += i;
                }
            }
            if sum == n {
                print(n)
            }
            n += p;
        }
    }
}

perfect_numbers()
