```rust
fn sum_divisors(n: Int) -> Int {
    let mut sum = 1;
    let limit = (n / 2) + 1;
    for i in range(2, limit) {
        if n % i == 0 {
            sum = sum + i;
        }
    }
    sum
}

fn main() {
    for num in range(1, 1000) {
        if num == sum_divisors(num) {
            print(num)
        }
    }
}
```
