fn is_prime(n: Int) -> Bool {
    if n <= 1 { return False }
    for i in range(2, n) {
        if n % i == 0 { return False }
    }
    True
}

fn count_primes(max: Int) -> Int {
    let mut count = 0
    for n in range(2, max) {
        if is_prime(n) { count = count + 1 }
    }
    count
}

fn main() {
    print(count_primes(100))
}
