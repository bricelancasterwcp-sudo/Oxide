fn count_s(text: Str) -> Int {
    let mut count = 0;
    for char in chars(text) {
        if char == "s" {
            count += 1;
        }
    }
    count
}

fn distinct_chars(text: Str) -> Int {
    let mut chars_set = Vec::new();
    for char in chars(text) {
        if !chars_set.contains(&char) {
            chars_set.push(char);
        }
    }
    len(chars_set)
}

fn main() {
    let text = "mississippi";
    print(count_s(text));
    print_str("\n");
    print(distinct_chars(text));
}
