fn reverse_print(v: &Vec<Int>) -> Vec<Int> {
    let mut result = vec();
    for i in 0..len(v) {
        let x = get(v, len(v) - 1 - i);
        print_str(&x);
        push(result, x);
    }
    result
}

fn main() {
    let nums = vec(1, 2, 3, 4, 5);
    reverse_print(&nums);
    drop nums
}
