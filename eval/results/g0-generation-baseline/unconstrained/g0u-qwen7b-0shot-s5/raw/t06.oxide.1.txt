fn perfect_number(n: Int) -> Bool {
    let mut sum = 0
    for i in range(1, n) {
        if n % i == 0 {
            sum = sum + i
        }
    }
    sum == n
}

fn main() {
    for n in range(1, 1000) {
        if perfect_number(n) {
            print(n)
            print_str("\n")
        }
    }
}
