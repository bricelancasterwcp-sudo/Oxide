```rust
struct Item { name: Str, unit_price: Int, quantity: Int }

fn calculate_cost(item: Item) -> Int {
    item.unit_price * item.quantity
}

fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 };
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 };
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 };

    let total_cost = calculate_cost(pen.clone()) + calculate_cost(pad.clone()) + calculate_cost(ink.clone());
    print(total_cost);

    let mut items = vec![pen, pad, ink];
    items.sort_by_key(|item| calculate_cost(item.clone()));
    let most_expensive = items.last().unwrap();
    print_str(most_expensive.name);
}
```
