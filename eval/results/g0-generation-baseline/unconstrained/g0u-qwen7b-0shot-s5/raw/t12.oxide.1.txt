fn reverse(s: Str) -> Str {
    let mut chars: Vec<Str> = chars(s);
    let mut reversed: Vec<Str> = vec();
    while len(chars) > 0 {
        reversed = push(reversed, pop(chars));
    }
    concat(reversed, "")
}

fn main() {
    let s = "stack";
    let reversed_s = reverse(s);
    print_str(reversed_s);
}
