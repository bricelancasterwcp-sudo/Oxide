fn main() {
    let inputs = vec!["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;

    for s in inputs.iter() {
        match parse_int(s) {
            Some(n) => {
                sum += n;
            }
            None => {
                count += 1;
            }
        }
    }

    print(sum);
    print(count);
}
