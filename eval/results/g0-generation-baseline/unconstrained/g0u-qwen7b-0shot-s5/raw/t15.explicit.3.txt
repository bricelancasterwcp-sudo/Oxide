fn main() {
    let nums = vec!["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;
    
    for s in &nums {
        match parse_int(s) {
            Some(n) => {
                sum = sum + n;
            },
            None => {
                count = count + 1;
            }
        }
    }
    
    print(sum);
    drop sum;
    print(count);
    drop count;
}
