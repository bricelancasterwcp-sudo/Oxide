```rust
fn fibonacci(n: Int) -> Int {
    if n <= 1 {
        1
    } else {
        let (mut a, mut b) = (1, 1);
        for _ in 2..n {
            let temp = a + b;
            drop a;
            a = b;
            b = temp;
        }
        b
    }
}

fn main() {
    let term = fibonacci(20);
    print(term);
    drop term;
}
```
