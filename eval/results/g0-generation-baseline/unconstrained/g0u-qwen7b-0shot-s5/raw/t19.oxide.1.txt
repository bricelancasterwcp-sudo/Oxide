struct Rect { width: Int, height: Int }

fn area(r: Rect) -> Int {
    r.width * r.height
}

fn perimeter(r: Rect) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let r = Rect { width: 7, height: 4 };
    print(area(r));         // Consumes r for area calculation
    let p = Rect { ..r };   // Clones r for perimeter calculation
    print(perimeter(p));    // Consumes p for perimeter calculation
}
