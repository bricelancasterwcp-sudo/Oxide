use oxide_explicit::prelude::*;

fn is_palindrome(text: &str) -> bool {
    for i in 0..(text.len() / 2) {
        if text.chars().nth(i) != text.chars().nth(text.len() - i - 1) {
            return false;
        }
    }
    true
}

fn main() {
    let level = "level";
    let stereo = "stereo";
    let rotor = "rotor";

    print(is_palindrome(&level));
    print(is_palindrome(&stereo));
    print(is_palindrome(&rotor));
}
