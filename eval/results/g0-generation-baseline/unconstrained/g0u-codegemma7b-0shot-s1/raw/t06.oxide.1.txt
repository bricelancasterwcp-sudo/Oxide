use std::collections::HashSet;

fn divisor_sum(n: Int) -> Int {
    let mut sum = 0;
    for i in 1..(n / 2 + 1) {
        if n % i == 0 {
            sum += i;
        }
    }
    sum
}

fn main() {
    for n in 0..1000 {
        if divisor_sum(n) == n && n != 0 {
            print!("{}\n", n);
        }
    }
}
