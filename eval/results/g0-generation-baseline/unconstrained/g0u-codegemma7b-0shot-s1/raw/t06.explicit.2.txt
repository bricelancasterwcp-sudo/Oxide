use oxide::ox;

fn is_perfect(n: int) -> bool {
    let mut sum = 0;
    for i in range(1, n) {
        if n % i == 0 {
            sum += i;
        }
    }
    return sum == n;
}

fn main() {
    for n in range(1, 1000) {
        if is_perfect(n) {
            print(n);
            print_str("\n");
        }
    }
}
