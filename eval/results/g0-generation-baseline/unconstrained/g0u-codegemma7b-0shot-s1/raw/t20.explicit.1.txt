use Oxide_Explicit as ox;

fn main() {
    let pen = ox::Tuple(3, 4);
    let pad = ox::Tuple(9, 2);
    let ink = ox::Tuple(5, 3);

    let cost_pen = pen.0 * pen.1;
    let cost_pad = pad.0 * pad.1;
    let cost_ink = ink.0 * ink.1;

    let total_cost = cost_pen + cost_pad + cost_ink;

    print(total_cost);

    if cost_pen >= cost_pad {
        if cost_pen >= cost_ink {
            print("pen");
        } else {
            print("ink");
        }
    } else {
        if cost_pad >= cost_ink {
            print("pad");
        } else {
            print("ink");
        }
    }
}
