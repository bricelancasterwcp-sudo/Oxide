#![no_std]

use std::io::{stdin, stdout, Write};

fn main() {
    let mut sum = 0;
    let mut fail = 0;

    for line in stdin().lines() {
        let line = line.expect("Failed to read line");
        if let Ok(num) = line.parse::<i32>() {
            sum += num;
        } else {
            fail += 1;
        }
    }

    print!("Sum: {}\n", sum);
    print!("Failed: {}\n", fail);
}
