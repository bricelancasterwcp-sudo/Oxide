fn sort_and_print(nums: &mut Vec<Int>) {
    for i in 0..nums.len() {
        for j in i + 1..nums.len() {
            if nums[i] > nums[j] {
                std::mem::swap(&mut nums[i], &mut nums[j]);
            }
        }
    }
    for num in &nums {
        print!("{}\n", num)
    }
}

fn main() {
    let mut numbers = vec![5, 3, 8, 1, 9, 2];
    sort_and_print(&mut numbers);
}
