fn count_chars(s: &Str) -> (Int, Int) {
    let mut count = 0;
    let mut seen = Vec::new();
    for c in chars(s) {
        if c == "s" {
            count = count + 1;
        }
        if !seen.contains(&c) {
            seen = push(seen, c);
        }
    }
    (count, seen.len())
}

fn main() {
    let text = "mississippi";
    let (s_count, distinct_count) = count_chars(&text);
    print_str("Number of 's': ");
    print(s_count);
    print_str("\nDistinct characters: ");
    print(distinct_count);
    drop text;
}
