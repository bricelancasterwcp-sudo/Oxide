fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v;
    for i in 0..sorted.len() {
        for j in 0..sorted.len() - 1 - i {
            if sorted[j] > sorted[j + 1] {
                let temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }
    sorted
}

fn main() {
    let nums = push(push(push(push(push(vec(), 5), 3), 8), 1), 9), 2);
    let sorted = sort(nums);
    for x in &sorted {
        print_str(x.to_string())
    }
    drop sorted
}
