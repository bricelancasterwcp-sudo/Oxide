fn transform(n: Int) -> Int {
    let mut count = 0;
    let mut x = n;
    while x != 1 {
        if x % 2 == 0 {
            x = x / 2;
        } else {
            x = 3 * x + 1;
        }
        count = count + 1;
    }
    count
}

fn main() {
    let steps = transform(27);
    print(steps);
}
