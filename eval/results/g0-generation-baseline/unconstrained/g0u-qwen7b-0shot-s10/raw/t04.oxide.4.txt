fn is_prime(n: Int) -> Bool {
    if n <= 1 { return false }
    for i in range(2, to_int(to_float(n).sqrt() + 1)) {
        if n % i == 0 { return false }
    }
    true
}

fn count_primes() -> Int {
    let mut count = 0
    for n in range(2, 100) {
        if is_prime(n) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes())
}
