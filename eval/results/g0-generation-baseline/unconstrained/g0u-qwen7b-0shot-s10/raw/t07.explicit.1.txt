```rust
fn sum_multiples() -> Int {
    let mut sum = 0
    let mut count = 0
    let mut i = 1
    while count < 8 {
        let multiple = i * 3
        sum = sum + multiple
        count = count + 1
        i = i + 1
    }
    sum
}

fn main() {
    let total = sum_multiples()
    print(total)
}
```
