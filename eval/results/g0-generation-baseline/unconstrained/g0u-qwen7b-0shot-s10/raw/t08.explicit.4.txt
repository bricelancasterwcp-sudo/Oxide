fn count_pos(v: &Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if *x > 0 {
            count = count + 1;
        }
    }
    count
}

fn max_value(v: &Vec<Int>) -> Int {
    let mut max = i32::MIN;
    for x in v {
        if *x > max {
            max = *x;
        }
    }
    max
}

fn main() {
    let nums = vec![3, 8, -2, 12, 7];
    let pos_count = count_pos(&nums);
    let max = max_value(&nums);
    print(pos_count);
    print(max);
}
