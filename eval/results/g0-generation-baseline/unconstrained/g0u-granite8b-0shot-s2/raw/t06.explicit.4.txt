fn perfect(n: Int) -> Bool {
    let mut sum = 0
    let mut d = 1
    while d * d <= n {
        if n % d == 0 {
            sum = sum + d
            if n / d != d {
                sum = sum + n / d
            }
        }
        d = d + 1
    }
    sum == n
}

fn main() {
    let mut n = 2
    while n < 1000 {
        if perfect(n) {
            print(n)
        }
        n = n + 1
    }
}
