```text
fn parse_int(s: Str) -> Option<Int> {
    if s.len() == 0 {
        return None
    }
    let mut total = 0
    for c in &s {
        if c < '0' || c > '9' {
            return None
        }
        total = total * 10 + (c as Int - '0' as Int)
    }
    Some(total)
}

fn print_sum_of_integers() {
    let x = parse_int("12")
    let y = parse_int("x")
    let z = parse_int("30")
    let sum = match x {
        Some(i) => {
            match y {
                Some(j) => {
                    match z {
                        Some(k) => i + j + k,
                        None => 0,
                    }
                }
                None => 0,
            }
        }
        None => 0,
    }
    print(sum)
}

fn main() {
    print_sum_of_integers()
}
```

`parse_int` only reads the input string (`&Str` arg), so it must be.
`print_sum_of_integers` calls it, and returns its result, so it consumes.
Main calls it and discards the result; no drop needed.
