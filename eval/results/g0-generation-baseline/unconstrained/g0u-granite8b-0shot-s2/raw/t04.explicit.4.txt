fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }

    let mut is_prime = true
    let two = 2
    for i in range(two, n) {
        if n % i == 0 {
            is_prime = false
            break
        }
    }

    is_prime
}

fn main() {
    let prime_count = 0
    let two = 2
    for i in range(two, 100) {
        if is_prime(i) {
            let prime_count = prime_count + 1
        }
    }

    print(prime_count)
}
