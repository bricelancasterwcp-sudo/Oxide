let x = parse_int("12")
let y = parse_int("x")
let z = parse_int("30")

match x {
    Some(i) => {
        match y {
            Some(j) => {
                match z {
                    Some(k) => {
                        let sum = i + j + k
                        print(sum)
                    }
                    None => {
                        print_str("z was not an integer")
                    }
                }
            }
            None => {
                print_str("y was not an integer")
            }
        }
    }
    None => {
        print_str("x was not an integer")
    }
}

match x {
    Some(_) => {
        print(1)
    }
    None => {
        print(0)
    }
}
