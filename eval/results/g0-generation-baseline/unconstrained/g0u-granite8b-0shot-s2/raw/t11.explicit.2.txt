fn sort(v: &mut Vec<Int>) {
    for i in range(0, len(v)) {
        let mut smallest = i
        for j in range(i+1, len(v)) {
            if get(v, j) < get(v, smallest) {
                smallest = j
            }
        }
        if smallest != i {
            swap(v, i, smallest)
        }
    }
}

fn swap(v: &mut Vec<Int>, i: Int, j: Int) {
    let tmp = clone(get(v, i))
    drop v[i]
    v[i] = get(v, j)
    drop v[j]
    v[j] = tmp
}

fn main() {
    let mut values = push(push(vec(), 5), 40)
    sort(&mut values)
    for x in &values {
        print(x)
        print("\n")
    }
}
