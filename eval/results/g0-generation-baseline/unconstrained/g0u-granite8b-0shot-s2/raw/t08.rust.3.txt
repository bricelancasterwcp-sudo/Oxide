fn main() {
    let list = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut largest_value = i32::min_value();

    for num in list {
        if num > 0 {
            positive_count += 1;
        }
        if num > largest_value {
            largest_value = num;
        }
    }

    println!("Positive numbers: {}", positive_count);
    println!("Largest value: {}", largest_value);
}
