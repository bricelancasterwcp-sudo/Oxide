fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }

    for i in range(2, n) {
        if n % i == 0 {
            return false
        }
    }

    true
}

fn main() {
    let prime_count = 0
    for i in range(2, 100) {
        if is_prime(i) {
            let prime_count = prime_count + 1
        }
    }

    print(prime_count)
}
