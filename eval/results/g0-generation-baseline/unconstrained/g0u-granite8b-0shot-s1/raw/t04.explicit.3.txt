fn main() {
    let primes = vec();
    let mut i = 2;
    while i < 100 {
        if is_prime(i) {
            primes = push(primes, i)
        }
        i = i + 1
    }
    print(len(&primes))
    drop primes
}

fn is_prime(n: Int) -> Bool {
    if n == 0 || n == 1 {
        return false
    }
    let range = range(2, n)
    let mut is_prime = true
    for j in &range {
        if n % j == 0 {
            is_prime = false
            break
        }
    }
    is_prime
}
