fn is_perfect(n: Int) -> Bool {
    let mut sum = 0;
    let mut i = 1;
    while i < n {
        if n % i == 0 {
            sum = sum + i;
        }
        i = i + 1;
    }
    sum == n
}

fn main() {
    for i in 2..1000 {
        if is_perfect(i) {
            print_int(i);
            print_str("\n");
        }
    }
}
