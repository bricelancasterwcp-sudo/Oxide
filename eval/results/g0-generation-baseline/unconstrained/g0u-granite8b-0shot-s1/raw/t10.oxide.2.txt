struct List { head: Int, tail: Vec<Int> }

fn keep_bigger_than_3(xs: Vec<Int>) -> Vec<Int> {
    let mut kept = vec()
    for x in xs {
        if x > 3 {
            kept = push(kept, x)
        }
    }
    kept
}

fn main() {
    let xs = push(push(push(push(vec(), 4), 1), 7), 3)
    let kept = keep_bigger_than_3(xs)
    print(len(kept))
    for x in kept {
        print(x)
    }
}
