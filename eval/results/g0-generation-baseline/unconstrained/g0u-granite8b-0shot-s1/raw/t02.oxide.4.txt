fn transform(mut x: Int) -> Int {
    while x > 1 {
        if x % 2 == 0 {
            x = x / 2;
        } else {
            x = (x * 3) + 1;
        }
    }

}

    while n > 1 {
        print(n)
        n = transform(n)
    }

}
