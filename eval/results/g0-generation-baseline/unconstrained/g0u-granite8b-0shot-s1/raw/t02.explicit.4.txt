fn main() {
    let mut two = 2;
    let mut seven = 7;
    let mut result = 1;
    while result < 1000 {
        if result % two == 0 {
            result = result / two;
            drop two;
        } else {
            result = result * seven + 1;
            drop seven;
        }
        drop result;
        let result = result;
    }
    print(result);
}
