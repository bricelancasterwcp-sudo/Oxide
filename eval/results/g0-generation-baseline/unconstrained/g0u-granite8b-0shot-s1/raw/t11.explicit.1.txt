fn task(vec: &Vec<Int>) -> Vec<Int> {
    let sorted = vec
    let mut i = 0
    let mut j = 1
    while i < len(&sorted) - 1 {
        while j < len(&sorted) && get(&sorted, j) < get(&sorted, j - 1) {
            j = j + 1
        }
        if j < len(&sorted) {
            let temp = get(&sorted, i)
            set(&sorted, i, get(&sorted, j))
            set(&sorted, j, temp)
            i = i + 1
            j = i + 1
        }
    }
    sorted
}

fn main() {
    let list = push(push(push(push(vec(), 5), 3), 8), 1), 9)
    let sorted = task(&list)
    for i in range(0, len(&sorted)) {
        print(get(&sorted, i))
        print_str("\n")
    }
    drop list
    drop sorted
}
