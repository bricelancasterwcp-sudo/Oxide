fn main() {
    let a = "12".parse::<i32>().unwrap_or(0);
    let b = "x".parse::<i32>().unwrap_or(0);
    let c = "30".parse::<i32>().unwrap_or(0);

    let sum = vec![a, b, c].into_iter().sum();
    println!("{}", sum);

    let count = vec![a, b, c].into_iter().filter_map(Result::ok).count();
    println!("{}", count);
}
