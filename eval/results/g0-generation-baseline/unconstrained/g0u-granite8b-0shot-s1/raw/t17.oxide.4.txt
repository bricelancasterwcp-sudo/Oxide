struct List { head: Int, tail: Vec<Int> }

fn lookup(list: &List, pos: Int) -> Int {
    if pos == 0 {
        return list.head
    }
    for x in list.tail.iter() {
        if pos == 1 {
            return x
        }
        pos = pos - 1
    }
    -1
}

fn main() {
    let list = List { head: 10, tail: vec![20, 30] }
    print(lookup(&list, 1))
    print(lookup(&list, 5))
}
