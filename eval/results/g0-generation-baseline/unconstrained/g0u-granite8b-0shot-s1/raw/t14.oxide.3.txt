struct Text { bytes: Vec<Option<Bool>> }

fn count_letter(s: Str, letter: u8) -> Int {
    let mut count = 0
    for ch in chars(s) {
        if ch as u8 == letter {
            count = count + 1
        }
    }
    count
}

fn distinct_chars(s: Str) -> Int {
    let mut distinct = 0
    for ch in chars(s) {
        let c = ch as u8
        if get(s.bytes, c)? == None {
            distinct = distinct + 1
            s.bytes[c] = Some(true)
        }
    }
    distinct
}

fn main() {
    let s = Text { bytes: vec() }
    print(count_letter("mississippi", 's'))
    print(distinct_chars("mississippi"))
}
