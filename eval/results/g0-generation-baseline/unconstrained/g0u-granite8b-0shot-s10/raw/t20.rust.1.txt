```rust
fn main() {
    let pen = (3, 4);
    let pad = (9, 2);
    let ink = (5, 3);

    let total_cost = pen.0 * pen.1 + pad.0 * pad.1 + ink.0 * ink.1;
    let highest_cost = [pen, pad, ink]
        .iter()
        .max_by_key(|item| item.0 * item.1)
        .unwrap();

    println!("Total cost: {}", total_cost);
    println!("Item with the highest cost: {:?}", highest_cost);
}
```
