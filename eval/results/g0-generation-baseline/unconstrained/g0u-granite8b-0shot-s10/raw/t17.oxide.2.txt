fn list_get(l: Vec<Int>, pos: Int) -> Option<Int> {
    if pos < 0 || pos >= len(l) {
        return None
    }
    Some(get(l, pos)?)
}

fn main() {
    let l = push(push(push(vec(), 10), 20), 30)
    match list_get(l, 0) {
        Some(n) => print(n),
        None => print_str("out of bounds"),
    }
    match list_get(l, 4) {
        Some(n) => print(n),
        None => print_str("out of bounds"),
    }
}
