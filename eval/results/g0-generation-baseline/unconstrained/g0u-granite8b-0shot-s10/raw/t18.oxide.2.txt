struct Rectangle { width: Int, height: Int }
struct Triangle { side1: Int, side2: Int, side3: Int }

fn perimeter(shape: Variant) -> Int {
    match shape {
        Rectangle { width, height } => width + width + height + height,
        Triangle { side1, side2, side3 } => side1 + side2 + side3,
        _ => panic("unknown shape")
    }
}

fn main() {
    let rect = Rectangle { width: 3, height: 4 }
    let tri = Triangle { side1: 3, side2: 4, side3: 5 }
    let seq = push(push(push(vec(), rect), tri), Rectangle { width: 2, height: 5 })
    for shape in seq {
        print(perimeter(shape))
    }
}
