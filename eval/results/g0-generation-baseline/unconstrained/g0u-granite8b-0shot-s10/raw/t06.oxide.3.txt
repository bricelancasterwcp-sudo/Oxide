fn perfect_numbers(max: Int) -> Vec<Int> {
    let mut v = vec()
    for n in range(2, max) {
        let mut sum = 0
        for d in range(1, n) {
            if n % d == 0 {
                sum = sum + d
            }
        }
        if sum == n {
            v = push(v, n)
        }
    }
    v
}

fn main() {
    let v = perfect_numbers(1000)
    for n in v {
        print(n)
        print_str("\n")
    }
}
