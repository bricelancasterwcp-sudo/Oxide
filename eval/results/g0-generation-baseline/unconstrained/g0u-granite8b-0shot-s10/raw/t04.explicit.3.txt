fn is_prime(x: Int) -> Bool {
    if x <= 1 {
        return false
    }
    let mut y = 2
    while y < x {
        if x % y == 0 {
            return false
        }
        y = y + 1
    }
    true
}

fn main() {
    let count = 0
    for x in 2..100 {
        if is_prime(x) {
            count = count + 1
        }
    }
    print(count)
}
