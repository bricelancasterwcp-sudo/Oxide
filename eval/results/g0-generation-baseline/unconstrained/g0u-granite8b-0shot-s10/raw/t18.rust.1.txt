```rust
fn main() {
    println!("{}", Rectangle::new(3, 4).perimeter());
    println!("{}", Triangle::new(3, 4, 5).perimeter());
    println!("{}", Rectangle::new(2, 5).perimeter());
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn new(width: u32, height: u32) -> Rectangle {
        Rectangle { width, height }
    }

    fn perimeter(&self) -> u32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    a: u32,
    b: u32,
    c: u32,
}

impl Triangle {
    fn new(a: u32, b: u32, c: u32) -> Triangle {
        Triangle { a, b, c }
    }

    fn perimeter(&self) -> u32 {
        self.a + self.b + self.c
    }
}
```
