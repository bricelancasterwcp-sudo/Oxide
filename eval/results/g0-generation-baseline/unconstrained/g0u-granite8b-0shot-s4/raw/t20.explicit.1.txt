fn main() {
    let pen = Item { name: "pen", price: 3, quantity: 4 };
    let pad = Item { name: "pad", price: 9, quantity: 2 };
    let ink = Item { name: "ink", price: 5, quantity: 3 };

    let items = vec!(pen, pad, ink);

    let mut total_price = 0;
    for item in items {
        total_price += calculate_price(&item);
    }

    println!("Total cost: {}", total_price);

    let most_expensive_item = items.iter().max_by_key(|item| calculate_price(item)).unwrap();
    println!("Most expensive item: {}", most_expensive_item.name);
}

struct Item {
    name: String,
    price: u32,
    quantity: u32,
}

fn calculate_price(item: &Item) -> u32 {
    item.price * item.quantity
}
