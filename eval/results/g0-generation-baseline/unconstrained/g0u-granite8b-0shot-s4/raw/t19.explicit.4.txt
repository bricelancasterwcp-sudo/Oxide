fn main() {
    let rectangle = Rectangle { width: 7, height: 4 };
    let area = rectangle.area();
    let perimeter = rectangle.perimeter();
    print(area);
    print(perimeter);
}

struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn area(&self) -> Int {
        self.width * self.height
    }

    fn perimeter(&self) -> Int {
        2 * (self.width + self.height)
    }
}
