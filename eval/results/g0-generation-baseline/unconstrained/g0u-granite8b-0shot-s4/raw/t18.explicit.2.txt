```rust
struct Rectangle {
    width: i32,
    height: i32,
}

struct Triangle {
    side1: i32,
    side2: i32,
    side3: i32,
}

trait Shape {
    fn perimeter(&self) -> i32;
}

impl Shape for Rectangle {
    fn perimeter(&self) -> i32 {
        2 * (self.width + self.height)
    }
}

impl Shape for Triangle {
    fn perimeter(&self) -> i32 {
        self.side1 + self.side2 + self.side3
    }
}

fn main() {
    let rectangle = Rectangle { width: 3, height: 4 };
    let triangle = Triangle { side1: 3, side2: 4, side3: 5 };
    let sequence = [rectangle, triangle, rectangle];

    for shape in sequence {
        println!("{}", perimeter(&shape));
    }
}
```
