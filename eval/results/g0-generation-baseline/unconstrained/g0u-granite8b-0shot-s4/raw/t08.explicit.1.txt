fn main() {
    let nums = vec(3, 8, -2, 12, 7);
    let positive = 0;
    let largest = nums[0];
    for num in &nums {
        if num > 0 {
            positive = positive + 1;
        }
        if num > largest {
            largest = num;
        }
    }
    print(positive);
    print(largest);
}
