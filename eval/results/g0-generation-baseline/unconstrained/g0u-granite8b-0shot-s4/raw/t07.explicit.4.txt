fn main() {
    let mut v = vec();
    for i in range(1, 9) {
        if i % 3 == 0 {
            v = push(v, i);
        }
    }
    let sum = sum(&v);
    print(sum);
    drop sum;
    drop v;
}

fn sum(v: &Vec<Int>) -> Int {
    let mut acc = 0;
    for x in &v {
        acc = acc + x;
    }
    acc
}
