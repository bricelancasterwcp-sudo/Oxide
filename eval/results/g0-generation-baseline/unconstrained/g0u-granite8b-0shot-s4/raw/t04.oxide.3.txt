fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in range(2, n) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let count = len(range(2, 100))?
    for i in range(2, 100) {
        if is_prime(i) {
            print(i)
        }
    }
    print(count)
}
