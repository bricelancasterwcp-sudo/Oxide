use std::io;

fn main() {
    let mut list = vec![10, 20, 30];

    let pos1 = read_position();
    let pos5 = read_position();

    println!("{}", lookup(&list, pos1));
    println!("{}", lookup(&list, pos5));
}

fn lookup(list: &Vec<i32>, position: usize) -> i32 {
    match list.get(position) {
        Some(val) => *val,
        None => -1,
    }
}

fn read_position() -> usize {
    let mut input = String::new();

    match io::stdin().read_line(&mut input) {
        Ok(_) => {
            let trimmed = input.trim();
            match trimmed.parse::<usize>() {
                Ok(num) => num,
                Err(_) => 0,
            }
        }
        Err(_) => 0,
    }
}
