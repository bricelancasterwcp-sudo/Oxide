fn main() {
    let pen_cost = 3 * 4;
    let pad_cost = 9 * 2;
    let ink_cost = 5 * 3;
    let total_cost = pen_cost + pad_cost + ink_cost;
    print(total_cost);
    if pen_cost > pad_cost {
        if pen_cost > ink_cost {
            print("Pen");
        } else {
            print("Ink");
        }
    } else {
        if pad_cost > ink_cost {
            print("Pad");
        } else {
            print("Ink");
        }
    }
    drop pen_cost
    drop pad_cost
    drop ink_cost
    drop total_cost
}
