fn is_palindrome(s: &str) -> bool {
    let mut chars = s.chars().peekable();
    while let (Some(front), Some(back)) = (chars.next(), chars.next_back()) {
        if front != back {
            return false;
        }
    }
    true
}

fn main() {
    let texts = ["level", "stereo", "rotor"];
    for text in texts {
        println!("{}", is_palindrome(text));
    }
}
