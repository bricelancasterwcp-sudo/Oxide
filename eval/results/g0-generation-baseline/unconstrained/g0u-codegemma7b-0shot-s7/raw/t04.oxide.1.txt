fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }
    for i in 2..trunc(sqrt(n)) + 1 {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn count_primes(limit: Int) -> Int {
    let mut count = 0
    for i in 0..limit {
        if is_prime(i) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
