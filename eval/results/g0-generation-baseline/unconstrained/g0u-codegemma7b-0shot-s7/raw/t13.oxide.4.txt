fn is_palindrome(s: &str) -> bool {
    let mut i = 0;
    let mut j = s.len() - 1;
    while i < j {
        if s.chars().nth(i) != s.chars().nth(j) {
            return false;
        }
        i += 1;
        j -= 1;
    }
    true
}

fn main() {
    let texts = vec!["level", "stereo", "rotor"];
    for text in texts {
        println!("{}", is_palindrome(text));
    }
}
