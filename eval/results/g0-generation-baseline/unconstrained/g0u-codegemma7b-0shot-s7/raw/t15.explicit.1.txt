fn main() {
    let numbers = ["12", "x", "30"];
    let mut sum = 0;
    let mut count = 0;
    for number in numbers {
        match parse_int(number) {
            Some(n) => {
                sum = sum + n;
                count = count + 1;
            }
            None => (),
        }
    }
    print(sum)
    print(count)
}
