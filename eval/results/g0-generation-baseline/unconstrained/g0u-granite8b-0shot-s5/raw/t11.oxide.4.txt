fn main() {
    let v = range(5, 10)
    print(first_big(v));
    print(second(v).unwrap());
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}
