struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let pen = Reading { label: "pen", values: push(vec(), 3) }
    let pad = Reading { label: "pad", values: push(vec(), 2) }
    let ink = Reading { label: "ink", values: push(vec(), 3) }
    let r = Reading { label: "lab", values: push(push(push(vec(), 3), 42), 99) }
    let all = concat(concat(pen.label, " cost: "), int_to_str(pen.values[0] * pen.values.len()))
    print(all)
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}
