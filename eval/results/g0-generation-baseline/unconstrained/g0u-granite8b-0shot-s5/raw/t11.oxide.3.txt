fn main() {
    let v = range(5, 10)
    match second(v) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
    print(first_big(v))
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}
