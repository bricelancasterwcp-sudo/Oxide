fn main() {
    let nums = vec![3, 8, -2, 12, 7];
    let mut positive = 0;
    let mut largest = nums[0];
    for num in &nums {
        if *num > 0 {
            positive += 1;
            largest = std::cmp::max(largest, *num);
        }
    }
    print!("{} positive numbers, largest is {}", positive, largest);
}
