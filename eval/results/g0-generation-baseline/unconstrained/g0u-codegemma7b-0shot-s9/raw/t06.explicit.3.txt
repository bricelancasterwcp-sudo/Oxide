fn is_perfect(n: Int) -> Bool {
    let mut total = 0;
    for i in 1..(n/2 + 1) {
        if n % i == 0 {
            total = total + i;
        }
    }
    return total == n;
}

fn main() {
    for i in 0..1000 {
        if is_perfect(i) {
            print(i)
            print("\n")
        }
    }
}
