fn count_s(s: &str) -> (Int, Int) {
    let mut count = 0;
    let mut seen = Vec::new();
    for c in chars(s) {
        if c == 's' {
            count = count + 1;
        }
        seen.push(c)
    }
    (count, len(&seen))
}

fn main() {
    let (s_count, unique) = count_s("mississippi");
    print!("{} {}", s_count, unique)
}
