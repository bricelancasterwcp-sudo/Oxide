fn main() {
    let r1 = Rect { w: 3, h: 4 };
    let t1 = Triangle { a: 3, b: 4, c: 5 };
    let r2 = Rect { w: 2, h: 5 };
    print(r1.perimeter());
    print(t1.perimeter());
    print(r2.perimeter());
}

struct Rect { w: Int, h: Int }
struct Triangle { a: Int, b: Int, c: Int }

impl Rect {
    fn perimeter(&self) -> Int {
        2 * (self.w + self.h)
    }
}

impl Triangle {
    fn perimeter(&self) -> Int {
        self.a + self.b + self.c
    }
}
