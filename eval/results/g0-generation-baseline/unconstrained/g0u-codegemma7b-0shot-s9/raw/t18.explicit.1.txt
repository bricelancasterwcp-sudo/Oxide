fn perimeter(r: &Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn perimeter(t: &Triangle) -> Int {
    t.a + t.b + t.c
}

fn main() {
    let shapes = push(
        push(
            push(
                vec(),
                Rectangle {
                    width: 3,
                    height: 4,
                },
            ),
            Triangle {
                a: 3,
                b: 4,
                c: 5,
            },
        ),
        Rectangle {
            width: 2,
            height: 5,
        },
    );
    for s in &shapes {
        print(perimeter(s))
    }
    drop shapes
}

struct Rectangle {
    width: Int,
    height: Int,
}

struct Triangle {
    a: Int,
    b: Int,
    c: Int,
}
