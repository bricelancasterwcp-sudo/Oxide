fn main() {
    let pen = (3, 4);
    let pad = (9, 2);
    let ink = (5, 3);

    let cost_pen = pen.0 * pen.1;
    let cost_pad = pad.0 * pad.1;
    let cost_ink = ink.0 * ink.1;

    let total_cost = cost_pen + cost_pad + cost_ink;
    print(total_cost);

    if cost_pen >= cost_ink && cost_pen >= cost_pad {
        print_str("pen");
    } else if cost_pad >= cost_ink && cost_pad >= cost_pen {
        print_str("pad");
    } else {
        print_str("ink");
    }
}
