fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return False
    }
    for i in range(2, trunc(n.sqrt())) {
        if n % i == 0 {
            return False
        }
    }
    True
}

fn main() {
    let mut count = 0;
    for n in range(1, 100) {
        if is_prime(n) {
            count = count + 1
        }
    }
    print(count)
}
