```rust
fn main() {
    let numbers = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut max_value = i32::MIN;

    for &num in &numbers {
        if num > 0 {
            positive_count += 1;
        }
        if num > max_value {
            max_value = num;
        }
    }

    println!("Positive values: {}", positive_count);
    println!("Largest value: {}", max_value);
}
```
