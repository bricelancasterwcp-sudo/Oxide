fn main() {
    let mut list = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut max_value = list[0];

    for value in list.iter() {
        if *value > 0 {
            positive_count += 1;
        }
        if *value > max_value {
            max_value = *value;
        }
    }

    println!("Positive count: {}", positive_count);
    println!("Max value: {}", max_value);
}
