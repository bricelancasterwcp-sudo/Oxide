fn is_prime(n: Int) -> Bool {
    if n < 2 {
        False
    } else {
        for i in range(2, n) {
            if n % i == 0 {
                return False
            }
        }
        True
    }
}

fn count_primes(limit: Int) -> Int {
    let mut count = 0
    for i in range(2, limit) {
        if is_prime(i) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
