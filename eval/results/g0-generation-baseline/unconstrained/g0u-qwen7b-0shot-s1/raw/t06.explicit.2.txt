```rust
fn main() {
    let mut sums = vec![];
    for num in 1..1000 {
        let sum = (1..num).filter(|&x| num % x == 0).sum::<Int>();
        if sum == num {
            sums.push(num);
        }
    }
    for num in sums.iter() {
        print(&num);
        print_str("\n");
    }
    drop sums;
}
```
