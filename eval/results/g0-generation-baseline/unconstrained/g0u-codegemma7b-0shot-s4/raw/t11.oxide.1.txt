fn main() {
    let mut list = vec![5, 3, 8, 1, 9, 2];
    for x in range(0, len(list)) {
        for y in range(x + 1, len(list)) {
            if list[y] < list[x] {
                let tmp = list[x];
                list[x] = list[y];
                list[y] = tmp;
            }
        }
    }
    for x in list {
        print(x);
        print_str("\n");
    }
}
