fn is_prime(x: Int) -> Bool {
    if x <= 1 {
        return false
    }
    let mut i = 5
    while i * i <= x {
        if x % i == 0 || x % (i + 2) == 0 {
            return false
        }
        i += 6
    }
    true
}

fn count_primes(max: Int) -> Int {
    let mut count = 0
    for i in 2..=max {
        if is_prime(i) {
            count += 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
