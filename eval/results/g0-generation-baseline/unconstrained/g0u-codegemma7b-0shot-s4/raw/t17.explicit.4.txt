fn main() {
    let nums = vec![10, 20, 30];
    print_num(get(&nums, 1));
    print_num(get(&nums, 5));
}

fn get(v: &Vec<Int>, i: Int) -> Int {
    if i >= 0 && i < v.len() {
        v[i]
    } else {
        -1
    }
}

fn print_num(n: Int) {
    if n == -1 {
        print!("-1\n");
    } else {
        print!("{}\n", n);
    }
}
