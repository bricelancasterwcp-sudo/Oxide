```rust
fn divisors(n: Int) -> Vec<Int> {
    let mut v = vec![];
    for i in 1..n {
        if n % i == 0 {
            v.push(i)
        }
    }
    v
}

fn is_perfect(n: Int) -> Bool {
    let d = divisors(n);
    let sum = d.iter().fold(0, |acc, x| acc + x);
    sum == n
}

fn main() {
    for n in 1..1000 {
        if is_perfect(n) {
            print(n)
            print("\n")
        }
    }
}
```
