fn main() {
    let numbers = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut largest = numbers[0];

    for number in &numbers {
        if *number > 0 {
            positive_count += 1;
            if *number > largest {
                largest = *number;
            }
        }
    }

    print!("Positive count: {}\n", positive_count);
    print!("Largest number: {}\n", largest);
}
