fn is_prime(n: Int) -> Bool {
    n > 1 && for i in range(2, n) {
        if n % i == 0 { return false }
    }
    true
}

fn main() {
    let n = 100
    let primes = for x in range(2, n) {
        if is_prime(x) { Some(x) } else { None }
    }
    print(len(primes))
}
