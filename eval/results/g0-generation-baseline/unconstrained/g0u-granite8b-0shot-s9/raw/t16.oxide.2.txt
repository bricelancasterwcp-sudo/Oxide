fn divide(numerator: Int, denominator: Int) -> Option<Int> {
    if denominator == 0 {
        None
    } else {
        Some(numerator / denominator)
    }
}

fn main() {
    let numerator = 100;
    let quotient1 = divide(numerator, 5)?;
    let quotient2 = divide(numerator, 0)?;
    let quotient3 = divide(numerator, 4)?;

    print(quotient1);
    print(quotient2);
    print(quotient3);
}
