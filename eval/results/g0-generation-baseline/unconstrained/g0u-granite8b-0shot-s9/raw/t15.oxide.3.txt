fn parse_and_sum(s: Str) -> (Int, Int) {
    let n = len(s)
    let mut i = 0
    let mut sum = 0
    while i < n {
        if s[i] >= '0' && s[i] <= '9' {
            let c = s[i] - '0'
            let new_sum = sum * 10 + c
            if new_sum < sum {
                return (sum, 1)
            }
            sum = new_sum
        } else {
            return (sum, i)
        }
        i = i + 1
    }
    (sum, i)
}
