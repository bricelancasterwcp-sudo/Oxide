fn main() {
    let limit = 1000
    for x in range(2, limit) {
        let sum = perfect_sum(x)
        if sum? == x {
            print(x)
        }
    }
}

fn perfect_sum(x: Int) -> Option<Int> {
    let mut sum = 0
    for y in range(1, x) {
        if x % y == 0 {
            sum = sum + y
        }
    }
    Some(sum)
}
