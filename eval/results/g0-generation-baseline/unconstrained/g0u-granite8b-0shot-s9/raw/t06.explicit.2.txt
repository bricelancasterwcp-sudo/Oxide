fn main() {
    let n = 2;
    let mut sum = 0;
    let mut i = 1;
    while i < n {
        let mut j = 1;
        while j < i {
            if i % j == 0 {
                sum = sum + j;
            }
            j = j + 1;
        }
        if sum == i {
            print(i)
            sum = 0;
        }
        i = i + 1;
    }
}
