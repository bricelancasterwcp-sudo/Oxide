fn is_prime(n: Int) -> Bool {
    if n == 1 {
        return false
    }
    let upper = trunc(to_float(n) / 2)
    for i in range(2, upper + 1) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let count = 0
    for x in range(1, 100) {
        if is_prime(x) {
            count = count + 1
        }
    }
    print(count)
}
