```rust
fn main() {
    let nums = vec![4, 1, 7, 3, 9, 2];
    let mut i = 0;
    let mut count = 0;
    while i < nums.len() {
        let x = &nums[i];
        if *x > 3 {
            print!("{}", x);
            count = count + 1;
        }
        i = i + 1;
    }
    println!("\n{}", count);
}
```
