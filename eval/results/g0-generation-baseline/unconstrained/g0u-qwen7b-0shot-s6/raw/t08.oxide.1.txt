fn count_positive(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn max_value(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max? < x {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let values = push(push(push(push(vec(), 3), 8), -2), 12)
    values = push(values, 7)
    print(count_positive(values))
    print(max_value(values))
}
