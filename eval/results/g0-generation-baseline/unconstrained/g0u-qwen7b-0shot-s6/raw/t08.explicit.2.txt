fn count_and_max(values: &Vec<Int>) -> (Int, Int) {
    let mut count = 0;
    let mut max = i32::MIN;
    for &x in values {
        if x > 0 {
            count = count + 1;
        }
        if x > max {
            max = x;
        }
    }
    (count, max)
}

fn main() {
    let nums = push(push(push(push(vec(), 3), 8), -2), 12);
    nums = push(nums, 7);
    let (positive_count, max_value) = count_and_max(&nums);
    print(positive_count);
    print(max_value);
    drop nums;
}
