```rust
fn sum_multiples_of_3(limit: Int) -> Int {
    let mut sum = 0;
    let mut multiple = 3;
    while multiple <= limit {
        sum = sum + multiple;
        multiple = multiple + 3;
    }
    sum
}

fn main() {
    let limit = 24;
    let sum = sum_multiples_of_3(limit);
    print(sum);
}
```
