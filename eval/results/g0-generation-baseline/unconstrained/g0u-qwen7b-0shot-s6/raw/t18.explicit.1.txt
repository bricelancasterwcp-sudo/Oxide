```rust
struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn perimeter(&self) -> Int {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    a: Int,
    b: Int,
    c: Int,
}

impl Triangle {
    fn perimeter(&self) -> Int {
        self.a + self.b + self.c
    }
}

fn main() {
    let rect1 = Rectangle { width: 3, height: 4 };
    let tri = Triangle { a: 3, b: 4, c: 5 };
    let rect2 = Rectangle { width: 2, height: 5 };

    print(rect1.perimeter());
    print(tri.perimeter());
    print(rect2.perimeter());
}
```
