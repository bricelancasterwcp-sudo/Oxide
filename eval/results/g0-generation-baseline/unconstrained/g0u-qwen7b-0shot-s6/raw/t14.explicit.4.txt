```rust
fn count_s(s: Str) -> Int {
    let count = 0;
    for c in chars(s) {
        if c == "s" {
            count = count + 1;
        }
    }
    count
}

fn distinct_chars(s: Str) -> Int {
    let mut chars_map = {};
    for c in chars(s) {
        chars_map.insert(c, true);
    }
    chars_map.len()
}

fn main() {
    let text = "mississippi";
    print(count_s(&text));
    print(distinct_chars(&text));
}
```
