fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut v = v;
    for i in 0..v.len() {
        for j in 0..v.len() - i - 1 {
            if v[j] > v[j + 1] {
                let temp = v[j];
                v[j] = v[j + 1];
                v[j + 1] = temp;
            }
        }
    }
    v
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2);
    let sorted = sort(nums.clone());
    for x in sorted {
        print_str(x.to_string());
    }
    drop sorted;
}
