```rust
fn main() {
    println!("{}", rectangle_perimeter(3, 4));
    println!("{}", triangle_perimeter(3, 4, 5));
    println!("{}", rectangle_perimeter(2, 5));
}

fn rectangle_perimeter(width: u32, height: u32) -> u32 {
    2 * (width + height)
}

fn triangle_perimeter(a: u32, b: u32, c: u32) -> u32 {
    a + b + c
}
```
