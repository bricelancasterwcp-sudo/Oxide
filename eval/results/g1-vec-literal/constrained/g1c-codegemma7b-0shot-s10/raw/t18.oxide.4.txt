fn main() {
    let r1 = Rectangle { width: 3, height: 4 }
    let t1 = Triangle { side1: 3, side2: 4, side3: 5 }
    let r2 = Rectangle { width: 2, height: 5 }
    print(perimeter(r1))
    print(perimeter(t1))
    print(perimeter(r2))
}

fn perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn perimeter(t: Triangle) -> Int {
    t.side1 + t.side2 + t.side3
}
