fn main() {
    let r1 = Rectangle { width: 3, height: 4 };
    let t1 = Triangle { sides: [3, 4, 5] };
    let r2 = Rectangle { width: 2, height: 5 };

    println!("{}", r1.perimeter());
    println!("{}", t1.perimeter());
    println!("{}", r2.perimeter());
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn perimeter(&self) -> u32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    sides: [u32; 3],
}

impl Triangle {
    fn perimeter(&self) -> u32 {
        self.sides.iter().sum()
    }
}
