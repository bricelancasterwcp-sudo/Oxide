fn main() {
    let pen = Item { unit_price: 3, quantity: 4 }
    let pad = Item { unit_price: 9, quantity: 2 }
    let ink = Item { unit_price: 5, quantity: 3 }
    let total_cost = pen.cost() + pad.cost() + ink.cost()
    let costy_item = max(&pen, &pad, &ink, &compare_cost)
    print(total_cost)
    print(costy_item.name())
}

struct Item {
    unit_price: Int,
    quantity: Int,
}

fn compare_cost(item_a: &Item, item_b: &Item) -> Int {
    item_a.cost().cmp(&item_b)
}

fn max(item_a: &Item, item_b: &Item, item_c: &Item, compare: FnMutTwoArgs<Item, Item, Int>) -> Item {
    let mut max_item = item_a
    if compare(item_b, max_item) > 0 {
        max_item = item_b
    }
    if compare(item_c, max_item) > 0 {
        max_item = item_c
    }
    max_item.clone()
}

fn name(item: &Item) -> String {
    if item.unit_price == 3 {
        return "pen".to_string()
    } else if item.unit_price == 9 {
        return "pad".to_string()
    } else {
        return "ink".to_string()
    }
}

fn cost(item: &Item) -> Int {
    item.unit_price * item.quantity
}

    if x < y {
        -1

    let mut result = String()

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < v.len() {
        return Some(v.get(i).unwrap())
    }
    None
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a.min(b).to_int() + 1.to_int() - 1.to_int() + 1.to_int() {
        if i < b {
            result.push(i)
        }
    }
    result
}
