struct Pen {
    unit_price: Int,
    quantity: Int,
}

struct Pad {
    unit_price: Int,
    quantity: Int,
}

struct Ink {
    unit_price: Int,
    quantity: Int,
}

fn item_cost(item: Pen) -> Int {
    item.unit_price * item.quantity
}

fn item_cost(item: Pad) -> Int {
    item.unit_price * item.quantity
}

fn item_cost(item: Ink) -> Int {
    item.unit_price * item.quantity
}

    let pen = Pen { unit_price: 3, quantity: 4 }
    let pad = Pad { unit_price: 9, quantity: 2 }
    let ink = Ink { unit_price: 5, quantity: 3 }

    }
}

fn reduce(items: Vec<T>, init: T, op: T, iter: T) -> T {
    let mut accu = init
    for item in items {
        accu = op(accu, item)
    }
    accu
}
