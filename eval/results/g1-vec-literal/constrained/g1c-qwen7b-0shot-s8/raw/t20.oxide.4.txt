struct Item { name: Str, unit_price: Int, quantity: Int }

fn cost(item: Item) -> Int {
    item.unit_price * item.quantity
}

fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 }
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 }
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 }
    let total_cost = cost(pen.clone()) + cost(pad.clone()) + cost(ink.clone())
    print(total_cost)
    let mut max_cost = cost(pen)
    let mut max_item = pen.name.clone()
    if cost(pad) > max_cost {
        max_cost = cost(pad)
        max_item = pad.name.clone()
    }
    if cost(ink) > max_cost {
        max_cost = cost(ink)
        max_item = ink.name.clone()
    }
    print(max_item)
}
