use std::io;

fn is_perfect(number: i32) -> bool {
    let mut sum = 0;
    for i in 1..(number / 2 + 1) {
        if number % i == 0 {
            sum += i;
        }
    }
    sum == number
}

fn main() {
    for i in 1..=999 {
        if is_perfect(i) {
            println!("{}", i);
        }
    }
}
