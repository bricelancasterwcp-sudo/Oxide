fn main() {
    let mut sum = 0
    let mut count = 0
    for s in vec_str("12", "x", "30") {
        match parse_int(s) {
            Some(n) => {
                sum = sum + n
                count = count + 1
            },
            None => {
                print_str("failed to parse ")
                print(s)
                print_str("\n")
            },
        }
    }
    print(sum)
    print(" ")
    print(count)
}

fn parse_int(s: Str) -> Option<Int> {
    match int_to_str(trunc(parse_float(s)?)) {
        Int(n) => Some(n),
        _ => None,
    }
}

fn parse_float(s: Str) -> Option<Float> {
    match str_to_float(s) {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn str_to_float(s: Str) -> Result<Float, Str> {
    match float_to_str(int_to_float(parse_int(s)?)?) {
        Ok(n) => Ok(n),
        Err(_) => Err(s),
    }
}

fn int_to_float(i: Int) -> Result<Float, Str> {
    match float_to_int(i) {
        Ok(n) => Ok(n),

    }
