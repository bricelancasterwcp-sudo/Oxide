use std::fmt::Display;

fn get_perimeter(shape: &str) -> u32 {
    match shape {
        "rectangle" => 2 * (3 + 4),
        "triangle" => 3 * 5,
        "rectangle_2" => 2 * (2 + 5),
        _ => 0,
    }
}

fn main() {
    let shapes = ["rectangle", "triangle", "rectangle_2"];

    for shape in shapes {
        let perimeter = get_perimeter(shape);
        println!("{}", perimeter);
    }
}
