struct Item { name: Str, price: Int, quantity: Int }

fn calculate_cost(items: Vec<Item>) -> Int {
    let mut cost = 0
    for item in items {
        cost = add(cost, mul(item.price, item.quantity))
    }
    cost
}

fn find_most_expensive(items: Vec<Item>) -> Str {
    let mut max = 0
    let mut best = ""
    for item in items {
        let cost = mul(item.price, item.quantity)
        if cost > max {
            max = cost
            best = item.name
        }
    }
    best
}

fn main() {
    let items = vec(Item { name: "pen", price: 3, quantity: 4 }, Item { name: "pad", price: 9, quantity: 2 }, Item { name: "ink", price: 5, quantity: 3 })
    print(calculate_cost(items))
    print(find_most_expensive(items))
}

fn add(a: Int, b: Int) -> Int {
    a + b
}

fn mul(a: Int, b: Int) -> Int {
    a * b
}

struct Point { x: Int, y: Int }

fn distance(a: Point, b: Point) -> Float {
    let dx = sub(a.x, b.x)
    let dy = sub(a.y, b.y)
    sqrt(add(mul(dx, dx), mul(dy, dy)))
}

fn distance_squared(a: Point, b: Point) -> Float {
    let dx = sub(a.x, b.x)
    let dy = sub(a.y, b.y)
    add(mul(dx, dx), mul(dy, dy))
}

fn main() {
    let a = Point { x: 3, y: 4 }
    let b = Point { x: 10, y: 12 }
    let dist = distance(a, b)
    let dist_squared = distance_squared(a, b)
    print_str("The distance is ")
    print(dist)
    print_str(", and the squared distance is ")
    print(dist_squared)
}

fn sub(a: Int, b: Int) -> Int {
    a - b
}

fn sqrt(x: Float) -> Float {
    x.sqrt()
}

struct Circle { center: Point, radius: Int }

fn area(c: Circle) -> Float {
    let r = c.radius
    let r2 = mul(r, r)
    mul(3.14, mul(r2, r))
}

fn main() {
    let c = Circle { center: Point { x: 0, y: 0 }, radius: 4 }
    print(area(c))
}
