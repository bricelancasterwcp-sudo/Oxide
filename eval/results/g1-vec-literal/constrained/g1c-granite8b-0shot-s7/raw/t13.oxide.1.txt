fn is_palindrome(s: Str) -> Bool {
    let len = str_len(s)
    for i in range(0, len / 2) {
        let j = len - 1 - i
        if get(chars(s), i)? != get(chars(s), j)? {
            return false
        }
    }
    true
}

fn get(v: Vec<T>, i: Int) -> Option<T> {
    match i < len(v)? {
        Some(true_val) => Some(clone(get(v, i)?)),
        Some(false_val) => None,
        _ => None,
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a < b? {
        v = push(v, i)
    }
    v
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    for i in range(0, str_len(s)) {
        v = push(v, str_sub(s, i, i + 1))
    }
    v
}

fn str_sub(s: Str, a: Int, b: Int) -> Str {
    let sub = clone(s)
    let start = trunc(to_float(a)?)
    let end = trunc(to_float(b)?)
    let len = end - start
    let chars = get(chars(sub), start)?
    concat(chars, get(chars(sub), end)?)
}

fn print(x: T) {
    match x {

fn print_str(s: Str) {
    let len = str_len(s)
    for i in range(0, len) {
        let c = get(chars(s), i)?
        print_char(c)
    }
}

fn print_char(c: Str) {
    match c {
        Str(refinner) => print_str(refinner.data),
        _ => panic("print_char() not implemented for this type"),
    }
}

fn print_int(n: Int) {
    let mut v = vec()
    let mut n = n
    while n > 0 {
        let digit = (n % 10)
        v = push(v, int_to_str(digit))
        n = trunc(to_float(n / 10)?)
    }
    print_str(concat(v, Str("")))
}
