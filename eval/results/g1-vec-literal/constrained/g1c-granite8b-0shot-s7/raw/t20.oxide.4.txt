struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Option<Int> {
    let mut found = None
    for x in v {
        if x > 10 {
            found = Some(x)
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    match first_big(r.values) {
        Some(n) => print(n),
        None => print_str("no big numbers"),
    }
    match second(r.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

struct Pen { price: Int, quantity: Int }

struct Pad { price: Int, quantity: Int }

struct Ink { price: Int, quantity: Int }

fn total_cost(pen: Pen, pad: Pad, ink: Ink) -> Int {
    let cost_of_pen: Int = pen.price * pen.quantity
    let cost_of_pad: Int = pad.price * pad.quantity
    let cost_of_ink: Int = ink.price * ink.quantity
    cost_of_pen + cost_of_pad + cost_of_ink
}

fn most_expensive(pen: Pen, pad: Pad, ink: Ink) -> Str {
    let total_cost_pen: Int = total_cost(pen, pad, ink)
    let total_cost_pad: Int = total_cost(pad, pen, ink)
    let total_cost_ink: Int = total_cost(ink, pen, pad)
    if (total_cost_pen > total_cost_pad) && (total_cost_pen > total_cost_ink) {
        return "Pen"
    } else if (total_cost_pad > total_cost_pen) && (total_cost_pad > total_cost_ink) {
        return "Pad"
    } else {
        return "Ink"
    }
}

fn main() {
    let pen = Pen { price: 3, quantity: 4 }
    let pad = Pad { price: 9, quantity: 2 }
    let ink = Ink { price: 5, quantity: 3 }
    let total = total_cost(pen, pad, ink)
    print(total)
    let most_expensive_item = most_expensive(pen, pad, ink)
    print(most_expensive_item)
}
