struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Option<Int> {
    let mut found = None
    for x in v {
        if x > 10 {
            found = Some(x)
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    match first_big(r.values) {
        Some(n) => print(n),
        None => print_str("no big numbers"),
    }
    match second(r.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}
