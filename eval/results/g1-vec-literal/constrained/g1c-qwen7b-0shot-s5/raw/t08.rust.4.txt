fn main() {
    let numbers = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut largest = i32::MIN;

    for &num in &numbers {
        if num > 0 {
            positive_count += 1;
        }
        if num > largest {
            largest = num;
        }
    }

    println!("Positive values count: {}", positive_count);
    println!("Largest value: {}", largest);
}
