fn main() {
    let pen_cost = (1.0 * 3.0 * 4.0).floor() / 1.0 * 100.0 / 100.0
    let pad_cost = (9.0 * 2.0).floor() / 1.0 * 100.0 / 100.0
    let ink_cost = (5.0 * 3.0).floor() / 1.0 * 100.0 / 100.0
    print(pen_cost + pad_cost + ink_cost)
    let items = vec(("pen".to_string() + ": " + &pen_cost), ("pad".to_string() + ": " + &pad_cost), ("ink".to_string() + ": " + &ink_cost))
    let mut max_cost = 0.0
    let mut max_item = "".to_string()
    for item in &items {
        let cost_parts: Vec<String> = item.split(": ").collect()
        let cost_float = cost_parts.get(1).unwrap().parse().unwrap()
        if cost_float > max_cost {
            max_cost = cost_float
            max_item = cost_parts.get(0).unwrap().to_string()
        }
    }
    print(max_item)
    drop pen_cost
    drop pad_cost
    drop ink_cost
    drop items
}
