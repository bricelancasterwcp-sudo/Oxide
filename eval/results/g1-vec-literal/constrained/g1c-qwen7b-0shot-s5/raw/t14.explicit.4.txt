fn main() {
    let text = "mississippi"
    let s_count = text.chars().filter(char_eq("s")).count()
    let distinct_count = text.chars().collect().len()
    print(s_count)
    print(distinct_count)
}

fn char_eq(c: Str) -> Str {
    move || c == "s"
}

fn count(f: Str) -> Int {
    let mut count = 0
    for _ in 0.range(count) {
        if f() {
            count = count + 1
        }
    }
    count
}

fn collect(f: Str) -> Vec<Str> {
    let mut result = Vec()
    for i in 0.range(26) {
        if f(to_char(i)) {
            result = push(result, to_char(i))
        }
    }
    result
}

fn to_char(n: Int) -> Str {
    from_int(n + 97)
}

fn from_int(n: Int) -> Str {
    ((n + 48) % 26).to_int().to_str()
}

fn print(x: Int) {
    print_str(x.to_str())
}

fn print_str(s: Str) {
    let mut chars = chars(s)
    while chars.len() > 0 {
        print_char(chars.get(0))
        chars = chars.iter().skip(1).collect()
    }
}

fn print_char(c: Str) {
    print_str(c)
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = Vec()
    let mut i = 0
    while i < str_len(s) {
        result = push(result, get(s, i))
        i = i + 1
    }
    result
}

fn get(s: Str, i: Int) -> Str {
    &s + i
}

fn str_len(s: Str) -> Int {
    len(s)
}

fn len(v: Str) -> Int {
    v.len()
}

fn push(v: Vec<Str>, x: Str) -> Vec<Str> {
    concat(v, x)
}

fn concat(a: Vec<Str>, b: Str) -> Vec<Str> {
    let mut result = a
    let mut chars = chars(b)
    while chars.len() > 0 {
        result = push(result, chars.get(0))
        chars = chars.iter().skip(1).collect()
    }
    result
}
