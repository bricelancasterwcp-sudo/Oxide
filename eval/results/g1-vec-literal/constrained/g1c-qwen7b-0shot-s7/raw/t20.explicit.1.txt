fn main() {
    let pen = (3.0 * 4.0).to_int()
    let pad = (9.0 * 2.0).to_int()
    let ink = (5.0 * 3.0).to_int()
    let total_cost = pen + pad + ink
    print(total_cost)
    if pen > pad {
        if pen > ink {
            print_str("pen")
        } else {
            print_str("ink")
        }
    } else {
        if pad > ink {
            print_str("pad")
        } else {
            print_str("ink")
        }
    }
}
