fn factors(n: Int) -> Vec<Int> {
    let mut fs = vec()
    for i in range(2, trunc(n / 2) + 1) {
        if n % i == 0 {
            fs.push(i)
        }
    }
    fs
}

fn is_perfect(n: Int) -> Bool {
    n == sum(factors(n))
}

fn main() {
    for i in range(1, 1000) {
        if is_perfect(i) {
            print(i)
            print_str("\n")
        }
    }
}
