fn keep_large(v: &Vec<Int>) -> Vec<Int> {
    let mut kept = vec()
    for x in v {
        if x > &mut && 3 {
            kept = push(kept, x)
        }
    }
    kept
}

fn main() {
    let values = vec(4, 1, 7, 3, 9, 2)
    let kept = keep_large(&values)
    for x in &kept {
        print(x)
    }
    print(len(&kept))
}
