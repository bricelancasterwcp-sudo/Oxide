fn len_square(s: &Shape) -> Int {
    match s {
        Square(n) => 4 * n,
        _ => 0,
    }
}

fn len_rectangle(s: &Shape) -> Int {
    match s {
        Rectangle(l, w) => 2 * (l + w),
        _ => 0,
    }
}

fn len_triangle(s: &Shape) -> Int {
    match s {
        Triangle(a, b, c) => a + b + c,
        _ => 0,
    }
}

fn main() {
    let shapes = vec(Square(3), Triangle(3, 4, 5), Rectangle(2, 5))
    for s in shapes {
        print(len_square(&s))
        print(len_rectangle(&s))
        print(len_triangle(&s))
    }
}

enum Shape {
    Square(Int),
    Triangle(Int, Int, Int),
    Rectangle(Int, Int),
}

fn push(v: Vec<Shape>, x: Shape) -> Vec<Shape> {
    let mut v = v
    v.push(x)
    v
}

fn vec(s: Shape) -> Vec<Shape> {
    let mut v = vec()
    v.push(s)
    v
}

fn print(x: Int) {
    println(x)
}

fn len(v: &Vec<Shape>) -> Int {
    v.len()
}

fn get(v: &Vec<Shape>, i: Int) -> Option<Shape> {
    if i >= len(v) {
        None
    } else {
        Some(v.get(i))
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a + 1 < b {
        v.push(i)
    }
    v
}

fn clone(x: &Shape) -> Shape {
    match x {
        Square(n) => Square(n),
        Triangle(a, b, c) => Triangle(a, b, c),
        Rectangle(l, w) => Rectangle(l, w),
    }
}

fn str_len(s: &Str) -> Int {
    s.len()
}

fn concat(a: Str, b: Str) -> Str {
    a + b
}

fn chars(s: &Str) -> Vec<Str> {
    let mut v = vec()
    for c in s.chars() {
        v.push(c.to_string())
    }
    v
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn parse_int(s: &Str) -> Option<Int> {
    match s.parse() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn to_float(n: Int) -> Float {
    n.to_float()
}

fn trunc(x: Float) -> Int {
    x.trunc()
}
