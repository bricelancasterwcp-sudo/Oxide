fn factors(n: Int) -> Vec<Int> {
    let mut x = 2
    let mut result = vec()
    while x * x <= n {
        if n % x == 0 {
            result = push(result, x)
            if x != n / x {
                result = push(result, n / x)
            }
        }
        x = x + 1
    }
    result
}

fn is_perfect(n: Int) -> Bool {
    if n == 1 {
        return False
    }
    let f = factors(n)
    let sum = f.iter().fold(0, add)
    return sum == n
}

fn main() {
    let mut x = 2
    while x < 1000 {
        if is_perfect(x) {
            print(x)
            print("\n")
        }
        x = x + 1
    }
}
