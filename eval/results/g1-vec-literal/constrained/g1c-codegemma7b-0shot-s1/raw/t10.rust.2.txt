use std::collections::LinkedList;

fn main() {
    let mut values = LinkedList::new();
    values.push_back(4);
    values.push_back(1);
    values.push_back(7);
    values.push_back(3);
    values.push_back(9);
    values.push_back(2);

    let mut kept_count = 0;

    for value in values {
        if value > 3 {
            println!("{}", value);
            kept_count += 1;
        }
    }

    println!("{} values kept", kept_count);
}
