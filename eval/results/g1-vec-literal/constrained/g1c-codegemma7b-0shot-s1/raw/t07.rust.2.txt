use std::io::Write;

fn main() {
    let mut x = 0;
    let mut multiples_of_three = Vec::new();

    loop {
        x += 1;
        if x % 3 == 0 {
            multiples_of_three.push(x);
        }
        if multiples_of_three.len() == 8 {
            break;
        }
    }

    let mut sum = 0;
    for multiple in multiples_of_three {
        sum += multiple;
    }

    print!("The sum of the first 8 positive multiples of 3 is: ");
    print!("{}", sum);
}
