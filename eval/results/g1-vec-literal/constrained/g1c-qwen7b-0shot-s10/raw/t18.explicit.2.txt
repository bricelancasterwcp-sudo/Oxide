fn perimeter(rect: Rect, triangle: Tri, rect2: Rect) -> Vec<Int> {
    let mut perimeters = vec()
    perimeters.push(rect.perimeter())
    perimeters.push(triangle.perimeter())
    perimeters.push(rect2.perimeter())
    perimeters
}

fn main() {
    let rect = Rect { width: 3, height: 4 }
    let triangle = Tri { a: 3, b: 4, c: 5 }
    let rect2 = Rect { width: 2, height: 5 }
    let perimeters = perimeter(rect, triangle, rect2)
    for p in &perimeters {
        print(p)
    }
    drop perimeters
}

struct Rect {
    width: Int,
    height: Int,
}

struct Tri {
    a: Int,
    b: Int,
    c: Int,
}

fn perimeter_rect(rect: Rect) -> Int {
    2 * (rect.width + rect.height)
}

fn perimeter_tri(tri: Tri) -> Int {
    tri.a + tri.b + tri.c
}

fn main() {
    let rect = Rect { width: 3, height: 4 }
    let triangle = Tri { a: 3, b: 4, c: 5 }
    let rect2 = Rect { width: 2, height: 5 }
    let perimeters = perimeter(rect, triangle, rect2)
    for p in &perimeters {
        print(p)
    }
    drop perimeters
}
