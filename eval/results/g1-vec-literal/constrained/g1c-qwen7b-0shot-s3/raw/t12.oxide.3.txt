struct Reading { label: Str, values: Vec<Int> }

fn reverse(s: Str) -> Str {
    let chars = chars(s)
    let mut reversed = vec()
    for ch in chars {
        reversed = push(reversed, ch)
    }
    concat("", reversed)
}
