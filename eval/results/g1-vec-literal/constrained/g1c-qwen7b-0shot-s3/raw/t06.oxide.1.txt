struct PerfectNumber { value: Int }

fn is_perfect(n: Int) -> Bool {
    let mut sum = 0
    for i in range(1, n) {
        if n % i == 0 {
            sum = sum + i
        }
    }
    sum == n
}

fn main() {
    for i in range(1, 1000) {
        if is_perfect(i) {
            print(i)
        }
    }
}
