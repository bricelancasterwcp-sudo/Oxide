fn count_and_max(v: Vec<Int>) -> Tuple<Int, Int> {
    let mut count = 0
    let mut max = -1000000000
    for x in v {
        if x > 0 {
            count = count + 1
        }
        if x > max {
            max = x
        }
    }
    Tuple(count, max)
}

fn main() {
    let v = push(push(push(push(vec(), 3), 8), -2), 12)
    v = push(v, 7)
    let result = count_and_max(v)
    print(result.ctor0)
    print(result.ctor1)
}
