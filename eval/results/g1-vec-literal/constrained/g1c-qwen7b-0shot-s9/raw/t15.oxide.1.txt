fn main() {
    let mut sum = 0
    let mut count = 0
    for s in vec("12", "x", "30") {
        match parse_int(s) {
            Some(n) => {
                sum = sum + n
                count = count + 1
            },
            None => count == count + 1,
        }
    }
    print(sum)
    print_str(" ")
    print(count)
}
