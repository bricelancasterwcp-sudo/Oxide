fn count_s(text: &Str) -> Int {
    let mut count = 0
    for c in chars(text) {
        if c == "s" {
            count = count + 1
        }
    }
    count
}

fn distinct_chars(text: &Str) -> Int {
    let mut chars = chars(text)
    let mut seen = vec()
    let mut count = 0
    while letSome(c) == chars.next() {
        if !contains(seen, c) {
            seen = push(seen, c)
            count = count + 1
        }
    }
    count
}

fn main() {
    let text = "mississippi"
    print(count_s(&text))
    drop text
    print(distinct_chars(&text))
    drop text
}
