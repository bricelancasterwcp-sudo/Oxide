```rust
use std::io::Write;

fn main() {
    let mut total_cost = 0;
    let mut max_cost = 0;
    let mut most_expensive_item = "";

    let items = [
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];

    for (name, unit_price, quantity) in items {
        let cost = unit_price * quantity;
        total_cost += cost;

        if cost > max_cost {
            max_cost = cost;
            most_expensive_item = name;
        }
    }

    println!("Total cost: {}", total_cost);
    println!("Most expensive item: {}", most_expensive_item);
}
```
