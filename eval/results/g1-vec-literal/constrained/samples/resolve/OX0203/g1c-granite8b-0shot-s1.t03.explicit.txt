fn main() {
    print(gcd(252, 105))
}

fn gcd(x: Int, y: Int) -> Int {
    let mut x = x
    let mut y = y
    while y != 0 {
        let z = y
        y = x % y
        x = z
    }
    x
}

fn divides(x: Int, y: Int) -> Bool {
    x % y == 0
}

fn is_prime(x: Int) -> Bool {
    if x < 2 {
        return false
    }
    for i in 2.to(x / 2) {
        if divides(x, i) {
            return false
        }
    }
    true
}

fn main() {
    let limit = 100
    let mut total = 0
    for i in 2.to(limit) {
        if is_prime(i) {
            total = total + i
        }
    }
    print(total)
}
