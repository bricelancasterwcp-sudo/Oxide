use std::io::{ErrorKind, Result};

fn main() -> Result<()> {
    let mut sum = 0;
    let mut errors = 0;

    for line in ["12", "x", "30"] {
        match line.parse::<i32>() {
            Ok(num) => sum += num,
            Err(e) if e.kind() == ErrorKind::InvalidInput => errors += 1,
            Err(_) => unreachable!(),
        }
    }

    println!("Sum: {}", sum);
    println!("Errors: {}", errors);

    Ok(())
}
