fn main() {
    let nums = vec(3, 8, -2, 12, 7)
    let mut positive = 0
    let mut largest = nums.first().unwrap()
    for num in nums {
        if num > 0 {
            positive = positive + 1
        }
        if num > largest {
            largest = num
        }
    }
    print(positive)
    print(largest)
    drop nums
}
