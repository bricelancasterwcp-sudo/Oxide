fn main() {
    let str = "mississippi".to_string()
    let mut count = 0
    let mut distinct = 0
    for c in str.chars() {
        if c == "s" {
            count = count + 1
        }
        if !distinct.contains(&c) {
            distinct = distinct + 1
        }
    }
    print(count)
    print(distinct)
}

fn contains(s: &Str, c: &Char) -> Bool {
    for ch in s.chars() {
        if ch == c {
            return true
        }
    }
    false
}

    chars: Vec<Char>,
}

    Str(s.clone())
}

fn len(s: &Str) -> Int {
    s.chars.len()
}

fn chars(s: &Str) -> Vec<Char> {
    s.chars.clone()
}
