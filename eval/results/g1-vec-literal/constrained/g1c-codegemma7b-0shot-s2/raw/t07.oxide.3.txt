fn first_multiples(n: Int) -> Vec<Int> {
    let mut v = vec()
    for i in 1.to(n) {
        if i % 3 == 0 {
            v = push(v, i)
        }
    }
    v
}

fn sum(v: Vec<Int>) -> Int {
    let mut acc = 0
    for x in v {
        acc = acc + x
    }
    acc
}

fn main() {
    let v = first_multiples(24)
    print(sum(v))
}
