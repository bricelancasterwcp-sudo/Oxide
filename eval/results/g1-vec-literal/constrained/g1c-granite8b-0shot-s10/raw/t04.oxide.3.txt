struct Prime { next: Option<Prime>, value: Int }

fn is_prime(n: Int) -> Bool {
    if n == 2 {
        return true
    }
    if n <= 1 || n % 2 == 0 {
        return false
    }
    let mut i = 3
    while i * i <= n {
        if n % i == 0 {
            return false
        }
        i = i + 2
    }
    true
}

fn count_primes(n: Int) -> Int {
    let mut count = 0
    let mut p = Prime { next: None, value: 2 }
    while p.value < n {
        let next = Prime { next: None, value: p.value + 2 }
        p.next == Some(next)
        if is_prime(p.value) {
            count = count + 1
        }
        p = p.next?
    }
    count
}

fn main() {
    print(count_primes(100))
}
