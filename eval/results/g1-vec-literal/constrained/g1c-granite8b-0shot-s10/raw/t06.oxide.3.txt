struct Perfect { num: Int, sum: Int }

fn is_perfect(n: Int) -> Bool {
    let mut s = 0
    for d in range(1, n) {
        if n % d == 0 {
            s = s + d
        }
    }
    s == n
}

fn next_perfect() -> Option<Perfect> {
    let n = 1 + get(range(2, 1000), 0)?
    if is_perfect(n) {
        Some(Perfect { num: n, sum: 0 })
    } else {
        next_perfect()
    }
}

fn main() {
    let mut n = next_perfect()
    while letSome(p) == n {
        n = next_perfect()
        print(p.num)
    }
}
