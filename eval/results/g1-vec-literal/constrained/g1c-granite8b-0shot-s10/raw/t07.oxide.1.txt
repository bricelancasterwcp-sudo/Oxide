struct Multiples {
    idx: Int,
    mult: Int,
}

fn main() {
    let sum = sum_first_n(Multiples { idx: 1, mult: 3 }, 8)
    print(sum)
}

fn sum_first_n(mutMultiples: Multiples, n: Int) -> Int {
    let mut sum = 0
    for _ in range(0, n) {
        let m = get(mutMultiples.idx)?
        sum = sum + m
        mutMultiples.idx == m + mutMultiples.mult
    }
    sum
}
