fn get_total_cost(pen: &Item, pad: &Item, ink: &Item) -> Int {
    pen.price * pen.quantity + pad.price * pad.quantity + ink.price * ink.quantity
}

fn get_item_with_highest_cost(pen: &Item, pad: &Item, ink: &Item) -> Item {
    let mut highest_cost = 0
    let mut result = Item { price: 0, quantity: 0 }
    if pen.price * pen.quantity > highest_cost {
        highest_cost = pen.price * pen.quantity
        result = pen.clone()
    }
    if pad.price * pad.quantity > highest_cost {
        highest_cost = pad.price * pad.quantity
        result = pad.clone()
    }
    if ink.price * ink.quantity > highest_cost {
        highest_cost = ink.price * ink.quantity
        result = ink.clone()
    }
    result
}
