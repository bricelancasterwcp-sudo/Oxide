struct Prime { n: Int }

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return False
    }
    for d in range(2, n) {
        if n % d == 0 {
            return False
        }
    }
    True
}

fn count_primes_strictly_less_than(n: Int) -> Int {
    let mut count = 0
    for x in range(0, n) {
        if is_prime(x) {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count_primes_strictly_less_than(100))
}
