    for c in chars(s) {
        print(c)
    }
}

fn int_to_str(n: Int) -> Str {
    let mut digits = vec()
    let mut n = n
    while n > 0 {
        digits.push(int_to_char(n % 10))
        n = trunc(n / 10)
    }
    let mut s = str()
    for d in reverse(&digits) {
        s = concat(s, d)
    }
    if s == str() {
        s = "0"
    }
    s
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    for i in 0.to_int() {
        if s.len() <= i {
            break
        }
        v.push(str(s.chars().nth(i).unwrap()))
    }
    v
}

fn reverse(v: &Vec<Int>) -> Vec<Int> {
    let mut v = v.to_vec()
    let mut a = 0.to_int()
    let mut b = v.len().to_int() - 1.to_int()
    while a < b {
        let tmp = v.get(a).unwrap()
        v.set(a, v.get(b).unwrap())
        v.set(b, tmp)
        a = a + 1.to_int()
        b = b - 1.to_int()
    }
    v
}

fn int_to_char(n: Int) -> Str {
    if n == 0 {
        return "0"
    }
    let mut n = n
    let mut digits = vec()
    while n > 0 {
        digits.push(if n % 10 == 0 { 10 } else { n % 10 })
        n = trunc(n / 10)
    }
    let mut s = str()
    for d in reverse(&digits) {
        s = concat(s, int_to_char(d))
    }
    s
}
