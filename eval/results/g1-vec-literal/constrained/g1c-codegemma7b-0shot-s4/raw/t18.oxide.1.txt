fn area(s: Shape) -> Float {
    match s {
        Shape(e) => match e {
            ShapeType(s) => match s {
                ShapeSubType(a, b) => a * b,
                ShapeSubType(a, b, c) => 0.5 * a * b * c,
            },
            _ => panic("Invalid shape"),
        },
    }
}

fn perimeter(s: Shape) -> Float {
    match s {
        Shape(e) => match e {
            ShapeType(s) => match s {
                ShapeSubType(a, b) => 2.0 * (a + b),
                ShapeSubType(a, b, c) => a + b + c,
            },
            _ => panic("Invalid shape"),
        },
    }
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    let mut i = 0
    while i < str_len(s) {
        let c = get(s, i)
        if c != None {
            v.append(c)
            i = i + 1
        }
    }
    v
}

fn int_to_str(n: Int) -> Str {
    let mut v = vec()
    let mut c = n
    while c != 0 {
        let r = c % 10
        let s = int_to_char(r)
        v.append(s)
        c = trunc(c / 10)
    }
    let mut s = str()
    for i in range(len(v) - 1, -1, -1) {
        s = concat(s, get(v, i))
    }
    s
}

fn parse_int(s: Str) -> Option<Int> {
    if str_len(s) == 0 {
        return None
    }
    let mut c = s
    let mut n = 0
    while len(c) > 0 {
        let d = get(c, 0)
        if d != None {
            let x = char_to_int(d)
            n = n * 10 + x
            c = get(c, 1)
        } else {
            return None
        }
    }
    Some(n)
}
