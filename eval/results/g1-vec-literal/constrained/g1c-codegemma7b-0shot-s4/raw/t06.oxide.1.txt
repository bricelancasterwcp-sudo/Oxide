fn is_perfect(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    let mut sum = 0
    for i in 2.to_int() / 2.to_int() + 1.to_int() {
        if n % i == 0 {
            sum = sum + i
        }
    }
    return sum == n
}

fn main() {
    for i in 0.to_int() + 1.to_int() {
        if is_perfect(i) {
            print(i)
            print_str("\n")
        }
    }
}
