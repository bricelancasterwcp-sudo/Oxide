fn is_prime(n: Int) -> Bool {
    let upper_bound = trunc(sqrt(to_float(n)))
    for x in range(2, upper_bound + 1) {
        if n % x == 0 {
            return False
        }
    }
    True
}

fn sum_divisors(n: Int) -> Int {
    let mut result = 0
    for x in range(1, n) {
        if n % x == 0 {
            result = result + x
        }
    }
    result
}

fn perfect_numbers(n: Int) -> Vec<Int> {
    let mut result = vec()
    for m in range(1, n) {
        if sum_divisors(m) == m {
            result = push(result, m)
        }
    }
    result
}

fn main() {
    let p = perfect_numbers(1000)
    for n in p {
        print(n)
    }
}

fn sqrt(x: Float) -> Float {
    x.sqrt()
}

fn main() {
    let sum = accumulate(vec(), perfect_numbers(1000))
    print(sum)
}

fn accumulate(v: Vec<Int>, xs: Vec<Int>) -> Vec<Int> {
    for x in xs {
        v = push(v, x)
    }
    v
}

fn main() {
    let p = perfect_numbers(1000)
    let sum = accumulate(vec(), p)
    print(sum)
}
