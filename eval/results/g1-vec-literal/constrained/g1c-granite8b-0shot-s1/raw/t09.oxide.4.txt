fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v = v
    push(v, x)
}

fn pop(s: Vec<Int>) -> Option<Int> {
    match s {
        Vec(_) => None,
        Stack(_, rest) => Some(rest),
    }
}

fn main() {
    let mut v = vec()
    v = push(v, 1)
    v = push(v, 2)
    match pop(v) {
        Some(x) => print(x),
        None => print_str("empty"),
    }
}
