struct Stack { value: Int, rest: Vec<Int> }

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn pop(s: Vec<Int>) -> Option<Int> {
    match s {
        Stack(_, rest) => Some(rest),
        Vec(_) => None,
    }
}

fn main() {
    let mut v = vec()
    v = push(v, 1)
    v = push(v, 2)
    match pop(v) {
        Some(x) => print(x),
        None => print_str("empty"),
    }
}
