fn main() {
    let string = "mississippi".to_string().chars().collect().clone()
    let s_count = count_char(string, "s".to_string().chars().collect().clone())
    let d_count = distinct_chars(string)
    print(s_count)
    print(d_count)
}

fn count_char(string: Vec<Char>, char: Char) -> Int {
    let mut count = 0
    for c in string {
        if c == char {
            count = count + 1
        }
    }
    count
}

fn distinct_chars(string: Vec<Char>) -> Int {
    let mut distinct_chars = vec()
    for c in string {
        let found = get(distinct_chars, c)?
        if found == false {
            distinct_chars = push(distinct_chars, true)
        }
    }
    len(distinct_chars)
}

fn get(vector: Vec<T>, index: Int) -> Option<T> {
    if index < 0 || index >= len(vector) {
        return None
    } else {
        let mut i = 0
        let mut found = None
        while i < len(vector) {
            let item = get(vector, i)?
            if i == index {
                found = Some(item)
                break
            }
            i = i + 1
        }
        return found
    }
}

fn push(vector: Vec<T>, item: T) -> Vec<T> {
    let mut result = vec()
    for x in vector {
        result = push(result, x)
    }
    result = push(result, item)
    return result
}

    return vec()
}

    return len(vector)
}
