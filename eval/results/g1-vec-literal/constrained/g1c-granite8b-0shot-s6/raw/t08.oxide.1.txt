fn filter(v: Vec<Int>, f: Int) -> Vec<Int> {
    let mut out = vec()
    for x in v {
        if f(x) {
            out = push(out, x)
        }
    }
    out
}

fn is_positive(n: Int) -> Bool {
    n > 0
}

fn reduce(v: Vec<Int>, f: Int) -> Int {
    let mut acc = get(v, 0)?
    for x in slice(v, 1, None) {
        acc = f(acc, x)
    }
    acc
}

fn max(a: Int, b: Int) -> Int {
    if a > b {
        a
    } else {
        b
    }
}

fn slice(v: Vec<Int>, start: Int, end: Option<Int>) -> Vec<Int> {
    let mut out = vec()
    for i in range(start, end?) {
        out = push(out, get(v, i)?)
    }
    out
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= len(v) {
        None
    } else {
        Some(v.get(i))
    }
}

fn print(n: Int) {
    let s = int_to_str(n)
    print_str(s)
}

fn len(v: Vec<Int>) -> Int {
    v.len()
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut out = v
    out.push(x)
    out
}

fn vec() -> Vec<Int> {
    Vec()
}

struct Option {
    value: Int,
}

struct Some {
    value: Int,
}

struct None {
    value: Int,

fn clone(x: Int) -> Int {
    x
}

fn parse_int(s: Str) -> Option<Int> {
    s.parse().ok()
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn print_str(s: Str) {
    println(s)
}
