fn sum_of_divisors(x: Int) -> Int {
    let mut total = 0
    for d in divisors(x) {
        total = add(total, d)
    }
    total
}

fn divisors(x: Int) -> Vec<Int> {
    let mut divs = vec()
    let mut d = 1
    while d <= trunc(sqrt(to_float(x))) {
        if x % d == 0 {
            divs = push(divs, d)
            if d * d != x {
                divs = push(divs, x / d)
            }
        }
        d = add(d, 1)
    }
    divs
}

fn sqrt(x: Float) -> Float {
    let mut low = 0.0
    let mut high = x
    while low < high {
        let mid = (low + high) / 2.0
        if mid * mid <= x {
            low = mid
        } else {
            high = mid
        }
    }
    low
}
