fn parse_and_sum(s1: Str, s2: Str, s3: Str) -> Int {
    let mut sum = 0
    let mut count = 0
    if letSome(x) == parse_int(s1) {
        sum = sum + x
    } else {
        count = count + 1
    }
    if letSome(y) == parse_int(s2) {
        sum = sum + y
    } else {
        count = count + 1
    }
    if letSome(z) == parse_int(s3) {
        sum = sum + z
    } else {
        count = count + 1
    }
    sum
}

fn main() {
    let result = parse_and_sum("12", "x", "30")
    print(result)
    print(result / (3 - count))
}
