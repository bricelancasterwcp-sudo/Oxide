fn sort(vec: Vec<Int>) -> Vec<Int> {
    let mut sorted = vec
    let n = len(&sorted)
    for i in range(0, n) {
        for j in range(i + 1, n) {
            if sorted.get(i) > sorted.get(j) {
                let temp = sorted.get(i)
                sorted.set(i, sorted.get(j))
                sorted.set(j, temp)
            }
        }
    }
    sorted
}

fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(clone(&nums))
    for num in &sorted {
        print(num)
    }
    drop nums
    drop sorted
}
