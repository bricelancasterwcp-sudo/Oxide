fn count(s: &Str, c: Char) -> Int {
    let mut count = 0
    for ch in chars(&s) {
        if ch == c {
            count = count + 1
        }
    }
    count
}

fn distinct(s: &Str) -> Int {
    let mut found = vec()
    for ch in chars(&s) {
        if !contains(&found, ch) {
            found = push(found, ch)
        }
    }
    len(&found)
}

fn contains(v: &Vec<Char>, x: Char) -> Bool {
    for y in &v {
        if y == x {
            return true
        }
    }
    false
}

fn main() {
    let s = "mississippi"
    print(count(&s, "s"))
    drop s
    print(distinct(&s))
}
