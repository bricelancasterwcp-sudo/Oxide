fn push(v: Vec<T>, x: T) -> Vec<T> {
    let new_len = len(v) + 1
    let mut v2 = vec()
    for i in range(0, new_len) {
        let j = i - 1
        if i == 0 {
            v2 = push(v2, x)
        } else {
            v2 = push(v2, get(v, j)?)
        }
    }
    v2
}
