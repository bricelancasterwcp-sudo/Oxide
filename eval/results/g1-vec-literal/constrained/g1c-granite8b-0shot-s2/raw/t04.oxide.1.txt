fn is_prime(n: Int) -> Bool {
    let mut i = 2
    while i * i <= n {
        if n % i == 0 {
            return false
        }
        i = i + 1
    }
    true
}

fn main() {
    let mut count = 0
    for i in range(2, 100) {
        if is_prime(i) {
            count = count + 1
        }
    }
    print(count)
}
