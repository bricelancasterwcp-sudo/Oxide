fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut v = v
    for i in range(0, len(v)) {
        for j in range(i + 1, len(v)) {
            if get(v, j)? < get(v, i)? {
                swap(v, i, j)
            }
        }
    }
    v
}

    let v = push(push(push(push(vec(), 5), 3), 8), 1)
    print(len(sort(push(v, 9), 2)))
}
