fn main() {
    let mut numbers = vec(5, 3, 8, 1, 9, 2)
    let mut i = 0
    let mut j = 0
    let mut tmp = 0
    while i < 6 {
        if j < 5 {
            if numbers.get(j) > &numbers {
                tmp = numbers.get(j)
                numbers.set(j, numbers.get(j + 1))
                numbers.set(j + 1, tmp)
            }
            j = j + 1
        } else {
            print(numbers.get(i))
            i = i + 1
        }
    }
}
