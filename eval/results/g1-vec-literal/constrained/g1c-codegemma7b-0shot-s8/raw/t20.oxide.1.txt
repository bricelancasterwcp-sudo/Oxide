fn print(x: Int) {
    print_str(int_to_str(x))
}

fn print_str(s: Str) {
    print(str_len(s))
    for c in chars(s) {
        print(ord(c))
    }
}

fn print_char(c: Char) {
    print(ord(c))
}

fn int_to_str(n: Int) -> Str {
    if n == 0 {
        return "0"
    }
    let mut digits = vec()
    let mut nn = n
    while nn != 0 {
        digits.push(nn % 10)
        nn = trunc(nn / 10)
    }
    let mut s = vec()
    for d in digits {
        s.push(char(d + ord("0")))
    }
    Str(s)
}

fn str_len(s: Str) -> Int {
    let mut count = 0
    for _ in chars(s) {
        count = count + 1
    }
    count
}

fn chars(s: Str) -> Vec<Char> {
    let mut v = vec()
    let mut ss = s
    while len(ss) > 0 {
        let c = ss.chars().next().unwrap()
        v.push(c)
        ss = ss.slice(1, len(ss))
    }
    v
}

fn ord(c: Char) -> Int {
    c - "a" + 1
}

fn trunc(x: Float) -> Int {
    if x < 0.0 {
        return x.floor().trunc()
    }
    x.ceil().trunc()
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut vv = v
    vv.push(x)
    vv
}

fn len(v: Vec<Int>) -> Int {
    let mut count = 0
    for _ in v {
        count = count + 1
    }
    count
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(v) {
        return Some(v.get(i).unwrap())
    }
    None
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a + 1 < b {
        v.push(i)
    }
    v
}
