fn main() {
    let nums = push(push(push(push(vec(), 1), 2), 3), 4)
    let reversed = reverse(&nums)
    for x in reversed {
        print(x)
    }
    drop nums
    drop reversed
}

fn reverse(v: &Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    for x in &v {
        result = push(result, x)
    }
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new_v = v.clone()
    new_v.push(x)
    new_v
}
