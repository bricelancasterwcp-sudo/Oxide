fn main() {
    let mut numbers = vec(5, 3, 8, 1, 9, 2)
    numbers = sort(numbers)
    for x in numbers {
        print_str(x.to_string())
    }
}

fn sort(mutv: Vec<Int>) -> Vec<Int> {
    for i in 0 < mutv.len() {
        for j in 0 < mutv.len() - i - 1 {
            if mutv.get(j) > mutv.get(j + 1) {
                let temp = mutv.get(j)
                mutv.set(j, mutv.get(j + 1))
                mutv.set(j + 1, temp)
            }
        }
    }
    mutv
}

fn print_str(s: Str) {
    println(s)
}
