fn main() {
    let nums = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(&nums)
    for x in sorted {
        print(x)
    }
    drop sorted
    drop nums
}

    let mut copy = v.clone()
    copy.sort()
    copy
