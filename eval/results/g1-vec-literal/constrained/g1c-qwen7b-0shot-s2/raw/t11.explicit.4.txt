fn main() {
    let list = vec(5, 3, 8, 1, 9, 2)
    let sorted = sort(clone(&list))
    for x in &sorted {
        print_str(&x)
    }
    drop sorted
}

fn sort(mutv: Vec<Int>) -> Vec<Int> {
    bubble_sort(mutv)
}

fn bubble_sort(mutv: Vec<Int>) -> Vec<Int> {
    let n = len(mutv)
    for i in 0 < n {
        for j in 0 < n - i - 1 {
            if v(j) > v(j + 1) {
                let temp = v(j)
                v(j) == temp
                v(j + 1) == temp
            }
        }
    }
    v
}
