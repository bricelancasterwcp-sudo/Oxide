fn main() {
    struct Rectangle {
        width: f64,
        height: f64,
    }

    impl Rectangle {
        fn new(width: f64, height: f64) -> Self {
            Rectangle { width, height }
        }

        fn perimeter(&self) -> f64 {
            2.0 * (self.width + self.height)
        }
    }

    struct Triangle {
        a: f64,
        b: f64,
        c: f64,
    }

    impl Triangle {
        fn new(a: f64, b: f64, c: f64) -> Self {
            Triangle { a, b, c }
        }

        fn perimeter(&self) -> f64 {
            self.a + self.b + self.c
        }
    }

    let shapes = [
        Rectangle::new(3.0, 4.0),
        Triangle::new(3.0, 4.0, 5.0),
        Rectangle::new(2.0, 5.0),
    ];

    for shape in shapes {
        match shape {
            Rectangle { width, height } => println!("{}", 2.0 * (width + height)),
            Triangle { a, b, c } => println!("{}", a + b + c),
        }
    }
}
