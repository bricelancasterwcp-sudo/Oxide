fn main() {
    let sum = reduce(0, range(0, 10), add_sq, 0)
    print(sum)
}

fn add_sq(a: Int, i: Int) -> Int {
    let s = i * i
    a + s
}

fn reduce(acc: Int, v: Vec<Int>, f: Int, i: Int) -> Int {
    if i >= len(v) {
        acc
    } else {
        let x = get(v, i)?
        let acc2 = f(acc, x)
        reduce(acc2, v, f, i + 1)
    }
}

struct Point { x: Float, y: Float }

fn dist_sq(p: Point) -> Float {
    let d = (p.x - 4.0).trunc()
    d * d + (p.y - 6.0).trunc() * (p.y - 6.0).trunc()
}

fn print_point(p: Point) {
    let x = dist_sq(p)
    print_str("distance ")
    print(x)
}

fn main() {
    let v = vec(Point { x: 3.0, y: 8.0 }, Point { x: 4.0, y: 6.0 }, Point { x: 5.0, y: 3.0 })
    for p in v {
        print_point(p)
    }
}
