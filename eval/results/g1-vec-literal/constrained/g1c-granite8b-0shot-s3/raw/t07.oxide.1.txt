fn main() {
    let mut v = vec()
    for i in 1.step_to(8) {
        v = push(v, 3 * i)
    }
    let sum = reduce(v, 0, add)
    print(sum)
}

fn reduce(v: Vec<Int>, init: Int, op: Int, x: Int) -> Int {
    if v.is_empty() {
        init
    } else {
        op(reduce(v.tail(), init, op), v.head())
    }
}

fn add(a: Int, b: Int) -> Int {
    a + b
}

struct Vec { head: Int, tail: Vec }

fn vec() -> Vec {
    Vec(0, Vec())
}

fn push(v: Vec, x: Int) -> Vec {
    Vec(x, v)
}

fn is_empty(v: Vec) -> Bool {
    if v.tail.is_empty() {
        true
    } else {
        false
    }
}

fn tail(v: Vec) -> Vec {
    v.tail
}

fn head(v: Vec) -> Int {
    v.head
}

fn step_to(v: Int, n: Int) -> Vec {
    if v == n {
        push(vec(), v)
    } else {
        push(step_to(v + 1, n), v)
    }
}

    if i == p {
        return
    } else {
        let c = get(s, i)?

    }
}
