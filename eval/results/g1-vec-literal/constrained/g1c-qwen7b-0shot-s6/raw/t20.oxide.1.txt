struct Item {
    name: Str,
    unit_price: Int,
    quantity: Int,
}

fn main() {
    let items = vec(vec(Item { name: "pen", unit_price: 3, quantity: 4 }), vec(Item { name: "pad", unit_price: 9, quantity: 2 }), vec(Item { name: "ink", unit_price: 5, quantity: 3 }))
    let mut total_cost = 0
    let mut highest_cost_item = ""
    let mut highest_cost = 0
    for item in items {
        let cost = item.unit_price * item.quantity
        total_cost = total_cost + cost
        if cost > highest_cost {
            highest_cost = cost
            highest_cost_item = item.name
        }
    }
    print(total_cost)
    print_str(highest_cost_item)
}
