    let a = div(100, 5)
    print(&a)
    drop a
    let b = div(100, 0)
    if b.is_none() {
        print_str("undefined")
    } else {
        print(&b)
    }
    drop b
    let c = div(100, 4)
    print(&c)
    drop c
}

fn div(a: Int, b: Int) -> Option<Int> {
    if b == 0 {
        None
    } else {
        Some(a / b)
    }
}
